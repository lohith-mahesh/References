def short_name(column):
    value = str(column)

    if prefix and value.startswith(prefix):
        return value[len(prefix):]

    return value

def split_channel(column):
    name = short_name(column)

    if name.startswith("AC_"):
        return "AC", name[3:]

    if name.startswith("BC_"):
        return "BC", name[3:]

    return "", name

def robust_center_scale(values):
    x = np.asarray(values, dtype=float)
    x = x[np.isfinite(x)]

    if x.size == 0:
        return np.nan, np.nan

    center = float(np.median(x))

    mad = float(
        np.median(
            np.abs(x - center)
        )
    )

    scale = 1.4826 * mad

    if (
        not np.isfinite(scale)
        or scale <= AD_EPS
    ):
        q25, q75 = np.percentile(
            x,
            [25, 75]
        )

        scale = float(
            (q75 - q25) / 1.349
        )

    if (
        not np.isfinite(scale)
        or scale <= AD_EPS
    ):
        scale = float(
            np.std(x)
        )

    if (
        not np.isfinite(scale)
        or scale <= AD_EPS
    ):
        scale = AD_EPS

    return center, scale

def normal_p(z):
    if not np.isfinite(z):
        return 1.0

    return float(
        math.erfc(
            abs(float(z))
            / math.sqrt(2.0)
        )
    )

def percent_deviation(
    value,
    baseline
):
    if (
        not np.isfinite(value)
        or not np.isfinite(baseline)
        or abs(baseline) <= AD_EPS
    ):
        return np.nan

    return (
        100.0
        * (value - baseline)
        / abs(baseline)
    )

def linear_slope(values):
    y = np.asarray(
        values,
        dtype=float
    )

    mask = np.isfinite(y)

    if mask.sum() < 3:
        return np.nan

    y = y[mask]

    x = np.arange(
        len(y),
        dtype=float
    )

    x = x - x.mean()
    y = y - y.mean()

    denominator = float(
        np.dot(x, x)
    )

    if denominator <= AD_EPS:
        return np.nan

    return float(
        np.dot(x, y)
        / denominator
    )

def median_pairwise_slope(values):
    y = np.asarray(
        values,
        dtype=float
    )

    y = y[
        np.isfinite(y)
    ]

    n = len(y)

    if n < 3:
        return np.nan

    slopes = []

    for i in range(n - 1):
        for j in range(i + 1, n):
            slopes.append(
                (y[j] - y[i])
                / (j - i)
            )

    if not slopes:
        return np.nan

    return float(
        np.median(slopes)
    )

def directional_fraction(values):
    y = np.asarray(
        values,
        dtype=float
    )

    y = y[
        np.isfinite(y)
    ]

    if len(y) < 3:
        return 0.0, 0

    differences = np.diff(y)

    positive = np.sum(
        differences > 0
    )

    negative = np.sum(
        differences < 0
    )

    total = max(
        positive + negative,
        1
    )

    if positive >= negative:
        return (
            positive / total,
            1
        )

    return (
        negative / total,
        -1
    )

series_count = (
    feature_df[
        [
            "Parameter",
            "FlightPhase",
            "Feature"
        ]
    ]
    .drop_duplicates()
    .shape[0]
)

abrupt_alpha_per_series = (
    AD_ABRUPT_ALPHA
    / max(
        series_count,
        1
    )
)

detection_records = []

for (
    parameter,
    phase,
    feature
), series in feature_df.groupby(
    [
        "Parameter",
        "FlightPhase",
        "Feature"
    ],
    sort=False
):

    series = (
        series
        .sort_values(
            "FlightIndex",
            kind="stable"
        )
        .reset_index(
            drop=True
        )
    )

    if (
        len(series)
        < (
            AD_MIN_BASELINE_FLIGHTS
            + AD_MIN_SEQUENCE_POINTS
        )
    ):
        continue

    baseline_count = min(
        AD_BASELINE_FLIGHTS,
        max(
            AD_MIN_BASELINE_FLIGHTS,
            len(series)
            - AD_MIN_SEQUENCE_POINTS
        )
    )

    baseline_values = (
        series
        .iloc[
            :baseline_count
        ]["Value"]
        .to_numpy(
            dtype=float
        )
    )

    baseline_center, baseline_scale = (
        robust_center_scale(
            baseline_values
        )
    )

    if (
        not np.isfinite(
            baseline_center
        )
        or not np.isfinite(
            baseline_scale
        )
    ):
        continue

    values = (
        series["Value"]
        .to_numpy(
            dtype=float
        )
    )

    z_values = (
        (
            values
            - baseline_center
        )
        / max(
            baseline_scale,
            AD_EPS
        )
    )

    z_values = np.clip(
        z_values,
        -100.0,
        100.0
    )

    positive_cusum = 0.0
    negative_cusum = 0.0
    ewma = 0.0

    gradual_candidate_count = 0
    gradual_candidate_direction = 0

    for i in range(
        baseline_count,
        len(series)
    ):

        row = series.iloc[i]

        z = float(
            z_values[i]
        )

        ewma = (
            AD_EWMA_LAMBDA
            * z
            + (
                1.0
                - AD_EWMA_LAMBDA
            )
            * ewma
        )

        positive_cusum = max(
            0.0,
            positive_cusum
            + z
            - AD_CUSUM_K
        )

        negative_cusum = min(
            0.0,
            negative_cusum
            + z
            + AD_CUSUM_K
        )

        cusum_strength = max(
            positive_cusum,
            abs(
                negative_cusum
            )
        )

        cusum_direction = (
            1
            if (
                positive_cusum
                >= abs(
                    negative_cusum
                )
            )
            else -1
        )

        p_value = (
            normal_p(
                z
            )
        )

        abrupt = (
            abs(z)
            >= AD_ABRUPT_Z
            and p_value
            <= abrupt_alpha_per_series
        )

        recent_shift_start = max(
            baseline_count,
            i
            - AD_SHIFT_PERSISTENCE
            + 1
        )

        recent_shift = (
            z_values[
                recent_shift_start:
                i + 1
            ]
        )

        persistent_shift = False

        if (
            len(recent_shift)
            >= AD_SHIFT_PERSISTENCE
        ):
            signs = np.sign(
                recent_shift
            )

            persistent_shift = (
                np.all(
                    np.abs(
                        recent_shift
                    )
                    >= AD_SHIFT_Z
                )
                and (
                    np.all(
                        signs > 0
                    )
                    or np.all(
                        signs < 0
                    )
                )
            )

        gradual = False
        robust_slope = np.nan
        direction_fraction = 0.0
        net_z = np.nan
        gradual_direction = 0

        if (
            i
            - baseline_count
            + 1
            >= AD_GRADUAL_WINDOW
        ):

            recent_z = (
                z_values[
                    i
                    - AD_GRADUAL_WINDOW
                    + 1:
                    i + 1
                ]
            )

            robust_slope = (
                median_pairwise_slope(
                    recent_z
                )
            )

            (
                direction_fraction,
                gradual_direction
            ) = directional_fraction(
                recent_z
            )

            net_z = float(
                recent_z[-1]
                - recent_z[0]
            )

            direction_consistent = (
                gradual_direction
                != 0
                and np.sign(
                    net_z
                )
                == gradual_direction
                and np.sign(
                    robust_slope
                )
                == gradual_direction
                and cusum_direction
                == gradual_direction
            )

            gradual = (
                direction_consistent
                and direction_fraction
                >= AD_GRADUAL_MIN_DIRECTIONAL_FRACTION
                and abs(
                    net_z
                )
                >= AD_GRADUAL_MIN_NET_Z
                and abs(
                    robust_slope
                )
                >= AD_GRADUAL_MIN_SLOPE_Z
                and cusum_strength
                >= AD_GRADUAL_MIN_CUSUM
                and abs(
                    ewma
                )
                >= AD_GRADUAL_MIN_EWMA
            )

        if gradual:
            if (
                gradual_candidate_direction
                == gradual_direction
            ):
                gradual_candidate_count += 1
            else:
                gradual_candidate_direction = (
                    gradual_direction
                )
                gradual_candidate_count = 1
        else:
            gradual_candidate_count = 0
            gradual_candidate_direction = 0

        confirmed_gradual = (
            gradual_candidate_count
            >= AD_CONFIRMATION_POINTS
        )

        detection_type = None

        if abrupt:
            detection_type = (
                "ABRUPT EXTREME"
            )

        elif persistent_shift:
            detection_type = (
                "PERSISTENT LEVEL SHIFT"
            )

        elif confirmed_gradual:
            detection_type = (
                "GRADUAL DEGRADATION"
            )

        if detection_type is None:
            continue

        value = float(
            row["Value"]
        )

        pct = percent_deviation(
            value,
            baseline_center
        )

        evidence = []

        evidence.append(
            f"same-phase robust deviation "
            f"{z:+.2f}σ"
        )

        if np.isfinite(
            pct
        ):
            evidence.append(
                f"value deviation "
                f"{pct:+.3f}%"
            )

        if abrupt:
            evidence.append(
                f"extreme-event p="
                f"{p_value:.2e}"
            )

        if persistent_shift:
            evidence.append(
                f"{AD_SHIFT_PERSISTENCE} "
                f"consecutive same-direction "
                f"large deviations"
            )

        if confirmed_gradual:
            evidence.append(
                f"{AD_GRADUAL_WINDOW}-point "
                f"robust slope "
                f"{robust_slope:+.3f}σ/flight"
            )

            evidence.append(
                f"directional consistency "
                f"{direction_fraction:.0%}"
            )

            evidence.append(
                f"net movement "
                f"{net_z:+.2f}σ"
            )

            evidence.append(
                f"EWMA "
                f"{ewma:+.2f}σ"
            )

            evidence.append(
                f"CUSUM "
                f"{cusum_strength:.2f}"
            )

        if detection_type == "ABRUPT EXTREME":
            confidence = "VERY HIGH"

        elif (
            detection_type
            == "PERSISTENT LEVEL SHIFT"
            and abs(z) >= 5.0
        ):
            confidence = "VERY HIGH"

        elif (
            detection_type
            == "GRADUAL DEGRADATION"
            and direction_fraction >= 0.85
            and abs(net_z) >= 3.0
            and cusum_strength >= 12.0
        ):
            confidence = "VERY HIGH"

        else:
            confidence = "HIGH"

        detection_records.append(
            {
                "FlightStart":
                    row[
                        "FlightStart"
                    ],
                "FlightIndex":
                    int(
                        row[
                            "FlightIndex"
                        ]
                    ),
                "FlightPhase":
                    row[
                        "FlightPhase"
                    ],
                "Parameter":
                    parameter,
                "Feature":
                    feature,
                "DetectionType":
                    detection_type,
                "Confidence":
                    confidence,
                "Observed":
                    value,
                "Baseline":
                    baseline_center,
                "DeviationPct":
                    pct,
                "RobustZ":
                    z,
                "EWMA":
                    ewma,
                "CUSUM":
                    cusum_strength,
                "RobustSlope":
                    robust_slope,
                "DirectionalConsistency":
                    direction_fraction,
                "NetMovementZ":
                    net_z,
                "Reason":
                    "; ".join(
                        evidence
                    )
            }
        )

    def pretty_parameter(

    report_df = pd.DataFrame(

            "PHASE":

                    "FlightPhase"
                ].map(
                    lambda x:
                    f"{x:g}"
                ),

            "PARAMETER":
                anomaly_detail_df[
                    "Parameter"
                ].map(
                    pretty_parameter
                ),

            "DETECTION":
                anomaly_detail_df[
                    "DetectionType"
                ],

            "CONFIDENCE":
                anomaly_detail_df[
                    "Confidence"
                ],

            "FEATURE":
                anomaly_detail_df[
                    "Feature"
                ],

            "OBSERVED":
                anomaly_detail_df[
                    "Observed"
                ].map(
                    lambda x:
                    f"{x:,.6g}"
                ),

            "BASELINE":
                anomaly_detail_df[
                    "Baseline"
                ].map(
