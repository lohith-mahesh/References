    "minimum_robust_z": 6.0,
    "minimum_rate_z": 8.0,
    "catastrophic_z": 20.0,
    "episode_gap_seconds": 5,
    "minimum_episode_seconds": 2,

def _ead_col(name):
    return F.col(f"`{name.replace('`', '``')}`")


def _ead_clean_double(column):
    value = column.cast("double")
    return F.when(
        value.isNull() | F.isnan(value),
        F.lit(None).cast("double"),
    ).otherwise(value)


def _ead_short(name):
    return (
        name[len(PARAMETER_PREFIX):]
        if name.startswith(PARAMETER_PREFIX)
        else name
    )


def _ead_channel_signal(name):
    short = _ead_short(name)
    for channel in CHANNELS:
        prefix = f"{channel}_"
        if short.startswith(prefix):
            return channel, short[len(prefix):]
    return "", short


def _ead_quantiles(value):
    if (
        value is None
        or len(value) != 5
        or any(item is None for item in value)
    ):
        return None
    return tuple(float(item) for item in value)


def _ead_scale(q25, median, q75):
    return max(
        (q75 - q25) / 1.349,
        abs(median) * 1.0e-9,
        1.0e-12,
    )


def _ead_case(phase_expressions):
    expression = F.lit(None).cast("double")
    for phase, value in phase_expressions:
        expression = F.when(
            F.col("_phase") == F.lit(phase),
            value,
        ).otherwise(expression)
    return expression


def _ead_bh(values):
    if not values:
        return np.array([], dtype=float)
    p_values = np.asarray(values, dtype=float)
    order = np.argsort(p_values)
    ranked = p_values[order]
    adjusted = ranked * len(ranked) / np.arange(
        1,
        len(ranked) + 1,
    )
    adjusted = np.minimum.accumulate(adjusted[::-1])[::-1]
    result = np.empty_like(adjusted)
    result[order] = np.minimum(adjusted, 1.0)
    return result


def _ead_prepare(columns, include_rates=True):
    projections = [
        _ead_col("Timestamp").alias("Timestamp"),
        _ead_col("StartDatetime").alias("StartDatetime"),
        _ead_clean_double(
            _ead_col(FLIGHT_PHASE_COLUMNS["AC"])
        ).alias("_phase_ac"),
        _ead_clean_double(
            _ead_col(FLIGHT_PHASE_COLUMNS["BC"])
        ).alias("_phase_bc"),
    ]

    aliases = {}

    for index, name in enumerate(columns):
        alias = f"_v{index:03d}"
        aliases[name] = alias
        projections.append(
            _ead_clean_double(_ead_col(name)).alias(alias)
        )

    frame = cda.select(*projections).where(
        F.col("Timestamp").isNotNull()
        & F.col("StartDatetime").isNotNull()
    )

    frame = (
        frame.withColumn(
            "_phase_disagreement",
            F.col("_phase_ac").isNotNull()
            & F.col("_phase_bc").isNotNull()
            & (F.col("_phase_ac") != F.col("_phase_bc")),
        )
        .withColumn(
            "_phase",
            F.when(
                ~F.col("_phase_disagreement"),
                F.coalesce("_phase_ac", "_phase_bc"),
            ),
        )
        .withColumn(
            "_time_us",
            (
                F.col("Timestamp").cast("double")
                * F.lit(1_000_000)
            ).cast("long"),
        )
    )

    order_window = (
        Window.partitionBy("StartDatetime")
        .orderBy("_time_us")
    )

    run_window = order_window.rowsBetween(
        Window.unboundedPreceding,
        Window.currentRow,
    )

    previous_phase = F.lag("_phase").over(order_window)

    frame = (
        frame.withColumn(
            "_phase_change",
            F.when(
                ~F.col("_phase").eqNullSafe(previous_phase),
                1,
            ).otherwise(0),
        )
        .withColumn(
            "_phase_run",
            F.sum("_phase_change").over(run_window),
        )
    )

    boundary_us = int(
        ANOMALY_CONFIG["transition_boundary_seconds"]
        * 1_000_000
    )

    boundary_window = (
        Window.partitionBy("StartDatetime")
        .orderBy("_time_us")
        .rangeBetween(-boundary_us, boundary_us)
    )

    frame = (
        frame.withColumn(
            "_near_phase_boundary",
            F.max("_phase_change").over(boundary_window) > 0,
        )
        .withColumn(
            "_eligible",
            F.col("_phase").isNotNull()
            & ~F.col("_phase_disagreement")
            & ~F.col("_near_phase_boundary"),
        )
    )

    if include_rates:
        rate_window = (
            Window.partitionBy(
                "StartDatetime",
                "_phase_run",
            )
            .orderBy("_time_us")
        )

        frame = frame.withColumn(
            "_dt",
            (
                F.col("_time_us")
                - F.lag("_time_us").over(rate_window)
            )
            / F.lit(1_000_000.0),
        )

        for alias in aliases.values():
            frame = frame.withColumn(
                f"{alias}_rate",
                F.when(
                    (F.col("_dt") > 0)
                    & (
                        F.col("_dt")
                        <= ANOMALY_CONFIG[
                            "maximum_rate_gap_seconds"
                        ]
                    ),
                    (
                        F.col(alias)
                        - F.lag(alias).over(rate_window)
                    )
                    / F.col("_dt"),
                ),
            )

    return frame, aliases


def _ead_stats(frame, aliases):
    tail = ANOMALY_CONFIG["tail_probability"]
    probabilities = [
        tail,
        0.25,
        0.5,
        0.75,
        1.0 - tail,
    ]

    expressions = []

    for index, alias in enumerate(aliases.values()):
        expressions.extend(
            [
                F.count(alias).alias(f"_n{index:03d}"),
                F.approx_count_distinct(alias).alias(
                    f"_d{index:03d}"
                ),
                F.percentile_approx(
                    alias,
                    probabilities,
                    ANOMALY_CONFIG["quantile_accuracy"],
                ).alias(f"_q{index:03d}"),
                F.count(f"{alias}_rate").alias(
                    f"_rn{index:03d}"
                ),
                F.approx_count_distinct(
                    f"{alias}_rate"
                ).alias(f"_rd{index:03d}"),
                F.percentile_approx(
                    f"{alias}_rate",
                    probabilities,
                    ANOMALY_CONFIG["quantile_accuracy"],
                ).alias(f"_rq{index:03d}"),
            ]
        )

    rows = (
        frame.where("_eligible")
        .groupBy("_phase")
        .agg(*expressions)
        .collect()
    )

    lookup = {name: {} for name in aliases}

    for row in rows:
        values = row.asDict(recursive=True)
        phase = float(values["_phase"])

        for index, name in enumerate(aliases):
            value_q = _ead_quantiles(
                values.get(f"_q{index:03d}")
            )
            rate_q = _ead_quantiles(
                values.get(f"_rq{index:03d}")
            )

            lookup[name][phase] = {
                "count": int(
                    values.get(f"_n{index:03d}") or 0
                ),
                "distinct": int(
                    values.get(f"_d{index:03d}") or 0
                ),
                "q": value_q,
                "rate_count": int(
                    values.get(f"_rn{index:03d}") or 0
                ),
                "rate_distinct": int(
                    values.get(f"_rd{index:03d}") or 0
                ),
                "rate_q": rate_q,
            }

    return lookup


def _ead_metric_expressions(alias, parameter_stats):
    amplitude_z_cases = []
    rate_z_cases = []
    amplitude_tail_cases = []
    rate_tail_cases = []
    amplitude_continuous_cases = []
    rate_continuous_cases = []

    value = F.col(alias)
    rate = F.col(f"{alias}_rate")

    for phase, stats in parameter_stats.items():
        if (
            stats["count"]
            >= ANOMALY_CONFIG["minimum_phase_samples"]
            and stats["q"] is not None
        ):
            lower, q25, median, q75, upper = stats["q"]
            scale = _ead_scale(q25, median, q75)

            amplitude_z_cases.append(
                (
                    phase,
                    F.abs(value - F.lit(median))
                    / F.lit(scale),
                )
            )

            amplitude_tail_cases.append(
                (
                    phase,
                    F.when(
                        (value < F.lit(lower))
                        | (value > F.lit(upper)),
                        1.0,
                    ).otherwise(0.0),
                )
            )

            amplitude_continuous_cases.append(
                (
                    phase,
                    F.lit(
                        1.0
                        if stats["distinct"] > 20
                        else 0.0
                    ),
                )
            )

        if (
            stats["rate_count"]
            >= ANOMALY_CONFIG["minimum_phase_samples"]
            and stats["rate_q"] is not None
        ):
            lower, q25, median, q75, upper = (
                stats["rate_q"]
            )
            scale = _ead_scale(q25, median, q75)

            rate_z_cases.append(
                (
                    phase,
                    F.abs(rate - F.lit(median))
                    / F.lit(scale),
                )
            )

            rate_tail_cases.append(
                (
                    phase,
                    F.when(
                        (rate < F.lit(lower))
                        | (rate > F.lit(upper)),
                        1.0,
                    ).otherwise(0.0),
                )
            )

            rate_continuous_cases.append(
                (
                    phase,
                    F.lit(
                        1.0
                        if stats["rate_distinct"] > 20
                        else 0.0
                    ),
                )
            )

    amplitude_z = F.coalesce(
        _ead_case(amplitude_z_cases),
        F.lit(0.0),
    )

    rate_z = F.coalesce(
        _ead_case(rate_z_cases),
        F.lit(0.0),
    )

    amplitude_tail = (
        F.coalesce(
            _ead_case(amplitude_tail_cases),
            F.lit(0.0),
        )
        > 0
    )

    rate_tail = (
        F.coalesce(
            _ead_case(rate_tail_cases),
            F.lit(0.0),
        )
        > 0
    )

    amplitude_continuous = (
        F.coalesce(
            _ead_case(amplitude_continuous_cases),
            F.lit(0.0),
        )
        > 0
    )

    rate_continuous = (
        F.coalesce(
            _ead_case(rate_continuous_cases),
            F.lit(0.0),
        )
        > 0
    )

    amplitude_flag = (
        amplitude_tail
        & (
            amplitude_z
            >= ANOMALY_CONFIG["minimum_robust_z"]
        )
    )

    rate_flag = (
        rate_tail
        & (
            rate_z
            >= ANOMALY_CONFIG["minimum_rate_z"]
        )
    )

    catastrophic = (
        amplitude_continuous
        & (
            amplitude_z
            >= ANOMALY_CONFIG["catastrophic_z"]
        )
    ) | (
        rate_continuous
        & (
            rate_z
            >= ANOMALY_CONFIG["catastrophic_z"]
        )
    )

    flag = (
        value.isNotNull()
        & (
            amplitude_flag
            | rate_flag
            | catastrophic
        )
    )

    anomaly_type = F.concat_ws(
        "+",
        F.when(
            amplitude_flag,
            F.lit("PHASE_CONDITIONAL_TAIL"),
        ),
        F.when(
            rate_flag,
            F.lit("TEMPORAL_JUMP"),
        ),
        F.when(
            catastrophic,
            F.lit("EXTREME_ROBUST_DEVIATION"),
        ),
    )

    score = F.least(
        F.greatest(amplitude_z, rate_z),
        F.lit(100.0),
    )

    return flag, score, anomaly_type, amplitude_z


def _ead_score_batch(frame, aliases, batch_stats):
    events = []

    for name, alias in aliases.items():
        channel, signal = _ead_channel_signal(name)

        flag, score, anomaly_type, _ = (
            _ead_metric_expressions(
                alias,
                batch_stats[name],
            )
        )

        events.append(
            F.when(
                F.col("_eligible") & flag,
                F.struct(
                    F.lit(name).alias("Parameter"),
                    F.lit(signal).alias("Signal"),
                    F.lit(channel).alias("Channel"),
                    F.col(alias).alias("Value"),
                    F.lit(None)
                    .cast("string")
                    .alias("PeerParameter"),
                    F.lit(None)
                    .cast("double")
                    .alias("PeerValue"),
                    score.alias("Score"),
                    anomaly_type.alias("AnomalyType"),
                    F.lit(1.0).alias(
                        "AttributionConfidence"
                    ),
                ),
            )
        )

    return (
        frame.select(
            "Timestamp",
            "StartDatetime",
            F.col("_phase").alias("FlightPhase"),
            F.explode(
                F.array(*events)
            ).alias("_event"),
        )
        .where(F.col("_event").isNotNull())
        .select(
            "Timestamp",
            "StartDatetime",
            "FlightPhase",
            "_event.*",
        )
    )


def _ead_trends(frame, aliases):
    expressions = [
        F.percentile_approx(
            alias,
            0.5,
            ANOMALY_CONFIG["quantile_accuracy"],
        ).alias(alias)
        for alias in aliases.values()
    ]

    rows = (
        frame.where("_eligible")
        .groupBy("StartDatetime", "_phase")
        .agg(*expressions)
        .orderBy("StartDatetime")
        .collect()
    )

    if not rows:
        return []

    wide = pd.DataFrame(
        [row.asDict(recursive=True) for row in rows]
    )

    wide["StartDatetime"] = pd.to_datetime(
        wide["StartDatetime"],
        errors="coerce",
    )

    findings = []

    for name, alias in aliases.items():
        usable = wide[
            ["StartDatetime", "_phase", alias]
        ].dropna()

        for phase, group in usable.groupby("_phase"):
            group = (
                group.sort_values("StartDatetime")
                .drop_duplicates("StartDatetime")
            )

            if (
                len(group)
                < ANOMALY_CONFIG["minimum_trend_flights"]
            ):
                continue

            times = group["StartDatetime"]

            x = (
                times - times.iloc[0]
            ).dt.total_seconds().to_numpy(
                dtype=float
            ) / 86400.0

            y = group[alias].to_numpy(dtype=float)

            if (
                not np.isfinite(y).all()
                or x[-1] <= x[0]
            ):
                continue

            median = float(np.median(y))

            mad = (
                float(
                    np.median(
                        np.abs(y - median)
                    )
                )
                * 1.4826
            )

            q25, q75 = np.quantile(
                y,
                [0.25, 0.75],
            )

            scale = max(
                mad,
                float(q75 - q25) / 1.349,
                abs(median) * 1.0e-9,
                1.0e-12,
            )

            slope, _, _, _ = theilslopes(
                y,
                x,
                0.99,
            )

            tau_result = kendalltau(
                x,
                y,
                nan_policy="omit",
            )

            tau_value = getattr(
                tau_result,
                "statistic",
                tau_result[0],
            )

            tau = (
                float(tau_value)
                if np.isfinite(tau_value)
                else 0.0
            )

            trend_p = (
                float(tau_result.pvalue)
                if np.isfinite(tau_result.pvalue)
                else 1.0
            )

            total_change = float(
                slope * (x[-1] - x[0])
            )

            trend_effect_z = (
                abs(total_change) / scale
            )

            best_change_z = 0.0
            best_split = None

            for split in range(
                3,
                len(y) - 2,
            ):
                change_z = (
                    abs(
                        float(
                            np.median(y[split:])
                            - np.median(y[:split])
                        )
                    )
                    / scale
                )

                if change_z > best_change_z:
                    best_change_z = change_z
                    best_split = split

            change_p = min(
                1.0,
                2.0
                * norm.sf(best_change_z)
                * max(1, len(y) - 5),
            )

            combined_p = min(
                1.0,
                min(trend_p, change_p) * 2.0,
            )

            effect_qualified = (
                (
                    trend_effect_z
                    >= ANOMALY_CONFIG[
                        "minimum_trend_effect_z"
                    ]
                    and abs(tau)
                    >= ANOMALY_CONFIG[
                        "minimum_trend_tau"
                    ]
                )
                or best_change_z
                >= ANOMALY_CONFIG[
                    "minimum_trend_effect_z"
                ]
                * 1.5
            )

            channel, signal = (
                _ead_channel_signal(name)
            )

            findings.append(
                {
                    "Parameter": name,
                    "Signal": signal,
                    "Channel": channel,
                    "FlightPhase": float(phase),
                    "Flights": int(len(group)),
                    "FirstFlight": times.iloc[0],
                    "LastFlight": times.iloc[-1],
                    "ChangePointFlight": (
                        times.iloc[best_split]
                        if best_split is not None
                        else pd.NaT
                    ),
                    "TheilSenSlopePerDay": float(
                        slope
                    ),
                    "EstimatedWindowChange": (
                        total_change
                    ),
                    "TrendEffectZ": float(
                        trend_effect_z
                    ),
                    "KendallTau": tau,
                    "ChangePointZ": float(
                        best_change_z
                    ),
                    "PValue": combined_p,
                    "EffectQualified": bool(
                        effect_qualified
                    ),
                }
            )

    return findings


numeric_parameters = [
    field.name
    for field in cda.schema.fields
    if field.name.startswith(PARAMETER_PREFIX)
    and isinstance(
        field.dataType,
        (NumericType, BooleanType),
    )
    and field.name
    not in set(FLIGHT_PHASE_COLUMNS.values())
]

if not numeric_parameters:
    raise RuntimeError(
        "No eligible numeric or Boolean engine "
        "parameters were found in cda."
    )

cadence_window = (
    Window.partitionBy("StartDatetime")
    .orderBy("Timestamp")
)

cadence_row = (
    cda.select(
        (
            _ead_col("Timestamp").cast("double")
            - F.lag(
                _ead_col("Timestamp").cast("double")
            ).over(cadence_window)
        ).alias("_sample_interval")
    )
    .where(
        (F.col("_sample_interval") > 0)
        & (F.col("_sample_interval") <= 60)
    )
    .agg(
        F.percentile_approx(
            "_sample_interval",
            0.5,
            10000,
        ).alias("_median_interval")
    )
    .first()
)

observed_cadence_seconds = float(
    cadence_row._median_interval or 1.0
)

ANOMALY_CONFIG[
    "observed_cadence_seconds"
] = observed_cadence_seconds

ANOMALY_CONFIG["episode_gap_seconds"] = min(
    60,
    max(
        5,
        int(
            math.ceil(
                3.0 * observed_cadence_seconds
            )
        ),
    ),
)

print(
    f"Enterprise anomaly analysis: "

    f"{observed_cadence_seconds:.6g}s | "

    f"{ANOMALY_CONFIG['episode_gap_seconds']}s"
)

event_frames = []
trend_records = []
parameter_stats = {}

for batch_start in range(
    0,

    batch_columns = numeric_parameters[

    batch_frame, batch_aliases = (

        )
    )

    batch_frame = batch_frame.persist(

    )

    batch_stats = _ead_stats(

    parameter_stats.update(batch_stats)

    event_frame = _ead_score_batch(

    event_frame.count()
    event_frames.append(event_frame)

    trend_records.extend(
        _ead_trends(
            batch_frame,
            batch_aliases,
        )
    )

    batch_frame.unpersist()

    completed = min(
        batch_start
        + ANOMALY_CONFIG["batch_size"],
        len(numeric_parameters),
    )

    print(
        f"Profiled and scored "
        f"{completed:,} of "
        f"{len(numeric_parameters):,} parameters"
    )

    difference_aliases = {}

    for index, (signal, members) in enumerate(

        pair_frame = pair_frame.withColumn(

    tail = ANOMALY_CONFIG[

    probabilities = [
        tail,

    aggregate_expressions = []

    for index, alias in enumerate(
        difference_aliases.values()
    ):
        aggregate_expressions.extend(
            [
                F.count(alias).alias(
                    f"_pn{index:03d}"
                ),
                F.percentile_approx(
                    alias,
                    probabilities,
                    ANOMALY_CONFIG[
                        "quantile_accuracy"
                    ],
                ).alias(
                    f"_pq{index:03d}"
                ),
            ]
        )

    pair_rows = (
        pair_frame.where("_eligible")
        .groupBy("_phase")
        .agg(*aggregate_expressions)
        .collect()
    )

    pair_stats = {
        signal: {}
        for signal in difference_aliases
    }

    for row in pair_rows:
        values = row.asDict(recursive=True)
        phase = float(values["_phase"])

        for index, signal in enumerate(
            difference_aliases
        ):
            pair_stats[signal][phase] = {
                "count": int(
                    values.get(
                        f"_pn{index:03d}"
                    )
                    or 0
                ),
                "q": _ead_quantiles(
                    values.get(
                        f"_pq{index:03d}"
                    )
                ),
            }

    pair_structs = []

    for signal, members in selected_pairs:
        difference_alias = (
            difference_aliases[signal]
        )

        pair_z_cases = []
        pair_tail_cases = []

        for phase, stats in (
            pair_stats[signal].items()
        ):
            if (
                stats["count"]
                < ANOMALY_CONFIG[
                    "minimum_phase_samples"
                ]
                or stats["q"] is None
            ):
                continue

            lower, q25, median, q75, upper = (
                stats["q"]
            )

            scale = _ead_scale(
                q25,
                median,
                q75,
            )

            pair_z_cases.append(
                (
                    phase,
                    F.abs(
                        F.col(difference_alias)
                        - F.lit(median)
                    )
                    / F.lit(scale),
                )
            )

            pair_tail_cases.append(
                (
                    phase,
                    F.when(
                        (
                            F.col(
                                difference_alias
                            )
                            < F.lit(lower)
                        )
                        | (
                            F.col(
                                difference_alias
                            )
                            > F.lit(upper)
                        ),
                        1.0,
                    ).otherwise(0.0),
                )
            )

        pair_z = F.coalesce(
            _ead_case(pair_z_cases),
            F.lit(0.0),
        )

        pair_tail = (
            F.coalesce(
                _ead_case(pair_tail_cases),
                F.lit(0.0),
            )
            > 0
        )

        ac_alias = pair_aliases[
            members["AC"]
        ]

        bc_alias = pair_aliases[
            members["BC"]
        ]

        _, _, _, ac_z = (
            _ead_metric_expressions(
                ac_alias,
                parameter_stats[
                    members["AC"]
                ],
            )
        )

        _, _, _, bc_z = (
            _ead_metric_expressions(
                bc_alias,
                parameter_stats[
                    members["BC"]
                ],
            )
        )

        ac_wins = (
            ac_z
            >= F.greatest(
                bc_z * F.lit(1.5),
                F.lit(
                    ANOMALY_CONFIG[
                        "minimum_robust_z"
                    ]
                ),
            )
        )

        bc_wins = (
            bc_z
            >= F.greatest(
                ac_z * F.lit(1.5),
                F.lit(
                    ANOMALY_CONFIG[
                        "minimum_robust_z"
                    ]
                ),
            )
        )

        attributed_parameter = (
            F.when(
                ac_wins,
                F.lit(members["AC"]),
            )
            .when(
                bc_wins,
                F.lit(members["BC"]),
            )
            .otherwise(
                F.lit(
                    f"{members['AC']} | "
                    f"{members['BC']}"
                )
            )
        )

        attributed_channel = (
            F.when(
                ac_wins,
                F.lit("AC"),
            )
            .when(
                bc_wins,
                F.lit("BC"),
            )
            .otherwise(
                F.lit("AC|BC")
            )
        )

        attributed_value = (
            F.when(
                ac_wins,
                F.col(ac_alias),
            )
            .when(
                bc_wins,
                F.col(bc_alias),
            )
        )

        peer_parameter = (
            F.when(
                ac_wins,
                F.lit(members["BC"]),
            )
            .when(
                bc_wins,
                F.lit(members["AC"]),
            )
        )

        peer_value = (
            F.when(
                ac_wins,
                F.col(bc_alias),
            )
            .when(
                bc_wins,
                F.col(ac_alias),
            )
        )

        confidence = (
            F.when(
                ac_wins | bc_wins,
                F.greatest(ac_z, bc_z)
                / (
                    ac_z
                    + bc_z
                    + F.lit(1.0e-12)
                ),
            )
            .otherwise(F.lit(0.5))
        )

        pair_flag = (
            F.col(
                difference_alias
            ).isNotNull()
            & (
                (
                    pair_tail
                    & (
                        pair_z
                        >= ANOMALY_CONFIG[
                            "minimum_robust_z"
                        ]
                    )
                )
                | (
                    pair_z
                    >= ANOMALY_CONFIG[
                        "catastrophic_z"
                    ]
                )
            )
        )

        pair_structs.append(
            F.when(
                F.col("_eligible")
                & pair_flag,
                F.struct(
                    attributed_parameter.alias(
                        "Parameter"
                    ),
                    F.lit(signal).alias(
                        "Signal"
                    ),
                    attributed_channel.alias(
                        "Channel"
                    ),
                    attributed_value.cast(
                        "double"
                    ).alias("Value"),
                    peer_parameter.alias(
                        "PeerParameter"
                    ),
                    peer_value.cast(
                        "double"
                    ).alias("PeerValue"),
                    F.least(
                        pair_z,
                        F.lit(100.0),
                    ).alias("Score"),
                    F.lit(
                        "AC_BC_CHANNEL_DIVERGENCE"
                    ).alias("AnomalyType"),
                    confidence.alias(
                        "AttributionConfidence"
                    ),
                ),
            )
        )

    if pair_structs:
        pair_event_frame = (
            pair_frame.select(
                "Timestamp",
                "StartDatetime",
                F.col("_phase").alias(
                    "FlightPhase"
                ),
                F.explode(
                    F.array(*pair_structs)
                ).alias("_event"),
            )
            .where(
                F.col("_event").isNotNull()
            )
            .select(
                "Timestamp",
                "StartDatetime",
                "FlightPhase",
                "_event.*",
            )
            .persist(
                StorageLevel.DISK_ONLY
            )
        )

        pair_event_frame.count()
        pair_events.append(
            pair_event_frame
        )

paired_events = (
    reduce(
        lambda left, right: left.unionByName(
            right
        ),
        pair_events,
    )
    if pair_events
    else spark.createDataFrame(
        [],
        event_schema,
    )
)

raw_events = (
    univariate_events
    .unionByName(paired_events)
)

raw_events = (
    raw_events.groupBy(
        "Timestamp",
        "StartDatetime",
        "FlightPhase",
        "Parameter",
        "Signal",
        "Channel",
    )
    .agg(
        F.first(
            "Value",
            ignorenulls=True,
        ).alias("Value"),
        F.first(
            "PeerParameter",
            ignorenulls=True,
        ).alias("PeerParameter"),
        F.first(
            "PeerValue",
            ignorenulls=True,
        ).alias("PeerValue"),
        F.max("Score").alias("Score"),
        F.concat_ws(
            "+",
            F.sort_array(
                F.collect_set(
                    "AnomalyType"
                )
            ),
        ).alias("AnomalyType"),
        F.max(
            "AttributionConfidence"
        ).alias(
            "AttributionConfidence"
        ),
    )
)

event_window = (
    Window.partitionBy(
        "StartDatetime",
        "FlightPhase",
        "Parameter",
    )
    .orderBy("Timestamp")
)

event_run_window = (
    event_window.rowsBetween(
        Window.unboundedPreceding,
        Window.currentRow,
    )
)

events_with_runs = (
    raw_events.withColumn(
        "_new_episode",
        F.when(
            F.lag("Timestamp")
            .over(event_window)
            .isNull()
            | (
                F.col("Timestamp")
                .cast("long")
                - F.lag("Timestamp")
                .over(event_window)
                .cast("long")
                > ANOMALY_CONFIG[
                    "episode_gap_seconds"
                ]
            ),
            1,
        ).otherwise(0),
    )
    .withColumn(
        "EpisodeId",
        F.sum("_new_episode").over(
            event_run_window
        ),
    )
)

enterprise_anomaly_episodes = (
    events_with_runs.groupBy(
        "StartDatetime",
        "FlightPhase",
        "Parameter",
        "Signal",
        "Channel",
        "EpisodeId",
    )
    .agg(
        F.min("Timestamp").alias(
            "EpisodeStart"
        ),
        F.max("Timestamp").alias(
            "EpisodeEnd"
        ),
        F.count("*").alias(
            "AnomalousSamples"
        ),
        F.max("Score").alias(
            "MaxScore"
        ),
        F.avg("Score").alias(
            "MeanScore"
        ),
        F.concat_ws(
            "+",
            F.sort_array(
                F.collect_set(
                    "AnomalyType"
                )
            ),
        ).alias("AnomalyType"),
        F.max(
            "AttributionConfidence"
        ).alias(
            "AttributionConfidence"
        ),
    )
    .withColumn(
        "DurationSeconds",
        F.col("EpisodeEnd").cast(
            "double"
        )
        - F.col("EpisodeStart").cast(
            "double"
        ),
    )
    .where(
        (
            F.col("AnomalousSamples")
            >= ANOMALY_CONFIG[
                "minimum_episode_samples"
            ]
        )
        | (
            F.col("DurationSeconds")
            >= ANOMALY_CONFIG[
                "minimum_episode_seconds"
            ]
        )
        | (
            F.col("MaxScore")
            >= ANOMALY_CONFIG[
                "catastrophic_z"
            ]
        )
    )
    .withColumn(
        "AircraftIdentifier",
        F.lit(str(selected_acid)),
    )
    .withColumn(
        "EngineSerialNumber",
        F.lit(str(selected_esn)),
    )
    .persist(StorageLevel.DISK_ONLY)
)

confirmed_episode_keys = (
    enterprise_anomaly_episodes.select(
        "StartDatetime",
        "FlightPhase",
        "Parameter",
        "EpisodeId",
    )
)

enterprise_anomaly_samples = (
    events_with_runs.join(
        confirmed_episode_keys,
        [
            "StartDatetime",
            "FlightPhase",
            "Parameter",
            "EpisodeId",
        ],
        "inner",
    )
    .drop("_new_episode")
    .withColumn(
        "AircraftIdentifier",
        F.lit(str(selected_acid)),
    )
    .withColumn(
        "EngineSerialNumber",
        F.lit(str(selected_esn)),
    )
    .persist(StorageLevel.DISK_ONLY)
)

phase_frame, _ = _ead_prepare(

    include_rates=False,
)

enterprise_phase_issue_samples = (
    phase_frame.where(
        F.col("_phase_disagreement")
    )
    .select(
        "Timestamp",
        "StartDatetime",
        F.col("_phase_ac").alias(
            "ACFlightPhase"
        ),
        F.col("_phase_bc").alias(
            "BCFlightPhase"
        ),
        F.lit(
            "FLIGHT_PHASE_DISAGREEMENT"
        ).alias("AnomalyType"),
    )
)

phase_issue_window = (
    Window.partitionBy(
        "StartDatetime",
        "AnomalyType",
    )
    .orderBy("Timestamp")
)

phase_issue_run_window = (
    phase_issue_window.rowsBetween(
        Window.unboundedPreceding,
        Window.currentRow,
    )
)

enterprise_phase_issue_samples = (
    enterprise_phase_issue_samples
    .withColumn(
        "_new_episode",
        F.when(
            F.lag("Timestamp")
            .over(phase_issue_window)
            .isNull()
            | (
                F.col("Timestamp")
                .cast("long")
                - F.lag("Timestamp")
                .over(phase_issue_window)
                .cast("long")
                > ANOMALY_CONFIG[
                    "episode_gap_seconds"
                ]
            ),
            1,
        ).otherwise(0),
    )
    .withColumn(
        "EpisodeId",
        F.sum("_new_episode").over(
            phase_issue_run_window
        ),
    )
    .drop("_new_episode")
)

enterprise_phase_issue_episodes = (
    enterprise_phase_issue_samples
    .groupBy(
        "StartDatetime",
        "AnomalyType",
        "EpisodeId",
    )
    .agg(
        F.min("Timestamp").alias(
            "EpisodeStart"
        ),
        F.max("Timestamp").alias(
            "EpisodeEnd"
        ),
        F.count("*").alias(
            "Samples"
        ),
        F.collect_set(
            "ACFlightPhase"
        ).alias(
            "ACFlightPhases"
        ),
        F.collect_set(
            "BCFlightPhase"
        ).alias(
            "BCFlightPhases"
        ),
    )
    .withColumn(
        "DurationSeconds",
        F.col("EpisodeEnd").cast(
            "double"
        )
        - F.col("EpisodeStart").cast(
            "double"
        ),
    )
    .persist(StorageLevel.DISK_ONLY)
)

enterprise_anomaly_flights = (
    enterprise_anomaly_episodes
    .groupBy("StartDatetime")
    .agg(
        F.count("*").alias(
            "ConfirmedEpisodes"
        ),
        F.countDistinct(
            "Parameter"
        ).alias(
            "AnomalousParameters"
        ),
        F.collect_set(
            "Parameter"
        ).alias("Parameters"),
        F.collect_set(
            "FlightPhase"
        ).alias("FlightPhases"),
        F.min(
            "EpisodeStart"
        ).alias("FirstAnomaly"),
        F.max(
            "EpisodeEnd"
        ).alias("LastAnomaly"),
        F.max(
            "MaxScore"
        ).alias("MaximumScore"),
        F.sum(
            "AnomalousSamples"
        ).alias("AnomalousSamples"),
    )
    .join(
        enterprise_phase_issue_episodes
        .groupBy("StartDatetime")
        .agg(
            F.count("*").alias(
                "FlightPhaseIssueEpisodes"
            )
        ),
        "StartDatetime",
        "full",
    )
    .fillna(
        {
            "ConfirmedEpisodes": 0,
            "AnomalousParameters": 0,
            "AnomalousSamples": 0,
            "FlightPhaseIssueEpisodes": 0,
        }
    )
    .withColumn(
        "AircraftIdentifier",
        F.lit(str(selected_acid)),
    )
    .withColumn(
        "EngineSerialNumber",
        F.lit(str(selected_esn)),
    )
)

enterprise_parameter_summary = (
    enterprise_anomaly_episodes
    .groupBy(

        "Signal",
        "Channel",

    .agg(
        F.count("*").alias(

            "AnomalousSamples"

        F.desc("ConfirmedEpisodes"),
