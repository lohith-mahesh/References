for c in ("Timestamp", "StartDatetime"):

def qc(name):
    return F.col(f"`{str(name).replace('`', '``')}`")

def short(name):
    name = str(name)
    return name[len(PARAMETER_PREFIX):] if name.startswith(PARAMETER_PREFIX) else name

def channel_signal(name):
    name = short(name)
    if name.startswith("AC_"):
        return "AC", name[3:]
    if name.startswith("BC_"):
        return "BC", name[3:]
    return "", name

def finite_array(values):
    x = np.asarray(values, dtype=float)
    return x[np.isfinite(x)]

def robust_reference(values):
    x = finite_array(values)

    if x.size == 0:
        return np.nan, np.nan, False

    center = float(np.median(x))
    mad = float(np.median(np.abs(x - center)))
    scale = 1.4826 * mad

    if not np.isfinite(scale) or scale <= AD_EPS:
        q25, q75 = np.percentile(x, [25, 75])
        scale = float((q75 - q25) / 1.349)

    if not np.isfinite(scale) or scale <= AD_EPS:
        scale = float(np.std(x))

    constant = not np.isfinite(scale) or scale <= AD_EPS

    if constant:
        scale = AD_EPS

    return center, scale, constant

def normal_p(z):
    if not np.isfinite(z):
        return 1.0
    return float(math.erfc(abs(float(z)) / math.sqrt(2.0)))

def pct(value, baseline):
    if not np.isfinite(value) or not np.isfinite(baseline) or abs(baseline) <= AD_EPS:
        return np.nan
    return float(100.0 * (value - baseline) / abs(baseline))

def theilsen(values):
    y = finite_array(values)

    if len(y) < 3:
        return np.nan

    slopes = [(y[j] - y[i]) / (j - i)
              for i in range(len(y) - 1)
              for j in range(i + 1, len(y))]

    return float(np.median(slopes)) if slopes else np.nan

def directionality(values):
    y = finite_array(values)

    if len(y) < 3:
        return 0.0, 0

    d = np.diff(y)
    pos = int(np.sum(d > 0))
    neg = int(np.sum(d < 0))
    total = pos + neg

    if total == 0:
        return 0.0, 0

    return (pos / total, 1) if pos >= neg else (neg / total, -1)

def mann_kendall(values):
    y = finite_array(values)
    n = len(y)

    if n < 5:
        return 0.0, 1.0

    s = sum(int(np.sum(np.sign(y[i + 1:] - y[i]))) for i in range(n - 1))
    _, counts = np.unique(y, return_counts=True)

    tie = float(np.sum(counts * (counts - 1) * (2 * counts + 5)))
    var_s = (n * (n - 1) * (2 * n + 5) - tie) / 18.0

    if var_s <= 0:
        return 0.0, 1.0

    z = (s - 1) / math.sqrt(var_s) if s > 0 else (
        (s + 1) / math.sqrt(var_s) if s < 0 else 0.0
    )

    return float(z), float(math.erfc(abs(z) / math.sqrt(2.0)))

schema = {f.name: f.dataType for f in cda.schema.fields}

parameter_columns = [
    name for name, dtype in schema.items()
    if name.startswith(PARAMETER_PREFIX) and isinstance(dtype, (NumericType, BooleanType))
]

if not parameter_columns:
    raise RuntimeError("No numeric/Boolean Parameter columns exist in cda.")

phase_candidates = [c for c in parameter_columns if "flightphase" in c.lower()]
phase_cols = {}

for ch in ("AC", "BC"):
    lead = f"{PARAMETER_PREFIX}{ch}_".lower()
    matches = [c for c in phase_candidates if c.lower().startswith(lead)]

    if len(matches) != 1:
        raise RuntimeError(
            f"Could not uniquely identify {ch} FlightPhase. "
            f"Found {len(matches)} candidate(s): {matches}"
        )

    phase_cols[ch] = matches[0]

ac_phase = qc(phase_cols["AC"]).cast("double")
bc_phase = qc(phase_cols["BC"]).cast("double")

resolved_phase = (
    F.when(ac_phase.isNotNull() & bc_phase.isNotNull() & (ac_phase == bc_phase), ac_phase)
     .when(ac_phase.isNotNull() & bc_phase.isNull(), ac_phase)
     .when(ac_phase.isNull() & bc_phase.isNotNull(), bc_phase)
     .otherwise(F.lit(None).cast("double"))
)

parameters = [c for c in parameter_columns if c not in set(phase_cols.values())]

if not parameters:
    raise RuntimeError("No analyzable parameters remain after excluding FlightPhase.")

source = (
    cda
    .withColumn("_AD_Phase", resolved_phase)
    .where(
        F.col("Timestamp").isNotNull()
        & F.col("StartDatetime").isNotNull()
        & F.col("_AD_Phase").isNotNull()
    )
)

flight_rows = (
    source
    .select("StartDatetime")
    .distinct()
    .orderBy("StartDatetime")
    .collect()
)

flights = [pd.Timestamp(r["StartDatetime"]) for r in flight_rows]
flight_idx = {f: i for i, f in enumerate(flights)}

if len(flights) < AD_MIN_BASELINE + AD_WINDOW:
    raise RuntimeError(
        f"Only {len(flights)} flights are available. "
        f"At least {AD_MIN_BASELINE + AD_WINDOW} are required."
    )

print("=" * 126)

print("=" * 126)
print(f"Flights found:             {len(flights):,}")

print(f"AC FlightPhase:            {phase_cols['AC']}")

feature_parts = []
batch_total = math.ceil(len(parameters) / AD_BATCH)

print("STAGE 1/4 - Spark full-schema phase-conditioned aggregation")

for batch_no, start in enumerate(range(0, len(parameters), AD_BATCH), 1):
    batch = parameters[start:start + AD_BATCH]

    safe = source.select(
        "StartDatetime",
        "_AD_Phase",
        *[
            qc(p).cast("double").alias(f"_p{i}")
            for i, p in enumerate(batch)
        ]
    )

    aggs = []

    for i in range(len(batch)):
        x = F.col(f"_p{i}")

        aggs += [
            F.count(x).alias(f"_n{i}"),
            F.percentile_approx(x, [0.10, 0.50, 0.90], 1000).alias(f"_q{i}"),
            F.stddev_samp(x).alias(f"_sd{i}")
        ]

    try:
        pdf = (
            safe
            .groupBy("StartDatetime", "_AD_Phase")
            .agg(*aggs)
            .toPandas()
        )
    except Exception as e:
        raise RuntimeError(
            f"Stage 1 failed in batch {batch_no}/{batch_total}, "
            f"covering parameters {start + 1}-{start + len(batch)}. "
            f"All batch inputs were explicitly cast to double. "
            f"Original Spark error: {type(e).__name__}: {e}"
        ) from e

    rows = []

    for _, r in pdf.iterrows():
        fs = pd.Timestamp(r["StartDatetime"])

        if fs not in flight_idx or pd.isna(r["_AD_Phase"]):
            continue

        phase = float(r["_AD_Phase"])
        fi = flight_idx[fs]

        for i, parameter in enumerate(batch):
            n = r[f"_n{i}"]

            if pd.isna(n) or int(n) < AD_MIN_SAMPLES:
                continue

            q = r[f"_q{i}"]

            if q is None or len(q) != 3:
                continue

            try:
                q10, q50, q90 = map(float, q)
            except (TypeError, ValueError):
                continue

            if not all(np.isfinite([q10, q50, q90])):
                continue

            rows.append({
                "FlightStart": fs,
                "FlightIndex": fi,
                "FlightPhase": phase,
                "Parameter": parameter,
                "Feature": "LEVEL",
                "Value": q50,
                "Samples": int(n)
            })

            rows.append({
                "FlightStart": fs,
                "FlightIndex": fi,
                "FlightPhase": phase,
                "Parameter": parameter,
                "Feature": "SPREAD",
                "Value": q90 - q10,
                "Samples": int(n)
            })

            sd = r[f"_sd{i}"]

            if pd.notna(sd) and np.isfinite(float(sd)):
                rows.append({
                    "FlightStart": fs,
                    "FlightIndex": fi,
                    "FlightPhase": phase,
                    "Parameter": parameter,
                    "Feature": "STD",
                    "Value": float(sd),
                    "Samples": int(n)
                })

    if rows:
        feature_parts.append(pd.DataFrame(rows))

    print(f"  Batch {batch_no}/{batch_total} complete ({len(batch)} parameters)")

if not feature_parts:

series_count = (

print(f"  Independent series:      {series_count:,}")

print("STAGE 3/4 - Sequential consensus detection")

detections = []

for (parameter, phase, feature), series in features.groupby(
    ["Parameter", "FlightPhase", "Feature"],
    sort=False
):
    series = (
        series
        .sort_values("FlightIndex", kind="stable")
        .reset_index(drop=True)
    )

    if len(series) < AD_MIN_BASELINE + AD_WINDOW:
        continue

    baseline_n = min(AD_BASELINE_TARGET, len(series) - AD_WINDOW)

    if baseline_n < AD_MIN_BASELINE:
        continue

    baseline_values = series.iloc[:baseline_n]["Value"].to_numpy(dtype=float)
    center, scale, constant_baseline = robust_reference(baseline_values)

    if not np.isfinite(center) or not np.isfinite(scale):
        continue

    values = series["Value"].to_numpy(dtype=float)

    if constant_baseline:
        nonzero_changes = np.abs(values - center)
        positive_changes = nonzero_changes[nonzero_changes > AD_EPS]

        if positive_changes.size:
            fallback_scale = float(np.median(positive_changes))
            scale = max(fallback_scale, AD_EPS)
        else:
            continue

    z_all = np.clip((values - center) / scale, -100.0, 100.0)

    ewma = 0.0
    cplus = 0.0
    cminus = 0.0
    phplus = 0.0
    phminus = 0.0
    gradual_streak = 0
    prior_direction = 0

    for i in range(baseline_n, len(series)):
        row = series.iloc[i]
        z = float(z_all[i])

        ewma = AD_EWMA_LAMBDA * z + (1.0 - AD_EWMA_LAMBDA) * ewma

        cplus = max(0.0, cplus + z - AD_CUSUM_K)
        cminus = min(0.0, cminus + z + AD_CUSUM_K)

        cusum = max(cplus, abs(cminus))
        cusum_dir = 1 if cplus >= abs(cminus) else -1

        phplus = max(0.0, phplus + z - AD_PH_DELTA)
        phminus = max(0.0, phminus - z - AD_PH_DELTA)

        ph = max(phplus, phminus)
        ph_dir = 1 if phplus >= phminus else -1

        raw_p = normal_p(z)

        abrupt = (
            abs(z) >= AD_ABRUPT_Z
            and raw_p <= abrupt_p_limit
        )

        shift = z_all[max(baseline_n, i - AD_SHIFT_N + 1):i + 1]

        persistent_shift = (
            len(shift) >= AD_SHIFT_N
            and np.all(np.abs(shift) >= AD_SHIFT_Z)
            and (np.all(shift > 0) or np.all(shift < 0))
        )

        gradual = False
        slope = np.nan
        dir_fraction = np.nan
        direction = 0
        net_z = np.nan
        mk_z = np.nan
        mk_p = np.nan
        votes = 0

        if i - baseline_n + 1 >= AD_WINDOW:
            recent = z_all[i - AD_WINDOW + 1:i + 1]

            slope = theilsen(recent)
            dir_fraction, direction = directionality(recent)
            net_z = float(recent[-1] - recent[0])
            mk_z, mk_p = mann_kendall(recent)

            direction_ok = (
                direction != 0
                and np.isfinite(slope)
                and np.sign(slope) == direction
                and np.sign(net_z) == direction
                and np.sign(mk_z) == direction
                and np.sign(ewma) == direction
                and cusum_dir == direction
                and ph_dir == direction
            )

            checks = [
                dir_fraction >= AD_DIRECTION_MIN,
                abs(net_z) >= AD_NET_Z_MIN,
                abs(slope) >= AD_SLOPE_MIN,
                mk_p <= AD_MK_P_MAX,
                abs(ewma) >= AD_EWMA_WARN,
                cusum >= AD_CUSUM_WARN,
                ph >= AD_PH_WARN
            ]

            votes = sum(bool(x) for x in checks)

            gradual = (
                direction_ok
                and votes >= 6
                and (
                    abs(ewma) >= AD_EWMA_STRONG
                    or cusum >= AD_CUSUM_STRONG
                    or ph >= AD_PH_STRONG
                )
            )

        if gradual:
            if direction == prior_direction:
                gradual_streak += 1
            else:
                prior_direction = direction
                gradual_streak = 1
        else:
            gradual_streak = 0
            prior_direction = 0

        confirmed_gradual = gradual_streak >= AD_CONFIRM

        if abrupt:
            kind = "ABRUPT EXTREME"
        elif persistent_shift:
            kind = "PERSISTENT LEVEL SHIFT"
        elif confirmed_gradual:
            kind = "GRADUAL DEGRADATION"
        else:
            continue

        observed = float(row["Value"])
        deviation = pct(observed, center)

        reasons = [f"same-phase deviation {z:+.2f} robust sigma"]

        if np.isfinite(deviation):
            reasons.append(f"{deviation:+.3f}% versus frozen baseline")

        if abrupt:
            reasons.append(
                f"abrupt p={raw_p:.2e}; corrected threshold={abrupt_p_limit:.2e}"
            )

        if persistent_shift:
            reasons.append(
                f"{AD_SHIFT_N} consecutive same-direction values >= {AD_SHIFT_Z:.1f} sigma"
            )

        if confirmed_gradual:
            reasons += [
                f"Theil-Sen={slope:+.3f} sigma/flight",
                f"directional consistency={dir_fraction:.0%}",
                f"net movement={net_z:+.2f} sigma",
                f"Mann-Kendall p={mk_p:.3g}",
                f"EWMA={ewma:+.2f}",
                f"CUSUM={cusum:.2f}",
                f"Page-Hinkley={ph:.2f}",
                f"consensus={votes}/7",
                f"confirmed across {gradual_streak} evaluations"
            ]

        strength = sum([
            abs(z) >= 6.0,
            abs(ewma) >= AD_EWMA_STRONG,
            cusum >= AD_CUSUM_STRONG,
            ph >= AD_PH_STRONG,
            np.isfinite(mk_p) and mk_p <= 0.02,
            np.isfinite(dir_fraction) and dir_fraction >= 0.85
        ])

        very_high = (
            kind == "ABRUPT EXTREME"
            or (kind == "PERSISTENT LEVEL SHIFT" and strength >= 2)
            or (kind == "GRADUAL DEGRADATION" and strength >= 4)
        )

        detections.append({
            "FlightStart": pd.Timestamp(row["FlightStart"]),
            "FlightIndex": int(row["FlightIndex"]),
            "FlightPhase": float(row["FlightPhase"]),
            "Parameter": parameter,
            "Feature": feature,
            "Detection": kind,
            "Confidence": "VERY HIGH" if very_high else "HIGH",
            "Observed": observed,
            "Baseline": center,
            "DeviationPct": deviation,
            "RobustZ": z,
            "EWMA": ewma,
            "CUSUM": cusum,
            "PageHinkley": ph,
            "Trend": slope,
            "Directional": dir_fraction,
            "MannKendallP": mk_p,
            "Consensus": votes,
            "Reason": "; ".join(reasons)
        })

anomaly_detail_df = pd.DataFrame(detections)

if not anomaly_detail_df.empty:

    def pretty_parameter(p):

        "BASELINE": anomaly_detail_df["Baseline"].map(lambda x: f"{x:,.6g}"),

        "DEVIATION": anomaly_detail_df["DeviationPct"].map(
            lambda x: f"{x:+.3f}%" if np.isfinite(x) else "N/A"
        ),

        "ROBUST SIGMA": anomaly_detail_df["RobustZ"].map(lambda x: f"{x:+.2f}"),
        "EWMA": anomaly_detail_df["EWMA"].map(lambda x: f"{x:+.2f}"),
        "CUSUM": anomaly_detail_df["CUSUM"].map(lambda x: f"{x:.2f}"),
        "PAGE-HINKLEY": anomaly_detail_df["PageHinkley"].map(lambda x: f"{x:.2f}"),

        "TREND SIGMA/FLIGHT": anomaly_detail_df["Trend"].map(
            lambda x: f"{x:+.3f}" if np.isfinite(x) else "-"
        ),

        "DIRECTIONAL": anomaly_detail_df["Directional"].map(
            lambda x: f"{x:.0%}" if np.isfinite(x) else "-"
        ),

        "MK p": anomaly_detail_df["MannKendallP"].map(
            lambda x: f"{x:.3g}" if np.isfinite(x) else "-"
        ),

        "CONSENSUS": anomaly_detail_df["Consensus"].map(lambda x: f"{int(x)}/7"),
        "REASON": anomaly_detail_df["Reason"]
    })

    print()
    print("=" * 126)
    print("CONFIRMED ANOMALIES")

print()
print("STAGE 4/4 - Raw telemetry visualization")

if anomaly_detail_df.empty:
    print("No confirmed anomaly signals to plot.")

else:
    priority = (
        anomaly_detail_df
        .groupby("_Signal")
        .agg(
            Count=("Detection", "size"),
            VeryHigh=("Confidence", lambda x: int(np.sum(x == "VERY HIGH"))),
            MaxCUSUM=("CUSUM", "max"),
            MaxPH=("PageHinkley", "max"),
            MaxZ=("RobustZ", lambda x: float(np.max(np.abs(np.asarray(x, dtype=float)))))
        )
    )

    priority["Score"] = (
        5.0 * priority["VeryHigh"]
        + priority["Count"]
        + priority["MaxCUSUM"] / 10.0
        + priority["MaxPH"] / 10.0
        + priority["MaxZ"] / 6.0
    )

    plotted_signals = (
        priority
        .sort_values("Score", ascending=False)
        .head(AD_MAX_PLOTS)
        .index
        .tolist()
    )

    signal_columns = {}

    for signal in plotted_signals:
        members = [
            (channel_signal(p)[0], p)
            for p in parameters
            if channel_signal(p)[1] == signal
        ]

        members.sort(
            key=lambda x: (
                0 if x[0] == "AC" else 1 if x[0] == "BC" else 2,
                x[1]
            )
        )

        if members:
            signal_columns[signal] = members

    raw_columns = list(dict.fromkeys(
        p
        for members in signal_columns.values()
        for _, p in members
    ))

    if not raw_columns:
        print("No raw source columns correspond to the confirmed anomaly signals.")

    else:
        try:
            raw = (
                source
                .select(
                    "Timestamp",
                    "StartDatetime",
                    "_AD_Phase",
                    *[qc(c).cast("double").alias(c) for c in raw_columns]
                )
                .orderBy("StartDatetime", "Timestamp")
                .toPandas()
            )
        except Exception as e:
            raise RuntimeError(
                "Detection completed, but raw anomaly plotting retrieval failed. "
                f"{type(e).__name__}: {e}"
            ) from e

        raw["Timestamp"] = pd.to_datetime(raw["Timestamp"], errors="coerce")
        raw["StartDatetime"] = pd.to_datetime(raw["StartDatetime"], errors="coerce")

        raw = (
            raw
            .dropna(subset=["Timestamp", "StartDatetime"])
            .sort_values(["StartDatetime", "Timestamp"], kind="stable")
            .reset_index(drop=True)
        )

        flight_info = []
        cursor = 0.0

        for flight in flights:
            frame = raw.loc[raw["StartDatetime"] == flight]

            if frame.empty:
                continue

            start = frame["Timestamp"].min()
            end = frame["Timestamp"].max()
            duration = max((end - start).total_seconds(), 1.0)

            flight_info.append({
                "FlightStart": flight,
                "ActualStart": start,
                "ActualEnd": end,
                "PlotStart": cursor,
                "PlotEnd": cursor + duration
            })

            cursor += duration + AD_GAP_SECONDS

        transformed = []

        for info in flight_info:
            frame = raw.loc[
                raw["StartDatetime"] == info["FlightStart"]
            ].copy()

            frame["_PlotX"] = (
                info["PlotStart"]
                + (frame["Timestamp"] - info["ActualStart"]).dt.total_seconds()
            )

            transformed.append(frame)

        if transformed:
            plot_df = pd.concat(transformed, ignore_index=True)
            flight_lookup = {x["FlightStart"]: x for x in flight_info}
            markers = defaultdict(dict)

            for _, detection in anomaly_detail_df.iterrows():
                signal = detection["_Signal"]

                if signal not in signal_columns:
                    continue

                flight = pd.Timestamp(detection["FlightStart"])
                phase = float(detection["FlightPhase"])

                phase_rows = plot_df.loc[
                    (plot_df["StartDatetime"] == flight)
                    & (
                        pd.to_numeric(plot_df["_AD_Phase"], errors="coerce")
                        == phase
                    )
                ]

                if phase_rows.empty:
                    if flight not in flight_lookup:
                        continue
                    marker_x = flight_lookup[flight]["PlotStart"]

                else:
                    phase_rows = phase_rows.sort_values("Timestamp", kind="stable")
                    marker_x = float(
                        phase_rows.iloc[len(phase_rows) // 2]["_PlotX"]
                    )

                markers[signal][(flight, phase)] = marker_x

            fig, axes = plt.subplots(
                len(signal_columns),
                1,
                figsize=(22, max(6, 3.4 * len(signal_columns))),
                sharex=True,
                squeeze=False
            )

            axes = axes[:, 0]

            colors = globals().get(
                "CHANNEL_COLORS",
                {"AC": "#1f77b4", "BC": "#ff7f0e"}
            )

            for ax, (signal, members) in zip(axes, signal_columns.items()):
                for channel, column in members:
                    first = True

                    for info in flight_info:
                        seg = plot_df.loc[
                            plot_df["StartDatetime"] == info["FlightStart"],
                            ["Timestamp", "_PlotX", column]
                        ].copy()

                        seg[column] = pd.to_numeric(seg[column], errors="coerce")
                        seg = seg.dropna(subset=["_PlotX", column]).sort_values("Timestamp")

                        if seg.empty:
                            continue

                        values = seg[column].dropna().unique()

                        discrete = (
                            len(values) <= 12
                            and np.allclose(values, np.round(values), equal_nan=True)
                        )

                        dt = seg["Timestamp"].diff().dt.total_seconds()
                        positive = dt[dt > 0]
                        cadence = float(positive.median()) if not positive.empty else 1.0
                        chunk = dt.gt(max(3.0 * cadence, 5.0)).cumsum()

                        for _, piece in seg.groupby(chunk, sort=True):
                            ax.plot(
                                piece["_PlotX"],
                                piece[column],
                                linewidth=1.1,
                                color=colors.get(channel),
                                drawstyle="steps-post" if discrete else "default",
                                label=(channel or short(column)) if first else "_nolegend_",
                                zorder=4
                            )
                            first = False

                for marker_x in markers.get(signal, {}).values():
                    ax.axvline(
                        marker_x,
                        color="white",
                        linewidth=2.0,
                        alpha=0.95,
                        zorder=20
                    )

                sd = anomaly_detail_df.loc[
                    anomaly_detail_df["_Signal"] == signal
                ]

                gradual = int(np.sum(sd["Detection"] == "GRADUAL DEGRADATION"))
                shift = int(np.sum(sd["Detection"] == "PERSISTENT LEVEL SHIFT"))
                abrupt = int(np.sum(sd["Detection"] == "ABRUPT EXTREME"))

                ax.set_title(
                    f"{signal} | gradual={gradual}, shift={shift}, abrupt={abrupt}",
                    loc="left",
                    fontsize=10
                )

                ax.set_ylabel(signal, fontsize=9)
                ax.grid(True, alpha=0.20)
                ax.margins(y=0.08)

                handles, labels = ax.get_legend_handles_labels()

                if handles:
                    ax.legend(handles, labels, loc="upper right", framealpha=0.85)

            ticks = [
                (x["PlotStart"] + x["PlotEnd"]) / 2.0
                for x in flight_info
            ]

            labels = [
                x["FlightStart"].strftime("%Y-%m-%d\n%H:%M")
                for x in flight_info
            ]

            axes[-1].set_xticks(ticks)
            axes[-1].set_xticklabels(labels, rotation=45, ha="right", fontsize=7)
            axes[-1].set_xlabel("Flight timeline - calendar gaps compressed")

            if flight_info:
                axes[-1].set_xlim(
                    flight_info[0]["PlotStart"],
                    flight_info[-1]["PlotEnd"]
                )

            fig.suptitle(
                "Confirmed Full-Schema Phase-Conditioned Engine Anomalies",
                fontsize=15
            )

            fig.text(
                0.5,
                0.012,
                "White line = centre of the affected FlightPhase interval",
                ha="center",
                fontsize=9
            )

            fig.tight_layout(rect=[0.02, 0.035, 0.99, 0.965])
            plt.show()
            plt.close(fig)
