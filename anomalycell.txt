def _fad_short(name):
    return name[len(PARAMETER_PREFIX):] if name.startswith(PARAMETER_PREFIX) else name


def _fad_channel_signal(name):
    short = _fad_short(name)
    for channel in CHANNELS:
        prefix = f"{channel}_"
        if short.startswith(prefix):
            return channel, short[len(prefix):]
    return "", short


def _fad_scale(values, center=None):
    finite = np.asarray(values, dtype=float)
    finite = finite[np.isfinite(finite)]
    if finite.size == 0:
        return np.nan, np.nan
    if center is None:
        center = float(np.median(finite))
    mad = float(np.median(np.abs(finite - center))) * 1.4826
    q25, q75 = np.quantile(finite, [0.25, 0.75])
    scale = max(mad, float(q75 - q25) / 1.349, abs(center) * 1.0e-9, 1.0e-12)
    return float(center), float(scale)


def _fad_bh(values):
    p_values = np.asarray(values, dtype=float)
    if p_values.size == 0:
        return np.array([], dtype=float)
    order = np.argsort(p_values)
    ranked = p_values[order]
    adjusted = ranked * len(ranked) / np.arange(1, len(ranked) + 1)
    adjusted = np.minimum.accumulate(adjusted[::-1])[::-1]
    result = np.empty_like(adjusted)
    result[order] = np.minimum(adjusted, 1.0)
    return result


def _fad_rank(values):
    values = np.asarray(values, dtype=float)
    order = np.argsort(values, kind="stable")
    ranks = np.empty(len(values), dtype=float)
    ranks[order] = np.arange(len(values), dtype=float)
    return ranks


def _fad_screen_parameter(values, phases, starts):
    values = np.asarray(values, dtype=float)
    finite = np.isfinite(values)
    coverage = float(finite.mean())
    if finite.sum() < FAST_ANOMALY_CONFIG["screen_minimum_phase_samples"]:
        return {
            "Score": -np.inf,
            "Coverage": coverage,
            "Distinct": int(np.unique(values[finite]).size),
            "ExtremeZ": 0.0,
            "RareStateScore": 0.0,
            "DriftScore": 0.0,
            "PhaseContrast": 0.0,
        }
    distinct = int(np.unique(values[finite]).size)
    extreme_z = 0.0
    rare_score = 0.0
    drift_score = 0.0
    phase_centers = []
    phase_scales = []
    for phase in np.unique(phases[np.isfinite(phases)]):
        mask = (phases == phase) & finite
        phase_values = values[mask]
        if phase_values.size < FAST_ANOMALY_CONFIG["screen_minimum_phase_samples"]:
            continue
        center, scale = _fad_scale(phase_values)
        unique_values, counts = np.unique(phase_values, return_counts=True)
        phase_centers.append(center)
        phase_scales.append(scale)
        if len(unique_values) <= FAST_ANOMALY_CONFIG["maximum_discrete_values"]:
            frequencies = counts / counts.sum()
            minimum_frequency = float(frequencies.min())
            if minimum_frequency <= 0.02:
                rare_score = max(rare_score, min(12.0, -math.log10(max(minimum_frequency, 1.0e-12)) * 2.0))
        else:
            z = np.abs(phase_values - center) / scale
            extreme_z = max(extreme_z, float(np.quantile(z, 0.999)))
        phase_frame = pd.DataFrame(
            {
                "StartDatetime": starts[mask],
                "Value": phase_values,
            }
        )
        flight_values = (
            phase_frame.groupby("StartDatetime", sort=True)["Value"]
            .median()
            .dropna()
        )
        if len(flight_values) >= FAST_ANOMALY_CONFIG["minimum_trend_flights"]:
            x = np.arange(len(flight_values), dtype=float)
            y = flight_values.to_numpy(dtype=float)
            rho = float(np.corrcoef(_fad_rank(x), _fad_rank(y))[0, 1])
            if np.isfinite(rho):
                effect = abs(float(y[-1] - y[0])) / scale
                drift_score = max(drift_score, min(12.0, abs(rho) * effect))
    phase_contrast = 0.0
    if len(phase_centers) >= 2:
        phase_contrast = min(
            6.0,
            float(np.ptp(phase_centers)) / max(float(np.median(phase_scales)), 1.0e-12),
        )
    variability_score = min(4.0, math.log1p(distinct) / 2.0)
    score = (
        min(extreme_z, 20.0)
        + rare_score
        + drift_score
        + 0.25 * phase_contrast
        + variability_score
    ) * min(1.0, coverage / 0.20)
    return {
        "Score": float(score),
        "Coverage": coverage,
        "Distinct": distinct,
        "ExtremeZ": float(extreme_z),
        "RareStateScore": float(rare_score),
        "DriftScore": float(drift_score),
        "PhaseContrast": float(phase_contrast),
    }


def _fad_episode_table(events, gap_seconds, minimum_samples, minimum_seconds, catastrophic_z):
    if events.empty:
        empty = events.copy()
        empty["EpisodeId"] = pd.Series(dtype="int64")
        return empty, pd.DataFrame(
            columns=[
                "StartDatetime",
                "FlightPhase",
                "Parameter",
                "Signal",
                "Channel",
                "EpisodeId",
                "EpisodeStart",
                "EpisodeEnd",
                "DurationSeconds",
                "AnomalousSamples",
                "MaxScore",
                "MeanScore",
                "AnomalyType",
                "AttributionConfidence",
            ]
        )
    key_columns = ["StartDatetime", "FlightPhase", "Parameter"]
    events = events.sort_values(key_columns + ["Timestamp"], kind="stable").reset_index(drop=True)
    same_key = np.ones(len(events), dtype=bool)
    for column in key_columns:
        same_key &= events[column].eq(events[column].shift()).to_numpy()
    time_gap = events["Timestamp"].diff().dt.total_seconds().to_numpy()
    new_episode = ~same_key | ~np.isfinite(time_gap) | (time_gap > gap_seconds)
    events["EpisodeId"] = np.cumsum(new_episode).astype(np.int64)
    episodes = (
        events.groupby(
            ["StartDatetime", "FlightPhase", "Parameter", "Signal", "Channel", "EpisodeId"],
            dropna=False,
            sort=False,
        )
        .agg(
            EpisodeStart=("Timestamp", "min"),
            EpisodeEnd=("Timestamp", "max"),
            AnomalousSamples=("Timestamp", "size"),
            MaxScore=("Score", "max"),
            MeanScore=("Score", "mean"),
            AnomalyType=("AnomalyType", lambda values: "+".join(sorted(set(values)))),
            AttributionConfidence=("AttributionConfidence", "max"),
        )
        .reset_index()
    )
    episodes["DurationSeconds"] = (
        episodes["EpisodeEnd"] - episodes["EpisodeStart"]
    ).dt.total_seconds()
    episodes = episodes.loc[
        (episodes["AnomalousSamples"] >= minimum_samples)
        | (episodes["DurationSeconds"] >= minimum_seconds)
        | (episodes["MaxScore"] >= catastrophic_z)
    ].reset_index(drop=True)
    confirmed_ids = set(episodes["EpisodeId"].tolist())
    samples = events.loc[events["EpisodeId"].isin(confirmed_ids)].reset_index(drop=True)
    return samples, episodes

        is_discrete = len(distinct_values) <= FAST_ANOMALY_CONFIG["maximum_discrete_values"]
        baselines[phase] = {
            "center": center,
            "scale": scale,
            "discrete": is_discrete,
            "coverage": float(finite_values.size / len(indices)),
        }

        local_z = np.zeros(len(indices), dtype=float)
        local_z[finite_mask] = np.abs(finite_values - center) / scale
        amplitude_z[indices] = np.minimum(local_z, 100.0).astype(np.float32)

        if is_discrete:
            frequencies = distinct_counts / distinct_counts.sum()
            rare_values = distinct_values[
                frequencies <= FAST_ANOMALY_CONFIG["rare_discrete_fraction"]
            ]
            local_flag = finite_mask & np.isin(phase_data, rare_values)
        else:
            tail_probability = max(
                FAST_ANOMALY_CONFIG["tail_probability"],
                min(
                    0.01,
                    FAST_ANOMALY_CONFIG["minimum_tail_observations"] / finite_values.size,
                ),
            )
            lower, upper = np.quantile(
                finite_values,
                [tail_probability, 1.0 - tail_probability],
            )
            local_flag = finite_mask & (
                ((phase_data < lower) | (phase_data > upper))
                & (local_z >= FAST_ANOMALY_CONFIG["minimum_amplitude_z"])
                | (local_z >= FAST_ANOMALY_CONFIG["catastrophic_z"])
            )

        amplitude_flag[indices] = local_flag

        if finite_values.size / len(indices) >= 0.995:
            dropout_flag[indices] = ~finite_mask

        phase_rate_positions = indices[np.isfinite(rates[indices])]
        phase_rates = rates[phase_rate_positions]
        if phase_rates.size >= FAST_ANOMALY_CONFIG["minimum_phase_samples"]:
            rate_center, rate_scale = _fad_scale(phase_rates)
            rate_distinct = np.unique(phase_rates).size
            local_rate_z = np.abs(phase_rates - rate_center) / rate_scale
            rate_z[phase_rate_positions] = np.minimum(local_rate_z, 100.0).astype(np.float32)
            if rate_distinct > FAST_ANOMALY_CONFIG["maximum_discrete_values"]:
                rate_tail_probability = max(
                    FAST_ANOMALY_CONFIG["tail_probability"],
                    min(
                        0.01,
                        FAST_ANOMALY_CONFIG["minimum_tail_observations"] / phase_rates.size,
                    ),
                )
                rate_lower, rate_upper = np.quantile(
                    phase_rates,
                    [rate_tail_probability, 1.0 - rate_tail_probability],
                )
                rate_flag[phase_rate_positions] = (
                    (((phase_rates < rate_lower) | (phase_rates > rate_upper))
                    & (local_rate_z >= FAST_ANOMALY_CONFIG["minimum_rate_z"]))
                    | (local_rate_z >= FAST_ANOMALY_CONFIG["catastrophic_z"])
                )

    parameter_baselines[parameter] = baselines
    combined_flag = eligible & (amplitude_flag | rate_flag | dropout_flag)
    flagged_positions = np.flatnonzero(combined_flag)

    if flagged_positions.size:
        channel, signal = _fad_channel_signal(parameter)
        types = np.where(
            dropout_flag[flagged_positions],
            "UNEXPECTED_DATA_DROPOUT",
            np.where(
                amplitude_flag[flagged_positions] & rate_flag[flagged_positions],
                "PHASE_TAIL+TEMPORAL_JUMP",
                np.where(
                    amplitude_flag[flagged_positions],
                    "PHASE_CONDITIONAL_TAIL",
                    "TEMPORAL_JUMP",
                ),
            ),
        )
        scores = np.maximum(
            amplitude_z[flagged_positions],
            rate_z[flagged_positions],
        ).astype(float)
        scores[dropout_flag[flagged_positions]] = np.maximum(
            scores[dropout_flag[flagged_positions]],
            FAST_ANOMALY_CONFIG["minimum_amplitude_z"],
        )
        raw_event_frames.append(
            pd.DataFrame(
                {
                    "RowPosition": flagged_positions,
                    "Timestamp": metadata["Timestamp"].to_numpy()[flagged_positions],
                    "StartDatetime": metadata["StartDatetime"].to_numpy()[flagged_positions],
                    "FlightPhase": phase_values[flagged_positions],
                    "Parameter": parameter,
                    "Signal": signal,
                    "Channel": channel,
                    "Value": values[flagged_positions],
                    "PeerParameter": None,
                    "PeerValue": np.nan,
                    "Score": scores,
                    "AnomalyType": types,
                    "AttributionConfidence": 1.0,
                }
            )
        )

    eligible_values = values[eligible]
    median_series = pd.Series(eligible_values).groupby(group_codes).median()
    count_series = pd.Series(eligible_values).groupby(group_codes).count()
    for code, median_value in median_series.items():
        if not np.isfinite(median_value) or count_series.loc[code] == 0:
            continue
        start_datetime, phase = group_keys[code]
        flight_phase_median_records.append(
            {
                "StartDatetime": start_datetime,
                "FlightPhase": float(phase),
                "Parameter": parameter,
                "MedianValue": float(median_value),
                "Samples": int(count_series.loc[code]),
            }
        )

    if parameter_index % 20 == 0 or parameter_index == len(analysis_parameters):
        print(f"Scored {parameter_index:,} of {len(analysis_parameters):,} parameters")

        if phase_difference.size < FAST_ANOMALY_CONFIG["minimum_phase_samples"]:
            continue
        center, scale = _fad_scale(phase_difference)
        pair_z = np.abs(phase_difference - center) / scale
        distinct_values, distinct_counts = np.unique(phase_difference, return_counts=True)
        if len(distinct_values) <= FAST_ANOMALY_CONFIG["maximum_discrete_values"]:
            frequencies = distinct_counts / distinct_counts.sum()
            rare_values = distinct_values[
                frequencies <= FAST_ANOMALY_CONFIG["pair_rare_discrete_fraction"]
            ]
            pair_flag = np.isin(phase_difference, rare_values)
        else:
            tail_probability = max(
                FAST_ANOMALY_CONFIG["tail_probability"],
                min(
                    0.01,
                    FAST_ANOMALY_CONFIG["minimum_tail_observations"] / phase_difference.size,
                ),
            )
            lower, upper = np.quantile(
                phase_difference,
                [tail_probability, 1.0 - tail_probability],
            )
            pair_flag = (
                (((phase_difference < lower) | (phase_difference > upper))
                & (pair_z >= FAST_ANOMALY_CONFIG["minimum_amplitude_z"]))
                | (pair_z >= FAST_ANOMALY_CONFIG["catastrophic_z"])
            )

        flagged_local = np.flatnonzero(pair_flag)
        if flagged_local.size == 0:
            continue
        flagged_positions = finite_positions[flagged_local]
        ac_baseline = parameter_baselines.get(ac_parameter, {}).get(phase)
        bc_baseline = parameter_baselines.get(bc_parameter, {}).get(phase)
        if ac_baseline is None or bc_baseline is None:
            continue
        ac_z = np.abs(ac_values[flagged_positions] - ac_baseline["center"]) / ac_baseline["scale"]
        bc_z = np.abs(bc_values[flagged_positions] - bc_baseline["center"]) / bc_baseline["scale"]
        ac_wins = (
            ac_z >= FAST_ANOMALY_CONFIG["minimum_amplitude_z"]
        ) & (
            ac_z >= FAST_ANOMALY_CONFIG["pair_attribution_ratio"] * bc_z
        )
        bc_wins = (
            bc_z >= FAST_ANOMALY_CONFIG["minimum_amplitude_z"]
        ) & (
            bc_z >= FAST_ANOMALY_CONFIG["pair_attribution_ratio"] * ac_z
        )
        attributed_parameter = np.where(
            ac_wins,
            ac_parameter,
            np.where(
                bc_wins,
                bc_parameter,
                f"{ac_parameter} | {bc_parameter}",
            ),
        )
        attributed_channel = np.where(ac_wins, "AC", np.where(bc_wins, "BC", "AC|BC"))
        attributed_value = np.where(
            ac_wins,
            ac_values[flagged_positions],
            np.where(bc_wins, bc_values[flagged_positions], np.nan),
        )
        peer_parameter = np.where(
            ac_wins,
            bc_parameter,
            np.where(bc_wins, ac_parameter, None),
        )
        peer_value = np.where(
            ac_wins,
            bc_values[flagged_positions],
            np.where(bc_wins, ac_values[flagged_positions], np.nan),
        )
        confidence = np.where(
            ac_wins | bc_wins,
            np.maximum(ac_z, bc_z) / (ac_z + bc_z + 1.0e-12),
            0.5,
        )
        raw_event_frames.append(
            pd.DataFrame(
                {
                    "RowPosition": flagged_positions,
                    "Timestamp": metadata["Timestamp"].to_numpy()[flagged_positions],
                    "StartDatetime": metadata["StartDatetime"].to_numpy()[flagged_positions],
                    "FlightPhase": phase_values[flagged_positions],
                    "Parameter": attributed_parameter,
                    "Signal": signal,
                    "Channel": attributed_channel,
                    "Value": attributed_value,
                    "PeerParameter": peer_parameter,
                    "PeerValue": peer_value,
                    "Score": np.minimum(pair_z[flagged_local], 100.0),
                    "AnomalyType": "AC_BC_CHANNEL_DIVERGENCE",
                    "AttributionConfidence": confidence,
                }
            )
        )

trend_records = []

if not flight_phase_medians.empty:
    for (parameter, phase), group in flight_phase_medians.groupby(
        ["Parameter", "FlightPhase"], sort=False
    ):
        group = group.sort_values("StartDatetime").drop_duplicates("StartDatetime")
        if len(group) < FAST_ANOMALY_CONFIG["minimum_trend_flights"]:
            continue
        x = (
            group["StartDatetime"] - group["StartDatetime"].iloc[0]
        ).dt.total_seconds().to_numpy(dtype=float) / 86400.0
        y = group["MedianValue"].to_numpy(dtype=float)
        if x[-1] <= x[0] or not np.isfinite(y).all():
            continue
        center, scale = _fad_scale(y)
        slope, intercept, _, _ = theilslopes(y, x, 0.99)
        _, noise_scale = _fad_scale(y - (slope * x + intercept))
        noise_scale = max(noise_scale, abs(center) * 1.0e-6, scale * 0.02)
        tau_result = kendalltau(x, y, nan_policy="omit")
        tau_value = getattr(tau_result, "statistic", tau_result[0])
        tau = float(tau_value) if np.isfinite(tau_value) else 0.0
        trend_p = float(tau_result.pvalue) if np.isfinite(tau_result.pvalue) else 1.0
        estimated_change = float(slope * (x[-1] - x[0]))
        trend_effect_z = abs(estimated_change) / noise_scale
        relative_change = abs(estimated_change) / max(abs(center), scale, 1.0e-12)
        best_change_z = 0.0
        best_split = None
        for split in range(3, len(y) - 2):
            change_z = abs(float(np.median(y[split:]) - np.median(y[:split]))) / noise_scale
            if change_z > best_change_z:
                best_change_z = change_z
                best_split = split
        change_p = min(1.0, 2.0 * norm.sf(best_change_z) * max(1, len(y) - 5))
        combined_p = min(1.0, min(trend_p, change_p) * 2.0)
        effect_qualified = (
            (
                trend_effect_z >= FAST_ANOMALY_CONFIG["minimum_trend_effect_z"]
                and abs(tau) >= FAST_ANOMALY_CONFIG["minimum_trend_tau"]
                and relative_change >= FAST_ANOMALY_CONFIG["minimum_relative_trend_change"]
            )
            or (
                best_change_z >= FAST_ANOMALY_CONFIG["minimum_trend_effect_z"] * 1.5
                and relative_change >= FAST_ANOMALY_CONFIG["minimum_relative_trend_change"]
            )
        )
        channel, signal = _fad_channel_signal(parameter)
        trend_records.append(
            {
                "Parameter": parameter,
                "Signal": signal,
                "Channel": channel,
                "FlightPhase": phase,
                "Flights": len(group),
                "FirstFlight": group["StartDatetime"].iloc[0],
                "LastFlight": group["StartDatetime"].iloc[-1],
                "ChangePointFlight": (
                    group["StartDatetime"].iloc[best_split]
                    if best_split is not None
                    else pd.NaT
                ),
                "TheilSenSlopePerDay": float(slope),
                "EstimatedWindowChange": estimated_change,
                "RelativeWindowChange": relative_change,
                "TrendEffectZ": float(trend_effect_z),
                "KendallTau": tau,
                "ChangePointZ": float(best_change_z),
                "PValue": combined_p,
                "EffectQualified": effect_qualified,
            }
        )

contextual_records = []

if not flight_phase_medians.empty:
    for phase, group in flight_phase_medians.groupby("FlightPhase", sort=False):
        matrix = group.pivot_table(
            index="StartDatetime",
            columns="Parameter",
            values="MedianValue",
            aggfunc="median",
        ).sort_index()
        if len(matrix) < FAST_ANOMALY_CONFIG["contextual_minimum_flights"]:
            continue
        coverage = matrix.notna().mean()
        usable_columns = coverage.index[coverage >= 0.8].tolist()
        if len(usable_columns) < 3:
            continue
        matrix = matrix[usable_columns]
        centers = matrix.median(axis=0)
        scales = pd.Series(index=usable_columns, dtype=float)
        for column in usable_columns:
            _, scales[column] = _fad_scale(matrix[column].dropna().to_numpy(), centers[column])
        standardized = matrix.subtract(centers, axis=1).divide(scales, axis=1).fillna(0.0)
        standardized = standardized.clip(-10.0, 10.0)
        values = standardized.to_numpy(dtype=float)
        if min(values.shape) < 3:
            continue
        _, singular_values, components = np.linalg.svd(values, full_matrices=False)
        variance = singular_values ** 2
        cumulative = np.cumsum(variance) / max(float(variance.sum()), 1.0e-12)
        variance_components = int(np.searchsorted(cumulative, 0.90) + 1)
        component_count = max(
            1,
            min(
                variance_components,
                8,
                values.shape[0] // 3,
                values.shape[1] // 3,
            ),
        )
        reconstruction = (values @ components[:component_count].T) @ components[:component_count]
        residual = np.abs(values - reconstruction)
        for column_index, parameter in enumerate(usable_columns):
            residual_center, residual_scale = _fad_scale(residual[:, column_index])
            residual_z = np.abs(residual[:, column_index] - residual_center) / residual_scale
            raw_z = np.abs(values[:, column_index])
            flagged = np.flatnonzero(
                (residual_z >= FAST_ANOMALY_CONFIG["contextual_minimum_z"])
                & (raw_z >= FAST_ANOMALY_CONFIG["contextual_minimum_raw_z"])
            )
            channel, signal = _fad_channel_signal(parameter)
            for row_index in flagged:
                contextual_records.append(
                    {
                        "StartDatetime": matrix.index[row_index],
                        "FlightPhase": float(phase),
                        "Parameter": parameter,
                        "Signal": signal,
                        "Channel": channel,
                        "MedianValue": matrix.iloc[row_index, column_index],
                        "RawRobustZ": float(raw_z[row_index]),
                        "ContextualResidualZ": float(residual_z[row_index]),
                        "Components": component_count,
                    }
                )
