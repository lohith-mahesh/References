def ad_qcol(name):
    return F.col(f"`{name.replace('`', '``')}`")

def ad_short(name):
    name = str(name)
    return (
        name[len(PARAMETER_PREFIX):]
        if name.startswith(PARAMETER_PREFIX)
        else name
    )

def ad_split_channel(name):
    name = ad_short(name)
    if name.startswith("AC_"):
        return "AC", name[3:]
    if name.startswith("BC_"):
        return "BC", name[3:]
    return "", name

def ad_robust_center_scale(values):
    x = np.asarray(values, dtype=float)
    x = x[np.isfinite(x)]

    if x.size == 0:
        return np.nan, np.nan

    center = float(np.median(x))
    mad = float(np.median(np.abs(x - center)))
    scale = 1.4826 * mad

    if not np.isfinite(scale) or scale <= AD_EPS:
        q25, q75 = np.percentile(x, [25.0, 75.0])
        scale = float((q75 - q25) / 1.349)

    if not np.isfinite(scale) or scale <= AD_EPS:
        scale = float(np.std(x))

    if not np.isfinite(scale) or scale <= AD_EPS:
        scale = AD_EPS

    return center, scale

def ad_normal_p(z):
    if not np.isfinite(z):
        return 1.0

    return float(
        math.erfc(
            abs(float(z)) / math.sqrt(2.0)
        )
    )

def ad_percent_deviation(value, baseline):
    if (
        not np.isfinite(value)
        or not np.isfinite(baseline)
        or abs(baseline) <= AD_EPS
    ):
        return np.nan

    return float(
        100.0
        * (value - baseline)
        / abs(baseline)
    )

def ad_theilsen(values):
    y = np.asarray(values, dtype=float)
    y = y[np.isfinite(y)]

    n = len(y)

    if n < 3:
        return np.nan

    slopes = []

    for i in range(n - 1):
        for j in range(i + 1, n):
            slopes.append(
                (y[j] - y[i])
                / float(j - i)
            )

    return (
        float(np.median(slopes))
        if slopes
        else np.nan
    )

def ad_directionality(values):
    y = np.asarray(values, dtype=float)
    y = y[np.isfinite(y)]

    if len(y) < 3:
        return 0.0, 0

    d = np.diff(y)

    positive = int(np.sum(d > 0))
    negative = int(np.sum(d < 0))

    total = positive + negative

    if total == 0:
        return 0.0, 0

    if positive >= negative:
        return positive / total, 1

    return negative / total, -1

def ad_mann_kendall(values):
    y = np.asarray(values, dtype=float)
    y = y[np.isfinite(y)]

    n = len(y)

    if n < 5:
        return 0.0, 1.0

    s = 0

    for i in range(n - 1):
        s += int(
            np.sum(
                np.sign(
                    y[i + 1:] - y[i]
                )
            )
        )

    _, counts = np.unique(
        y,
        return_counts=True
    )

    tie_term = float(
        np.sum(
            counts
            * (counts - 1)
            * (2 * counts + 5)
        )
    )

    variance = (
        n * (n - 1) * (2 * n + 5)
        - tie_term
    ) / 18.0

    if variance <= 0:
        return 0.0, 1.0

    if s > 0:
        z = (s - 1) / math.sqrt(variance)
    elif s < 0:
        z = (s + 1) / math.sqrt(variance)
    else:
        z = 0.0

    p = math.erfc(
        abs(z) / math.sqrt(2.0)
    )

    return float(z), float(p)

detections = []

for (
    parameter,
    phase,
    feature
), series in feature_df.groupby(
    [
        "Parameter",
        "FlightPhase",
        "Feature"
    ],
    sort=False
):
    series = (
        series
        .sort_values(
            "FlightIndex",

    if (
        len(series)
        < AD_MIN_BASELINE_FLIGHTS
        + AD_SEQUENCE_WINDOW
    ):
        continue

    baseline_count = min(
        AD_TARGET_BASELINE_FLIGHTS,
        len(series)
        - AD_SEQUENCE_WINDOW
    )

    if (
        baseline_count
        < AD_MIN_BASELINE_FLIGHTS
    ):
        continue

    baseline_values = (
        series
        .iloc[
            :baseline_count
        ]["Value"]
        .to_numpy(
            dtype=float
        )
    )

    baseline_center, baseline_scale = (
        ad_robust_center_scale(
            baseline_values
        )
    )

    if (
        not np.isfinite(
            baseline_center
        )
        or not np.isfinite(
            baseline_scale
        )
    ):
        continue

    values = (
        series["Value"]
        .to_numpy(
            dtype=float
        )
    )

    z_values = np.clip(
        (
            values
            - baseline_center
        )
        / max(
            baseline_scale,
            AD_EPS
        ),
        -100.0,
        100.0
    )

    ewma = 0.0
    cplus = 0.0
    cminus = 0.0
    ph_plus = 0.0
    ph_minus = 0.0

    gradual_streak = 0
    previous_gradual_direction = 0

    for i in range(
        baseline_count,
        len(series)
    ):
        row = series.iloc[i]

        z = float(
            z_values[i]
        )

        ewma = (
            AD_EWMA_LAMBDA
            * z
            + (
                1.0
                - AD_EWMA_LAMBDA
            )
            * ewma
        )

        cplus = max(
            0.0,
            cplus
            + z
            - AD_CUSUM_K
        )

        cminus = min(
            0.0,
            cminus
            + z
            + AD_CUSUM_K
        )

        cusum = max(
            cplus,
            abs(cminus)
        )

        cusum_direction = (
            1
            if cplus >= abs(cminus)
            else -1
        )

        ph_plus = max(
            0.0,
            ph_plus
            + z
            - AD_PH_DELTA
        )

        ph_minus = max(
            0.0,
            ph_minus
            - z
            - AD_PH_DELTA
        )

        ph_score = max(
            ph_plus,
            ph_minus
        )

        ph_direction = (
            1
            if ph_plus >= ph_minus
            else -1
        )

        raw_p = ad_normal_p(
            z
        )

        abrupt = (
            abs(z)
            >= AD_ABRUPT_Z
            and raw_p
            <= abrupt_p_limit
        )

        shift_start = max(
            baseline_count,
            i
            - AD_SHIFT_COUNT
            + 1
        )

        recent_shift = (
            z_values[
                shift_start:
                i + 1
            ]
        )

        persistent_shift = (
            len(recent_shift)
            >= AD_SHIFT_COUNT
            and np.all(
                np.abs(
                    recent_shift
                )
                >= AD_SHIFT_Z
            )
            and (
                np.all(
                    recent_shift > 0
                )
                or np.all(
                    recent_shift < 0
                )
            )
        )

        gradual = False
        theilsen = np.nan
        direction_fraction = np.nan
        direction = 0
        net_z = np.nan
        mk_z = np.nan
        mk_p = np.nan
        consensus_votes = 0

        if (
            i
            - baseline_count
            + 1
            >= AD_SEQUENCE_WINDOW
        ):
            recent_z = (
                z_values[
                    i
                    - AD_SEQUENCE_WINDOW
                    + 1:
                    i + 1
                ]
            )

            theilsen = ad_theilsen(
                recent_z
            )

            (
                direction_fraction,
                direction
            ) = ad_directionality(
                recent_z
            )

            net_z = float(
                recent_z[-1]
                - recent_z[0]
            )

            (
                mk_z,
                mk_p
            ) = ad_mann_kendall(
                recent_z
            )

            direction_checks = [
                direction != 0,
                np.isfinite(
                    theilsen
                )
                and np.sign(
                    theilsen
                )
                == direction,
                np.sign(
                    net_z
                )
                == direction,
                np.sign(
                    mk_z
                )
                == direction,
                np.sign(
                    ewma
                )
                == direction,
                cusum_direction
                == direction,
                ph_direction
                == direction
            ]

            direction_agreement = all(
                direction_checks
            )

            evidence_checks = [
                direction_fraction
                >= AD_MIN_DIRECTIONAL_FRACTION,

                abs(net_z)
                >= AD_MIN_NET_Z,

                np.isfinite(
                    theilsen
                )
                and abs(
                    theilsen
                )
                >= AD_MIN_THEILSEN,

                np.isfinite(
                    mk_p
                )
                and mk_p
                <= AD_MK_P,

                abs(ewma)
                >= AD_EWMA_WARN,

                cusum
                >= AD_CUSUM_WARN,

                ph_score
                >= AD_PH_WARN
            ]

            consensus_votes = int(
                sum(
                    bool(x)
                    for x in evidence_checks
                )
            )

            strong_sequential = (
                abs(ewma)
                >= AD_EWMA_STRONG
                or cusum
                >= AD_CUSUM_STRONG
                or ph_score
                >= AD_PH_STRONG
            )

            gradual = (
                direction_agreement
                and consensus_votes >= 6
                and strong_sequential
            )

        if gradual:
            if (
                direction
                == previous_gradual_direction
            ):
                gradual_streak += 1
            else:
                previous_gradual_direction = (
                    direction
                )
                gradual_streak = 1
        else:
            gradual_streak = 0
            previous_gradual_direction = 0

        confirmed_gradual = (
            gradual_streak
            >= AD_CONFIRMATIONS
        )

        detection_type = None

        if abrupt:
            detection_type = (
                "ABRUPT EXTREME"
            )

        elif persistent_shift:
            detection_type = (
                "PERSISTENT LEVEL SHIFT"
            )

        elif confirmed_gradual:
            detection_type = (
                "GRADUAL DEGRADATION"
            )

            continue

        )

                observed,
                baseline_center
            )
        )

            f"same-FlightPhase deviation {z:+.2f} robust sigma"
        ]

            deviation_pct
        ):
            reason.append(
                f"{deviation_pct:+.3f}% versus frozen baseline"
            )

        if abrupt:
            reason.append(
                f"abrupt-event p={raw_p:.2e}, "
                f"threshold={abrupt_p_limit:.2e}"
            )

        if persistent_shift:
            reason.append(
                f"{AD_SHIFT_COUNT} consecutive same-direction "
                f"observations >= {AD_SHIFT_Z:.1f} sigma"
            )

        if confirmed_gradual:
            reason.extend(
                [
                    f"Theil-Sen={theilsen:+.3f} sigma/flight",
                    f"directional consistency={direction_fraction:.0%}",
                    f"net {AD_SEQUENCE_WINDOW}-observation movement={net_z:+.2f} sigma",
                    f"Mann-Kendall p={mk_p:.3g}",
                    f"EWMA={ewma:+.2f} sigma",
                    f"CUSUM={cusum:.2f}",
                    f"Page-Hinkley={ph_score:.2f}",
                    f"consensus={consensus_votes}/7",
                    f"confirmed on {gradual_streak} successive evaluations"
                ]
            )

        strong_votes = int(
            sum(
                [
                    abs(z) >= 6.0,
                    abs(ewma) >= AD_EWMA_STRONG,
                    cusum >= AD_CUSUM_STRONG,
                    ph_score >= AD_PH_STRONG,
                    np.isfinite(mk_p)
                    and mk_p <= 0.02,
                    np.isfinite(
                        direction_fraction
                    )
                    and direction_fraction >= 0.85
                ]
            )
        )

        if detection_type == "ABRUPT EXTREME":
            confidence = "VERY HIGH"

        elif (
            detection_type
            == "PERSISTENT LEVEL SHIFT"
            and strong_votes >= 2
        ):
            confidence = "VERY HIGH"

        elif (
            detection_type
            == "GRADUAL DEGRADATION"
            and strong_votes >= 4
        ):
            confidence = "VERY HIGH"

        else:
            confidence = "HIGH"

        detections.append(
            {
                "FlightStart": pd.Timestamp(
                    row["FlightStart"]
                ),
                "FlightIndex": int(
                    row["FlightIndex"]
                ),
                "FlightPhase": float(
                    row["FlightPhase"]

                "MannKendallP": mk_p,
                "NetMovementZ": net_z,
                "ConsensusVotes": consensus_votes,
                "Reason": "; ".join(

            }
