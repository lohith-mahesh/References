AD_DIRECTION_MIN_Z = 1.50
AD_TREND_MIN = 0.35
AD_FWER_ALPHA = 0.01
AD_EPS = 1e-12
AD_MAX_TABLE_ROWS = 500
AD_MAX_LABELS = 30

    .reset_index(drop=True)

    raise RuntimeError("No valid telemetry rows remain after timestamp/phase filtering.")

flight_starts = sorted(
    pd.Timestamp(x)
    for x in ad_df["StartDatetime"].dropna().unique()
)

flight_index = {
    flight: i
    for i, flight in enumerate(flight_starts)
}

ad_df["_AD_FlightIndex"] = ad_df["StartDatetime"].map(flight_index)

parameter_columns = [
    c for c in PARAMETER_COLUMNS
    if c in ad_df.columns and c != FLIGHT_PHASE_COLUMN
]

if not parameter_columns:
    raise RuntimeError("No parameter columns are available for anomaly detection.")

def ad_short_name(column):
    prefix = globals().get("PARAMETER_PREFIX", "")
    if prefix and str(column).startswith(prefix):
        return str(column)[len(prefix):]
    return str(column)

def ad_channel_signal(column):
    short = ad_short_name(column)
    if short.startswith("AC_"):
        return "AC", short[3:]
    if short.startswith("BC_"):
        return "BC", short[3:]
    return "", short

def ad_center_scale(values):
    x = np.asarray(values, dtype=float)
    x = x[np.isfinite(x)]

    if x.size == 0:
        return np.nan, np.nan

    center = float(np.median(x))
    mad = float(np.median(np.abs(x - center)))
    scale = 1.4826 * mad

    if not np.isfinite(scale) or scale <= AD_EPS:
        q25, q75 = np.percentile(x, [25, 75])
        scale = float((q75 - q25) / 1.349)

    if not np.isfinite(scale) or scale <= AD_EPS:
        scale = float(np.std(x))

    if not np.isfinite(scale) or scale <= AD_EPS:
        scale = AD_EPS

    return center, scale

def ad_normal_two_sided_p(z):
    if not np.isfinite(z):
        return 1.0
    return float(math.erfc(abs(float(z)) / math.sqrt(2.0)))

def ad_pct(value, baseline):
    if not np.isfinite(value) or not np.isfinite(baseline):
        return np.nan

    if abs(baseline) <= AD_EPS:
        return np.nan

    return 100.0 * (value - baseline) / abs(baseline)

def ad_linear_slope(values):
    y = np.asarray(values, dtype=float)
    good = np.isfinite(y)

    if good.sum() < 3:
        return np.nan

    y = y[good]
    x = np.arange(len(y), dtype=float)

    x -= x.mean()
    y -= y.mean()

    denom = float(np.dot(x, x))

    if denom <= AD_EPS:
        return np.nan

    return float(np.dot(x, y) / denom)

def ad_confidence_label(z, p_corrected, consensus, persistence):
    if (
        abs(z) >= 7.0
        and p_corrected <= 1e-6
        and consensus >= 3
        and persistence >= AD_PERSISTENCE_MIN
    ):
        return "EXTREME"

    if (
        abs(z) >= 6.0
        and p_corrected <= 1e-4
        and consensus >= 3
        and persistence >= AD_PERSISTENCE_MIN
    ):
        return "VERY HIGH"

    return "HIGH"

numeric_parameters = []

for column in parameter_columns:
    converted = pd.to_numeric(ad_df[column], errors="coerce")

    if converted.notna().sum() >= AD_MIN_PHASE_SAMPLES:
        ad_df[column] = converted
        numeric_parameters.append(column)

if not numeric_parameters:
    raise RuntimeError("No sufficiently populated numeric parameters were found.")

print(
    f"High-confidence anomaly detector starting: "
    f"{len(flight_starts):,} flights, "
    f"{len(numeric_parameters):,} numeric parameters."
)

feature_records = []

grouped = ad_df.groupby(
    ["StartDatetime", "_AD_Phase"],
    sort=True,
    dropna=True
)

for (flight_start, phase), group in grouped:
    fi = flight_index[pd.Timestamp(flight_start)]

    for column in numeric_parameters:
        values = group[column].to_numpy(dtype=float)
        values = values[np.isfinite(values)]

        if values.size < AD_MIN_PHASE_SAMPLES:
            continue

        median = float(np.median(values))
        q10, q90 = np.percentile(values, [10, 90])
        mad = float(np.median(np.abs(values - median)))

        feature_records.append(
            {
                "FlightStart": pd.Timestamp(flight_start),
                "FlightIndex": fi,
                "FlightPhase": float(phase),
                "Parameter": column,
                "Feature": "LEVEL",
                "Value": median,
                "Samples": int(values.size)
            }
        )

        feature_records.append(
            {
                "FlightStart": pd.Timestamp(flight_start),
                "FlightIndex": fi,
                "FlightPhase": float(phase),
                "Parameter": column,
                "Feature": "SPREAD",
                "Value": float(q90 - q10),
                "Samples": int(values.size)
            }
        )

        feature_records.append(
            {
                "FlightStart": pd.Timestamp(flight_start),
                "FlightIndex": fi,
                "FlightPhase": float(phase),
                "Parameter": column,
                "Feature": "VARIABILITY",
                "Value": mad,
                "Samples": int(values.size)
            }
        )

        slope = ad_linear_slope(values)

        if np.isfinite(slope):
            feature_records.append(
                {
                    "FlightStart": pd.Timestamp(flight_start),
                    "FlightIndex": fi,
                    "FlightPhase": float(phase),
                    "Parameter": column,
                    "Feature": "WITHIN_PHASE_TREND",
                    "Value": slope,
                    "Samples": int(values.size)
                }
            )

pair_map = defaultdict(dict)

for column in numeric_parameters:
    channel, signal = ad_channel_signal(column)

    if channel in ("AC", "BC"):
        pair_map[signal][channel] = column

for signal, channels in pair_map.items():
    if "AC" not in channels or "BC" not in channels:
        continue

    ac = channels["AC"]
    bc = channels["BC"]

    pair_data = ad_df[
        [
            "StartDatetime",
            "_AD_Phase",
            ac,
            bc
        ]
    ].copy()

    pair_data["_AD_Delta"] = (
        pd.to_numeric(pair_data[ac], errors="coerce")
        - pd.to_numeric(pair_data[bc], errors="coerce")
    )

    for (flight_start, phase), group in pair_data.groupby(
        ["StartDatetime", "_AD_Phase"],
        sort=True,
        dropna=True
    ):
        values = group["_AD_Delta"].to_numpy(dtype=float)
        values = values[np.isfinite(values)]

        if values.size < AD_MIN_PHASE_SAMPLES:
            continue

        fi = flight_index[pd.Timestamp(flight_start)]

        median = float(np.median(values))
        q10, q90 = np.percentile(values, [10, 90])

        feature_records.append(
            {
                "FlightStart": pd.Timestamp(flight_start),
                "FlightIndex": fi,
                "FlightPhase": float(phase),
                "Parameter": f"AC-BC::{signal}",
                "Feature": "CHANNEL_OFFSET",
                "Value": median,
                "Samples": int(values.size)
            }
        )

        feature_records.append(
            {
                "FlightStart": pd.Timestamp(flight_start),
                "FlightIndex": fi,
                "FlightPhase": float(phase),
                "Parameter": f"AC-BC::{signal}",
                "Feature": "CHANNEL_SPREAD",
                "Value": float(q90 - q10),
                "Samples": int(values.size)
            }
        )

feature_df = pd.DataFrame(feature_records)

if feature_df.empty:
    raise RuntimeError(
        "No phase-conditioned features could be constructed from the available data."
    )

number_of_tests = max(len(feature_df), 1)
bonferroni_limit = AD_FWER_ALPHA / number_of_tests

candidate_records = []

for (parameter, phase, feature), group in feature_df.groupby(
    ["Parameter", "FlightPhase", "Feature"],
    sort=False
):
    group = (
        group
        .sort_values("FlightIndex", kind="stable")
        .reset_index(drop=True)
    )

    raw_history = []
    z_history = []

    ewma = 0.0
    positive_cusum = 0.0
    negative_cusum = 0.0

    for _, row in group.iterrows():
        value = float(row["Value"])

        history = np.asarray(
            raw_history[-AD_MAX_HISTORY:],
            dtype=float
        )

        history = history[np.isfinite(history)]

        if history.size < AD_MIN_HISTORY:
            raw_history.append(value)
            continue

        baseline, scale = ad_center_scale(history)

        z = (value - baseline) / max(scale, AD_EPS)
        z = float(np.clip(z, -100.0, 100.0))

        ewma = (
            AD_EWMA_LAMBDA * z
            + (1.0 - AD_EWMA_LAMBDA) * ewma
        )

        positive_cusum = max(
            0.0,
            positive_cusum + z - AD_CUSUM_K
        )

        negative_cusum = min(
            0.0,
            negative_cusum + z + AD_CUSUM_K
        )

        cusum = max(
            positive_cusum,
            abs(negative_cusum)
        )

        direction = 1 if z > 0 else -1 if z < 0 else 0

        recent = (
            z_history[-(AD_PERSISTENCE_WINDOW - 1):]
            + [z]
        )

        same_direction = [
            x
            for x in recent
            if np.isfinite(x)
            and abs(x) >= AD_DIRECTION_MIN_Z
            and np.sign(x) == direction
        ]

        persistence = len(same_direction)

        recent_slope = np.nan

        if len(recent) >= AD_PERSISTENCE_WINDOW:
            recent_slope = ad_linear_slope(recent)

        raw_p = ad_normal_two_sided_p(z)
        corrected_p = min(1.0, raw_p * number_of_tests)

        evidence = []

        if abs(z) >= AD_Z_STRONG:
            evidence.append("EXTREME_PHASE_DEVIATION")

        if abs(ewma) >= AD_EWMA_STRONG:
            evidence.append("PERSISTENT_EWMA_DRIFT")

        if cusum >= AD_CUSUM_STRONG:
            evidence.append("ACCUMULATED_CUSUM_DRIFT")

        if persistence >= AD_PERSISTENCE_MIN:
            evidence.append("MULTI_FLIGHT_PERSISTENCE")

        if (
            np.isfinite(recent_slope)
            and abs(recent_slope) >= AD_TREND_MIN
            and np.sign(recent_slope) == direction
        ):
            evidence.append("CONSISTENT_DEGRADATION_TREND")

        multiplicity_pass = raw_p <= bonferroni_limit

        exceptional_pass = (
            abs(z) >= AD_Z_EXTREME
            and multiplicity_pass
            and len(evidence) >= 2
        )

        persistent_pass = (
            abs(z) >= AD_Z_STRONG
            and multiplicity_pass
            and persistence >= AD_PERSISTENCE_MIN
            and len(evidence) >= 3
            and (
                abs(ewma) >= AD_EWMA_STRONG
                or cusum >= AD_CUSUM_STRONG
            )
        )

        gradual_pass = (
            multiplicity_pass
            and persistence >= AD_PERSISTENCE_MIN
            and abs(ewma) >= AD_EWMA_STRONG
            and cusum >= AD_CUSUM_STRONG
            and len(evidence) >= 3
        )

        final_pass = (
            exceptional_pass
            or persistent_pass
            or gradual_pass
        )

        if final_pass:
            pct = ad_pct(value, baseline)

            if np.isfinite(pct):
                deviation_text = f"{pct:+.3f}%"
            else:
                deviation_text = "N/A"

            direction_text = (
                "above"
                if value > baseline
                else "below"
            )

            reason_parts = [
                f"{abs(z):.2f} robust σ {direction_text} same-phase baseline"
            ]

            if np.isfinite(pct):
                reason_parts.append(
                    f"{abs(pct):.3f}% magnitude deviation"
                )

            if persistence >= AD_PERSISTENCE_MIN:
                reason_parts.append(
                    f"same-direction evidence in "
                    f"{persistence}/{min(len(recent), AD_PERSISTENCE_WINDOW)} "
                    f"recent comparable flights"
                )

            if abs(ewma) >= AD_EWMA_STRONG:
                reason_parts.append(
                    f"EWMA={ewma:+.2f}σ"
                )

            if cusum >= AD_CUSUM_STRONG:
                reason_parts.append(
                    f"CUSUM={cusum:.2f}"
                )

            if (
                np.isfinite(recent_slope)
                and abs(recent_slope) >= AD_TREND_MIN
            ):
                reason_parts.append(
                    f"trend={recent_slope:+.2f}σ/flight"
                )

            reason_parts.append(
                f"multiplicity-corrected p≈{corrected_p:.2e}"
            )

            candidate_records.append(
                {
                    "FlightStart": row["FlightStart"],
                    "FlightIndex": int(row["FlightIndex"]),
                    "FlightPhase": row["FlightPhase"],
                    "Parameter": parameter,
                    "Feature": feature,
                    "Observed": value,
                    "Baseline": baseline,
                    "DeviationPct": pct,
                    "RobustZ": z,
                    "EWMA": ewma,
                    "CUSUM": cusum,
                    "Persistence": persistence,
                    "TrendSigmaPerFlight": recent_slope,
                    "RawP": raw_p,
                    "CorrectedP": corrected_p,
                    "Consensus": len(evidence),
                    "Evidence": " + ".join(evidence),
                    "Reason": "; ".join(reason_parts)
                }
            )

        raw_history.append(value)
        z_history.append(z)

anomaly_detail_df = pd.DataFrame(candidate_records)

        report_df["FlightPhase"]
        .map(lambda x: f"{x:g}")
    )

    report_df["Observed"] = report_df["Observed"].map(
        lambda x: f"{x:,.6g}"
    )

    report_df["Baseline"] = report_df["Baseline"].map(
        lambda x: f"{x:,.6g}"
    )

    report_df["Deviation"] = report_df["DeviationPct"].map(
        lambda x: (
            f"{x:+.3f}%"
            if np.isfinite(x)
            else "N/A"
        )
    )

    report_df["Robust σ"] = report_df["RobustZ"].map(
        lambda x: f"{x:+.2f}"
    )

    report_df["EWMA"] = report_df["EWMA"].map(
        lambda x: f"{x:+.2f}"
    )

    report_df["CUSUM"] = report_df["CUSUM"].map(
        lambda x: f"{x:.2f}"
    )

    report_df["Persistence"] = report_df["Persistence"].map(
        lambda x: f"{int(x)}/{AD_PERSISTENCE_WINDOW}"
    )

    report_df["Corrected p"] = report_df["CorrectedP"].map(
        lambda x: f"{x:.2e}"
    )

    report_df = report_df[
        [
            "Date / Flight",
            "Flight Phase",
            "Parameter",
            "Feature",
            "Confidence",
            "Observed",
            "Baseline",
            "Deviation",
            "Robust σ",
            "EWMA",
            "CUSUM",
            "Persistence",
            "Corrected p",
            "Reason"
        ]
    ]

    print()
    print("=" * 120)
    print("HIGH-CONFIDENCE PHASE-CONDITIONED ANOMALY REPORT")
    print("=" * 120)
    print(f"Aircraft:                 {globals().get('selected_acid', 'N/A')}")
    print(f"Engine:                   {globals().get('selected_esn', 'N/A')}")
    print(f"Flights analysed:         {len(flight_starts):,}")
    print(f"Numeric parameters:       {len(numeric_parameters):,}")
    print(f"Statistical tests:        {number_of_tests:,}")
    print(f"Family-wise alpha:        {AD_FWER_ALPHA}")
    print(f"Per-test threshold:       {bonferroni_limit:.3e}")
    print(f"Confirmed anomaly rows:   {len(report_df):,}")
    print(
        f"Anomalous flights:        "
        f"{anomaly_detail_df['FlightStart'].nunique():,}"
    )
    print()

    for flight, rows in anomaly_detail_df.groupby(
        "FlightStart",
        sort=True
    ):
        print("-" * 120)
        print(
            "FLIGHT: "
            + pd.Timestamp(flight).strftime("%Y-%m-%d %H:%M:%S")
        )

        top_rows = rows.head(AD_MAX_LABELS)

        for _, r in top_rows.iterrows():
            name = display_parameter_name(r["Parameter"])

            pct = (
                f"{r['DeviationPct']:+.3f}%"
                if np.isfinite(r["DeviationPct"])
                else "N/A"
            )

            print(
                f"  [{r['Confidence']}] "
                f"{name} | "
                f"Phase {r['FlightPhase']:g} | "
                f"{r['Feature']} | "
                f"Deviation {pct}"
            )

            print(
                f"      {r['Reason']}"
            )

    print()
    print("=" * 120)
    print("COLUMNIZED ANOMALY TABLE")
    print("=" * 120)

    table_to_show = report_df.head(AD_MAX_TABLE_ROWS)

    try:
        display(table_to_show)
    except Exception:
        try:
            from IPython.display import display as ipy_display
            ipy_display(table_to_show)
        except Exception:
            print(table_to_show.to_string(index=False))

    if len(report_df) > AD_MAX_TABLE_ROWS:
        print(
            f"Showing the first {AD_MAX_TABLE_ROWS:,} of "
            f"{len(report_df):,} confirmed anomaly rows."
        )

confirmed_flights = set()

if not anomaly_detail_df.empty:
    confirmed_flights = set(
        pd.to_datetime(
            anomaly_detail_df["FlightStart"]
        )
    )

plot_parameter_names = []

if "selected_parameters" in globals():
    for p in selected_parameters:
        candidate = str(p)

        prefix = globals().get("PARAMETER_PREFIX", "")

        if candidate in ad_df.columns:
            plot_parameter_names.append(candidate)

        elif prefix + candidate in ad_df.columns:
            plot_parameter_names.append(prefix + candidate)

if not plot_parameter_names and not anomaly_detail_df.empty:
    for parameter in anomaly_detail_df["Parameter"].unique():
        if str(parameter).startswith("AC-BC::"):
            signal = str(parameter).split("::", 1)[1]

            for channel in ("AC", "BC"):
                prefix = globals().get("PARAMETER_PREFIX", "")
                candidate = prefix + channel + "_" + signal

                if candidate in ad_df.columns:
                    plot_parameter_names.append(candidate)

        elif parameter in ad_df.columns:
            plot_parameter_names.append(parameter)

plot_parameter_names = list(dict.fromkeys(plot_parameter_names))

if plot_parameter_names:
    flight_plot_info = []

    cursor = 0.0
    visual_gap = 120.0

    for flight in flight_starts:
        fdf = ad_df.loc[
            ad_df["StartDatetime"] == flight
        ]

        if fdf.empty:
            continue

        start = fdf["Timestamp"].min()
        end = fdf["Timestamp"].max()

        duration = max(
            (end - start).total_seconds(),
            1.0
        )

        flight_plot_info.append(
            {
                "FlightStart": flight,
                "Start": start,
                "End": end,
                "PlotStart": cursor,
                "PlotEnd": cursor + duration
            }
        )

        cursor += duration + visual_gap

    plot_frames = []

    for info in flight_plot_info:
        fdf = ad_df.loc[
            ad_df["StartDatetime"] == info["FlightStart"]
        ].copy()

        fdf["_AD_PlotX"] = (
            info["PlotStart"]
            + (
                fdf["Timestamp"] - info["Start"]
            ).dt.total_seconds()
        )

        plot_frames.append(fdf)

    if plot_frames:
        plot_df = pd.concat(
            plot_frames,
            ignore_index=True
        )

        signal_groups = defaultdict(list)

        for column in plot_parameter_names:
            channel, signal = ad_channel_signal(column)
            signal_groups[signal].append(
                (channel, column)
            )

        nplots = len(signal_groups)

        fig, axes = plt.subplots(
            nplots,
            1,
            figsize=(20, max(5, 3.3 * nplots)),
            sharex=True,
            squeeze=False
        )

        axes = axes[:, 0]

        for axis, (signal, members) in zip(
            axes,
            signal_groups.items()
        ):
            for channel, column in members:
                first = True

                for info in flight_plot_info:
                    segment = plot_df.loc[
                        plot_df["StartDatetime"]
                        == info["FlightStart"],
                        ["_AD_PlotX", column]
                    ].copy()

                    segment[column] = pd.to_numeric(
                        segment[column],
                        errors="coerce"
                    )

                    segment = segment.dropna()

                    if segment.empty:
                        continue

                    axis.plot(
                        segment["_AD_PlotX"],
                        segment[column],
                        linewidth=1.0,
                        label=(
                            channel
                            if channel and first
                            else (
                                ad_short_name(column)
                                if first
                                else "_nolegend_"
                            )
                        )
                    )

                    first = False

            for info in flight_plot_info:
                if info["FlightStart"] in confirmed_flights:
                    axis.axvline(
                        info["PlotStart"],
                        color="white",
                        linewidth=2.0,
                        alpha=0.95,
                        zorder=20
                    )

            axis.set_ylabel(signal)
            axis.grid(True, alpha=0.20)

            handles, labels = axis.get_legend_handles_labels()

            if handles:
                axis.legend(
                    loc="upper right",
                    framealpha=0.85
                )

        tick_positions = []
        tick_labels = []

        for info in flight_plot_info:
            tick_positions.append(
                (
                    info["PlotStart"]
                    + info["PlotEnd"]
                ) / 2.0
            )

            tick_labels.append(
                info["FlightStart"].strftime("%Y-%m-%d\n%H:%M")
            )

        axes[-1].set_xticks(tick_positions)
        axes[-1].set_xticklabels(
            tick_labels,
            rotation=45,
            ha="right",
            fontsize=7
        )

        axes[-1].set_xlabel(
            "Flight timeline — inter-flight calendar gaps compressed"
        )

        fig.suptitle(
            "High-Confidence FlightPhase-Conditioned Anomaly Detection",
            fontsize=15
        )

        fig.tight_layout(
            rect=[0.02, 0.03, 0.99, 0.96]
        )

        plt.show()
        plt.close(fig)
