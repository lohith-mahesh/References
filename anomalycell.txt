def ad_short(column):
    value = str(column)
    return value[len(prefix):] if prefix and value.startswith(prefix) else value

def ad_split_channel(column):
    value = ad_short(column)
    if value.startswith("AC_"):
        return "AC", value[3:]
    if value.startswith("BC_"):
        return "BC", value[3:]
    return "", value

def ad_robust_center_scale(values):
    x = np.asarray(values, dtype=float)
    x = x[np.isfinite(x)]
    if x.size == 0:
        return np.nan, np.nan
    center = float(np.median(x))
    mad = float(np.median(np.abs(x - center)))
    scale = 1.4826 * mad
    if not np.isfinite(scale) or scale <= AD_EPS:
        q25, q75 = np.percentile(x, [25, 75])
        scale = float((q75 - q25) / 1.349)
    if not np.isfinite(scale) or scale <= AD_EPS:
        scale = float(np.std(x))
    if not np.isfinite(scale) or scale <= AD_EPS:
        scale = AD_EPS
    return center, scale

def ad_normal_p(z):
    if not np.isfinite(z):
        return 1.0
    return float(math.erfc(abs(float(z)) / math.sqrt(2.0)))

def ad_pct(value, baseline):
    if not np.isfinite(value) or not np.isfinite(baseline) or abs(baseline) <= AD_EPS:
        return np.nan
    return 100.0 * (value - baseline) / abs(baseline)

def ad_linear_slope(values):
    y = np.asarray(values, dtype=float)
    mask = np.isfinite(y)
    if mask.sum() < 3:
        return np.nan
    y = y[mask]
    x = np.arange(len(y), dtype=float)
    x -= x.mean()
    y -= y.mean()
    denom = float(np.dot(x, x))
    if denom <= AD_EPS:
        return np.nan
    return float(np.dot(x, y) / denom)

def ad_theilsen(values):
    y = np.asarray(values, dtype=float)
    y = y[np.isfinite(y)]
    if len(y) < 3:
        return np.nan
    slopes = []
    for i in range(len(y) - 1):
        for j in range(i + 1, len(y)):
            slopes.append((y[j] - y[i]) / (j - i))
    return float(np.median(slopes)) if slopes else np.nan

def ad_directionality(values):
    y = np.asarray(values, dtype=float)
    y = y[np.isfinite(y)]
    if len(y) < 3:
        return 0.0, 0
    d = np.diff(y)
    pos = int(np.sum(d > 0))
    neg = int(np.sum(d < 0))
    total = pos + neg
    if total == 0:
        return 0.0, 0
    if pos >= neg:
        return pos / total, 1
    return neg / total, -1

def ad_mann_kendall(values):
    y = np.asarray(values, dtype=float)
    y = y[np.isfinite(y)]
    n = len(y)
    if n < 5:
        return 0.0, 1.0
    s = 0
    for i in range(n - 1):
        s += int(np.sum(np.sign(y[i + 1:] - y[i])))
    _, counts = np.unique(y, return_counts=True)
    tie_term = float(np.sum(counts * (counts - 1) * (2 * counts + 5)))
    var_s = (n * (n - 1) * (2 * n + 5) - tie_term) / 18.0
    if var_s <= 0:
        return 0.0, 1.0
    if s > 0:
        z = (s - 1) / math.sqrt(var_s)
    elif s < 0:
        z = (s + 1) / math.sqrt(var_s)
    else:
        z = 0.0
    p = math.erfc(abs(z) / math.sqrt(2.0))
    return float(z), float(p)

def ad_feature_rows(frame, parameter, flight_start, flight_idx, phase):
    values = pd.to_numeric(frame[parameter], errors="coerce").to_numpy(dtype=float)
    values = values[np.isfinite(values)]
    if values.size < AD_MIN_PHASE_SAMPLES:
        return []
    median = float(np.median(values))
    q10, q90 = np.percentile(values, [10, 90])
    mad = float(np.median(np.abs(values - median)))
    slope = ad_linear_slope(values)
    result = [
        (parameter, "LEVEL", median),
        (parameter, "SPREAD", float(q90 - q10)),
        (parameter, "VARIABILITY", mad)
    ]
    if np.isfinite(slope):
        result.append((parameter, "WITHIN_PHASE_TREND", slope))
    return [
        {
            "FlightStart": pd.Timestamp(flight_start),
            "FlightIndex": int(flight_idx),
            "FlightPhase": float(phase),
            "Parameter": p,
            "Feature": f,
            "Value": float(v),
            "Samples": int(values.size)
        }
        for p, f, v in result
    ]

    values = series["Value"].to_numpy(dtype=float)
    z_values = np.clip(
        (values - baseline_center) / max(baseline_scale, AD_EPS),

    )

    ewma = 0.0
    cplus = 0.0
    cminus = 0.0
    ph_plus = 0.0
    ph_minus = 0.0

    gradual_streak = 0
    gradual_direction = 0

    for i in range(baseline_count, len(series)):
        row = series.iloc[i]
        z = float(z_values[i])

        ewma = (
            AD_EWMA_LAMBDA * z
            + (1.0 - AD_EWMA_LAMBDA) * ewma
        )

        cplus = max(
            0.0,
            cplus + z - AD_CUSUM_K
        )

        cminus = min(
            0.0,
            cminus + z + AD_CUSUM_K
        )

        cusum = max(cplus, abs(cminus))
        cusum_direction = 1 if cplus >= abs(cminus) else -1

        ph_plus = max(
            0.0,
            ph_plus + z - AD_PH_DELTA
        )

        ph_minus = max(
            0.0,
            ph_minus - z - AD_PH_DELTA
        )

        ph_score = max(ph_plus, ph_minus)
        ph_direction = 1 if ph_plus >= ph_minus else -1

        raw_p = ad_normal_p(z)
        abrupt = (
            abs(z) >= AD_ABRUPT_Z
            and raw_p <= abrupt_p_limit
        )

        shift_start = max(
            baseline_count,
            i - AD_SHIFT_COUNT + 1
        )

        shift_z = z_values[shift_start:i + 1]

        persistent_shift = (
            len(shift_z) >= AD_SHIFT_COUNT
            and np.all(np.abs(shift_z) >= AD_SHIFT_Z)
            and (
                np.all(shift_z > 0)
                or np.all(shift_z < 0)
            )
        )

        gradual = False
        theilsen = np.nan
        direction_fraction = np.nan
        direction = 0
        net_z = np.nan
        mk_z = np.nan
        mk_p = np.nan

        if i - baseline_count + 1 >= AD_SEQ_WINDOW:
            recent_z = z_values[i - AD_SEQ_WINDOW + 1:i + 1]

            theilsen = ad_theilsen(recent_z)
            direction_fraction, direction = ad_directionality(recent_z)
            net_z = float(recent_z[-1] - recent_z[0])
            mk_z, mk_p = ad_mann_kendall(recent_z)

            trend_direction_ok = (
                direction != 0
                and np.sign(net_z) == direction
                and np.sign(theilsen) == direction
                and np.sign(mk_z) == direction
            )

            sequential_direction_ok = (
                cusum_direction == direction
                and ph_direction == direction
                and np.sign(ewma) == direction
            )

            evidence_count = sum(
                [
                    direction_fraction >= AD_DIRECTIONAL_FRACTION,
                    abs(net_z) >= AD_MIN_NET_Z,
                    abs(theilsen) >= AD_MIN_THEILSEN,
                    mk_p <= AD_MK_P,
                    abs(ewma) >= AD_EWMA_WARN,
                    cusum >= AD_CUSUM_WARN,
                    ph_score >= AD_PH_WARN
                ]
            )

            gradual = (
                trend_direction_ok
                and sequential_direction_ok
                and evidence_count >= 6
                and (
                    abs(ewma) >= AD_EWMA_STRONG
                    or cusum >= AD_CUSUM_STRONG
                    or ph_score >= AD_PH_STRONG
                )
            )

        if gradual:
            if direction == gradual_direction:
                gradual_streak += 1
            else:
                gradual_direction = direction
                gradual_streak = 1
        else:
            gradual_streak = 0
            gradual_direction = 0

        confirmed_gradual = gradual_streak >= AD_CONFIRMATIONS

        detection_type = None

        if abrupt:
            detection_type = "ABRUPT EXTREME"
        elif persistent_shift:
            detection_type = "PERSISTENT LEVEL SHIFT"
        elif confirmed_gradual:
            detection_type = "GRADUAL DEGRADATION"

        if detection_type is None:
            continue

        pct = ad_pct(float(row["Value"]), baseline_center)

        evidence = []

        evidence.append(
            f"same-phase deviation {z:+.2f} robust sigma"
        )

        if np.isfinite(pct):
            evidence.append(
                f"{pct:+.3f}% from frozen healthy baseline"
            )

        if abrupt:
            evidence.append(
                f"family-wise corrected abrupt-event p evidence {raw_p:.2e}"
            )

        if persistent_shift:
            evidence.append(
                f"{AD_SHIFT_COUNT} consecutive same-direction >= {AD_SHIFT_Z:.1f} sigma observations"
            )

        if confirmed_gradual:
            evidence.extend(
                [
                    f"{AD_SEQ_WINDOW}-flight Theil-Sen trend {theilsen:+.3f} sigma/flight",
                    f"directional consistency {direction_fraction:.0%}",
                    f"net movement {net_z:+.2f} sigma",
                    f"Mann-Kendall p={mk_p:.3g}",
                    f"EWMA {ewma:+.2f} sigma",
                    f"CUSUM {cusum:.2f}",
                    f"Page-Hinkley evidence {ph_score:.2f}",
                    f"confirmed across {gradual_streak} successive evaluations"
                ]
            )

        strength_votes = 0

        if abs(z) >= 6.0:
            strength_votes += 1
        if abs(ewma) >= AD_EWMA_STRONG:
            strength_votes += 1
        if cusum >= AD_CUSUM_STRONG:
            strength_votes += 1
        if ph_score >= AD_PH_STRONG:
            strength_votes += 1
        if np.isfinite(mk_p) and mk_p <= 0.02:
            strength_votes += 1
        if np.isfinite(direction_fraction) and direction_fraction >= 0.85:
            strength_votes += 1

        if detection_type == "ABRUPT EXTREME":
            confidence = "VERY HIGH"
        elif detection_type == "PERSISTENT LEVEL SHIFT" and strength_votes >= 2:
            confidence = "VERY HIGH"
        elif detection_type == "GRADUAL DEGRADATION" and strength_votes >= 4:
            confidence = "VERY HIGH"
        else:
            confidence = "HIGH"

        detections.append(
            {
                "FlightStart": pd.Timestamp(row["FlightStart"]),
                "FlightIndex": int(row["FlightIndex"]),
                "FlightPhase": float(row["FlightPhase"]),
                "Parameter": parameter,
                "Feature": feature,
                "Detection": detection_type,
                "Confidence": confidence,
                "Observed": float(row["Value"]),
                "Baseline": float(baseline_center),
                "DeviationPct": pct,

                "EWMA": ewma,
                "CUSUM": cusum,
                "PageHinkley": ph_score,
                "TrendSigmaPerFlight": theilsen,
                "DirectionalConsistency": direction_fraction,
                "MannKendallP": mk_p,

            }

                    + x["MaxCUSUM"] / 10.0
                    + x["MaxPH"] / 10.0
                    + x["MaxZ"] / 6.0
            )
            .sort_values("Priority", ascending=False)
        )

        keep_signals = set(
            signal_priority.head(
                AD_MAX_PLOT_SIGNALS
            ).index
        )

        signal_columns = {
            k: v
            for k, v in signal_columns.items()
            if k in keep_signals
        }

    flight_info = []
    cursor = 0.0

    for flight in flight_starts:
        subset = ad_df.loc[
            ad_df["StartDatetime"] == flight
        ]

        if subset.empty:
            continue

        start = subset["Timestamp"].min()
        end = subset["Timestamp"].max()

        duration = max(
            (end - start).total_seconds(),
            1.0
        )

        flight_info.append(
            {
                "FlightStart": flight,
                "ActualStart": start,
                "ActualEnd": end,
                "PlotStart": cursor,
                "PlotEnd": cursor + duration
            }
        )

        cursor += duration + AD_COMPRESSED_GAP

    plot_frames = []

    for info in flight_info:
        subset = ad_df.loc[
            ad_df["StartDatetime"] == info["FlightStart"]
        ].copy()

        subset["_AD_PlotX"] = (
            info["PlotStart"]
            + (
                subset["Timestamp"]
                - info["ActualStart"]
            ).dt.total_seconds()
        )

        plot_frames.append(subset)

    if plot_frames and signal_columns:
        plot_df = pd.concat(
            plot_frames,
            ignore_index=True
        )

        flight_lookup = {
            x["FlightStart"]: x
            for x in flight_info
        }

        anomaly_markers = defaultdict(dict)

        for _, row in anomaly_detail_df.iterrows():
            signal = row["_Signal"]

            if signal not in signal_columns:
                continue

            flight = pd.Timestamp(
                row["FlightStart"]
            )

            phase = float(
                row["FlightPhase"]
            )

            phase_rows = plot_df.loc[
                (
                    plot_df["StartDatetime"]
                    == flight
                )
                & (
                    pd.to_numeric(
                        plot_df["_AD_Phase"],
                        errors="coerce"
                    )
                    == phase
                )
            ]

            if phase_rows.empty:
                if flight not in flight_lookup:
                    continue
                x = flight_lookup[flight]["PlotStart"]
                timestamp = flight
            else:
                phase_rows = phase_rows.sort_values(
                    "Timestamp",
                    kind="stable"
                )
                middle = phase_rows.iloc[
                    len(phase_rows) // 2
                ]
                x = float(
                    middle["_AD_PlotX"]
                )
                timestamp = pd.Timestamp(
                    middle["Timestamp"]
                )

            key = (
                flight,
                phase
            )

            anomaly_markers[signal][key] = {
                "x": x,
                "timestamp": timestamp,
                "flight": flight,
                "phase": phase
            }

        fig, axes = plt.subplots(
            len(signal_columns),
            1,
            figsize=(
                22,
                max(
                    6,
                    3.4 * len(signal_columns)
                )
            ),
            sharex=True,
            squeeze=False
        )

        axes = axes[:, 0]

        channel_colors = globals().get(
            "CHANNEL_COLORS",
            {
                "AC": "#1f77b4",
                "BC": "#ff7f0e"
            }
        )

        for axis, (signal, members) in zip(
            axes,
            signal_columns.items()
        ):
            for channel, column in members:
                label_pending = True

                for info in flight_info:
                    segment = plot_df.loc[
                        plot_df["StartDatetime"]
                        == info["FlightStart"],
                        [
                            "Timestamp",
                            "_AD_PlotX",
                            column
                        ]
                    ].copy()

                    segment[column] = pd.to_numeric(
                        segment[column],
                        errors="coerce"
                    )

                    segment = (
                        segment
                        .dropna(
                            subset=[
                                "_AD_PlotX",
                                column
                            ]
                        )
                        .sort_values(
                            "Timestamp",
                            kind="stable"
                        )
                    )

                    if segment.empty:
                        continue

                    unique_values = (
                        segment[column]
                        .dropna()
                        .unique()
                    )

                    discrete = (
                        len(unique_values) <= 12
                        and np.allclose(
                            unique_values,
                            np.round(unique_values),
                            equal_nan=True
                        )
                    )

                    positive_deltas = (
                        segment["Timestamp"]
                        .diff()
                        .dt.total_seconds()
                    )

                    cadence_values = positive_deltas[
                        positive_deltas > 0
                    ]

                    cadence = (
                        float(cadence_values.median())
                        if not cadence_values.empty
                        else 1.0
                    )

                    gap_threshold = max(
                        3.0 * cadence,
                        5.0
                    )

                    chunk_ids = (
                        positive_deltas
                        .gt(gap_threshold)
                        .cumsum()
                    )

                    for _, chunk in segment.groupby(
                        chunk_ids,
                        sort=True
                    ):
                        axis.plot(
                            chunk["_AD_PlotX"],
                            chunk[column],
                            linewidth=1.15,
                            color=channel_colors.get(
                                channel,
                                None
                            ),
                            drawstyle=(
                                "steps-post"
                                if discrete
                                else "default"
                            ),
                            label=(
                                channel or ad_short(column)
                                if label_pending
                                else "_nolegend_"
                            ),
                            zorder=4
                        )

                        label_pending = False

            for marker in anomaly_markers.get(
                signal,
                {}
            ).values():
                axis.axvline(
                    marker["x"],
                    color="white",
                    linewidth=2.0,
                    alpha=0.95,
                    zorder=20
                )

            signal_detections = anomaly_detail_df.loc[
                anomaly_detail_df["_Signal"]
                == signal
            ]

            gradual_count = int(
                (
                    signal_detections["Detection"]
                    == "GRADUAL DEGRADATION"
                ).sum()
            )

            shift_count = int(
                (
                    signal_detections["Detection"]
                    == "PERSISTENT LEVEL SHIFT"
                ).sum()
            )

            abrupt_count = int(
                (
                    signal_detections["Detection"]
                    == "ABRUPT EXTREME"
                ).sum()
            )

            axis.set_title(
                f"{signal} | gradual={gradual_count}, "
                f"shift={shift_count}, abrupt={abrupt_count}",
                loc="left",
                fontsize=10
            )

            axis.set_ylabel(
                signal,
                fontsize=9
            )

            axis.grid(
                True,
                alpha=0.20
            )

            axis.margins(
                y=0.08
            )

            handles, labels = (
                axis.get_legend_handles_labels()
            )

            if handles:
                axis.legend(
                    handles,
                    labels,
                    loc="upper right",
                    framealpha=0.85
                )

        tick_positions = []
        tick_labels = []

        for info in flight_info:
            tick_positions.append(
                (
                    info["PlotStart"]
                    + info["PlotEnd"]
                ) / 2.0
            )

            tick_labels.append(
                info["FlightStart"].strftime(
                    "%Y-%m-%d\n%H:%M"
                )
            )

        axes[-1].set_xticks(
            tick_positions
        )

        axes[-1].set_xticklabels(
            tick_labels,
            rotation=45,
            ha="right",
            fontsize=7
        )

        axes[-1].set_xlabel(
            "Flight timeline - calendar gaps compressed"
        )

        if flight_info:
            axes[-1].set_xlim(
                flight_info[0]["PlotStart"],
                flight_info[-1]["PlotEnd"]
            )

        fig.suptitle(
            "Confirmed Phase-Conditioned Engine Anomalies",
            fontsize=15
        )

        fig.text(
            0.5,
            0.012,
            "White line = centre of the FlightPhase interval in which the confirmed statistical anomaly was identified",
            ha="center",
            fontsize=9
        )

        fig.tight_layout(
            rect=[
                0.02,
                0.035,
                0.99,
                0.965
            ]
        )

        plt.show()
        plt.close(fig)
