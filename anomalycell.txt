def _ad_short_name(column):
    if column.startswith(PARAMETER_PREFIX):
        return column[len(PARAMETER_PREFIX):]
    return column

def _ad_channel_signal(column):
    short = _ad_short_name(column)
    if short.startswith("AC_"):
        return "AC", short[3:]
    if short.startswith("BC_"):
        return "BC", short[3:]
    return "", short

def _ad_robust_scale(values):
    x = np.asarray(values, dtype=float)
    x = x[np.isfinite(x)]
    if len(x) == 0:
        return np.nan, np.nan
    center = float(np.median(x))
    mad = float(np.median(np.abs(x - center)))
    scale = 1.4826 * mad
    if not np.isfinite(scale) or scale < AD_EPS:
        q25, q75 = np.percentile(x, [25, 75])
        scale = float((q75 - q25) / 1.349)
    if not np.isfinite(scale) or scale < AD_EPS:
        std = float(np.std(x))
        scale = std if np.isfinite(std) and std >= AD_EPS else AD_EPS
    return center, scale

def _ad_slope(values):
    y = np.asarray(values, dtype=float)
    valid = np.isfinite(y)
    if valid.sum() < 3:
        return np.nan
    y = y[valid]
    x = np.arange(len(y), dtype=float)
    x = x - x.mean()
    denom = float(np.dot(x, x))
    if denom <= AD_EPS:
        return np.nan
    return float(np.dot(x, y - y.mean()) / denom)

def _ad_pct(value, baseline):
    if not np.isfinite(value) or not np.isfinite(baseline):
        return np.nan
    if abs(baseline) > AD_EPS:
        return 100.0 * (value - baseline) / abs(baseline)
    return np.nan

        slope = _ad_slope(values)

        feature_rows.extend(

                    "FlightPhase": float(phase),

                    "Metric": "median",

                    "Samples": len(values),

                {
                    "FlightStart": pd.Timestamp(flight_start),
                    "FlightIndex": flight_index,
                    "FlightPhase": float(phase),
                    "Parameter": column,
                    "Metric": "spread",
                    "Value": q90 - q10,

                    "FlightIndex": flight_index,

scored_rows = []

for key, group in feature_df.groupby(
    ["Parameter", "FlightPhase", "Metric"],
    sort=False,
):
    group = group.sort_values("FlightIndex", kind="stable").reset_index(drop=True)

    history_values = []
    standardized_history = []
    ewma = 0.0
    cusum_positive = 0.0
    cusum_negative = 0.0

    for _, row in group.iterrows():
        value = float(row["Value"])

        history = np.asarray(
            history_values[-AD_BASELINE_MAX_FLIGHTS:],
            dtype=float,
        )
        history = history[np.isfinite(history)]

        if len(history) < AD_MIN_FEATURE_HISTORY:
            history_values.append(value)
            continue

        baseline, scale = _ad_robust_scale(history)
        robust_z = (value - baseline) / max(scale, AD_EPS)
        robust_z = float(np.clip(robust_z, -50.0, 50.0))

        ewma = (
            AD_EWMA_LAMBDA * robust_z
            + (1.0 - AD_EWMA_LAMBDA) * ewma
        )

        cusum_positive = max(
            0.0,
            cusum_positive + robust_z - AD_CUSUM_ALLOWANCE,
        )
        cusum_negative = min(
            0.0,
            cusum_negative + robust_z + AD_CUSUM_ALLOWANCE,
        )
        cusum_strength = max(
            cusum_positive,
            abs(cusum_negative),
        )

        recent_z = standardized_history[-(AD_TREND_WINDOW - 1):] + [robust_z]
        trend_strength = 0.0
        trend_direction = 0.0

        if len(recent_z) >= AD_TREND_WINDOW:
            recent_array = np.asarray(recent_z, dtype=float)
            x = np.arange(len(recent_array), dtype=float)
            trend_direction = float(np.polyfit(x, recent_array, 1)[0])
            trend_strength = abs(trend_direction) * AD_TREND_WINDOW

        point_component = max(
            0.0,
            abs(robust_z) - AD_ROBUST_Z_THRESHOLD,
        )
        ewma_component = max(
            0.0,
            abs(ewma) - AD_EWMA_THRESHOLD,
        )
        cusum_component = max(
            0.0,
            cusum_strength - AD_CUSUM_THRESHOLD,
        )
        trend_component = max(
            0.0,
            trend_strength - AD_TREND_THRESHOLD,
        )

        anomaly_score = (
            1.50 * point_component
            + 2.25 * ewma_component
            + 2.75 * cusum_component
            + 1.75 * trend_component
        )

        percentage_deviation = _ad_pct(value, baseline)

        reasons = []

        if abs(robust_z) >= AD_ROBUST_Z_THRESHOLD:
            if np.isfinite(percentage_deviation):
                reasons.append(
                    f"phase-conditioned value deviated {percentage_deviation:+.2f}% "
                    f"from its prior-flight baseline ({robust_z:+.2f} robust σ)"
                )
            else:
                reasons.append(
                    f"phase-conditioned value shifted {robust_z:+.2f} robust σ "
                    f"from its prior-flight baseline"
                )

        if abs(ewma) >= AD_EWMA_THRESHOLD:
            reasons.append(
                f"persistent EWMA drift reached {ewma:+.2f} σ"
            )

        if cusum_strength >= AD_CUSUM_THRESHOLD:
            direction = "upward" if cusum_positive >= abs(cusum_negative) else "downward"
            reasons.append(
                f"small repeated {direction} deviations accumulated to "
                f"CUSUM {cusum_strength:.2f}"
            )

        if trend_strength >= AD_TREND_THRESHOLD:
            direction = "increasing" if trend_direction > 0 else "decreasing"
            reasons.append(
                f"{AD_TREND_WINDOW}-flight standardized trend is persistently "
                f"{direction} ({trend_direction:+.3f} σ/flight)"
            )

        if anomaly_score >= AD_SCORE_THRESHOLD:
            scored_rows.append(
                {
                    "FlightStart": row["FlightStart"],
                    "FlightIndex": int(row["FlightIndex"]),
                    "FlightPhase": row["FlightPhase"],
                    "Parameter": row["Parameter"],
                    "Metric": row["Metric"],
                    "Value": value,
                    "Baseline": baseline,
                    "RobustZ": robust_z,
                    "PercentageDeviation": percentage_deviation,
                    "EWMA": ewma,
                    "CUSUM": cusum_strength,
                    "Trend": trend_direction,
                    "AnomalyScore": anomaly_score,
                    "Reason": "; ".join(reasons),
                }
            )

        standardized_history.append(robust_z)
        history_values.append(value)

        anomaly_detail_df[report_columns]

            ["FlightStart", "AnomalyScore"],
            ascending=[True, False],

    )
