    partition_filter = F.lit(False)

)

def _binary_state(series):
    normalized = series.astype("string").str.strip().str.lower()

    mapped = normalized.map(
        {
            "0": 0,
            "0.0": 0,
            "false": 0,
            "1": 1,
            "1.0": 1,
            "true": 1,
        }
    )

    numeric = pd.to_numeric(normalized, errors="coerce")
    mapped = mapped.fillna(numeric.where(numeric.isin([0, 1])))

    return mapped.astype("Int64")


def _engine_state_changes(day_df, engine_esn):
    engine_df = (
        day_df.loc[
            day_df["AssetIdentifier"] == engine_esn,
            ["Timestamp", TR_UNLOCKED_AC_COLUMN],
        ]
        .dropna(subset=["Timestamp"])
        .sort_values("Timestamp", kind="stable")
        .copy()
    )

    engine_df["State"] = _binary_state(engine_df[TR_UNLOCKED_AC_COLUMN])
    engine_df = engine_df.dropna(subset=["State"]).copy()

    if engine_df.empty:
        return pd.DataFrame(columns=["Timestamp", "Transition"])

    previous_state = engine_df["State"].shift(1)
    transition_mask = previous_state.notna() & engine_df["State"].ne(previous_state)

    transitions = engine_df.loc[
        transition_mask,
        ["Timestamp", "State"],
    ].copy()

    transitions["PreviousState"] = previous_state.loc[transition_mask].astype("Int64")
    transitions["Transition"] = (
        transitions["PreviousState"].astype(str)
        + " \u2192 "
        + transitions["State"].astype(str)
    )

    return transitions[["Timestamp", "Transition"]]


def _show_state_changes(change=None):
    selected_date = state_date_widget.value

    with state_change_output:
        clear_output(wait=True)

        day_df = state_change_df.loc[
            state_change_df["Timestamp"].dt.date == selected_date
        ].copy()

        engine_1_changes = _engine_state_changes(day_df, engine_1_esn).rename(
            columns={"Transition": engine_1_esn}
        )
        engine_2_changes = _engine_state_changes(day_df, engine_2_esn).rename(
            columns={"Transition": engine_2_esn}
        )

        result = (
            engine_1_changes.merge(
                engine_2_changes,
                on="Timestamp",
                how="outer",
            )
            .sort_values("Timestamp", kind="stable")
            .reset_index(drop=True)
        )

        result = result.rename(
            columns={
                engine_1_esn: f"Engine-1 ID: {engine_1_esn}",
                engine_2_esn: f"Engine-2 ID: {engine_2_esn}",
            }
        )

        if result.empty:
            print(f"No AC trIsUnlocked state changes on {selected_date}.")
            return

        display_result = result.copy()
        display_result["Timestamp"] = display_result["Timestamp"].dt.strftime(
            "%Y-%m-%d %H:%M:%S.%f"
        ).str[:-3]

        ipy_display(display_result)


state_date_widget.observe(_show_state_changes, names="value")

ipy_display(
    widgets.VBox(
        [
            state_date_widget,
            state_change_output,
        ]
    )
)

_show_state_changes()
