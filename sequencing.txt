def _ec_quote(name):
    return F.col(f"`{name.replace('`', '``')}`")


def _ec_parameter_label(base):
    if base in EC_LABELS:
        return EC_LABELS[base]
    label = base
    for ending in ("_data", "_flt"):
        if label.endswith(ending):
            label = label[:-len(ending)]
    return label


def _ec_full_column(channel, base):
    return f"{EC_PARAMETER_PREFIX}{channel}_{base}"


def _ec_scalar(value):
    return value.item() if isinstance(value, np.generic) else value


def _ec_number(value):
    value = _ec_scalar(value)
    if isinstance(value, (bool, np.bool_)):
        return float(bool(value))
    if isinstance(value, (int, float, np.integer, np.floating)):
        numeric = float(value)
        return numeric if np.isfinite(numeric) else None
    try:
        numeric = float(value)
        return numeric if np.isfinite(numeric) else None
    except Exception:
        return None


def _ec_value_key(value):
    value = _ec_scalar(value)
    number = _ec_number(value)
    if number is not None:
        return ("number", round(number, 9))
    return ("text", str(value))


def _ec_format_value(value):
    value = _ec_scalar(value)
    if isinstance(value, (bool, np.bool_)):
        return "1" if bool(value) else "0"
    if isinstance(value, str) and value.strip().lower() in {"true", "false"}:
        return "1" if value.strip().lower() == "true" else "0"
    number = _ec_number(value)
    if number is not None:
        if np.isclose(number, round(number), rtol=0.0, atol=1e-12):
            return str(int(round(number)))
        return f"{number:.9g}"
    return str(value)


def _ec_truthy(value):
    number = _ec_number(value)
    if number is not None:
        return not np.isclose(number, 0.0, rtol=0.0, atol=1e-12)
    return str(value).strip().lower() in {"true", "on", "yes", "open", "1"}


def _ec_event_text(base, resolved):
    label = _ec_parameter_label(base)
    if resolved["disagreement"]:
        return f"{label} AC={_ec_format_value(resolved['ac'])} | BC={_ec_format_value(resolved['bc'])}"
    value = resolved["value"]
    if base in EC_ON_OFF_PARAMETERS:
        return f"{label} {'ON' if _ec_truthy(value) else 'OFF'}"
    if base == "IThrustReverser_trDoorStowStatus_data":
        return f"{label} {'Close' if _ec_truthy(value) else 'Open'}"
    if isinstance(_ec_scalar(value), (bool, np.bool_)):
        return f"{label} {'ON' if bool(value) else 'OFF'}"
    return f"{label}={_ec_format_value(value)}"


def _ec_recognized_state(value, base):
    if base in EC_ANALOG_LVT_PARAMETERS:
        numeric = value.cast("double")
        return (
            F.when(numeric <= F.lit(EC_LVT_LOW_THRESHOLD), F.lit(0.0))
            .when(numeric >= F.lit(EC_LVT_HIGH_THRESHOLD), F.lit(100.0))
            .otherwise(F.lit(None).cast("double"))
        )
    recognized = F.lit(None).cast("double")
    for state in EC_PARAMETER_STATE_VALUES.get(base, EC_GENERIC_STATE_VALUES):
        recognized = F.when(
            F.abs(value.cast("double") - F.lit(float(state))) <= F.lit(EC_STATE_TOLERANCE),
            F.lit(float(state)),
        ).otherwise(recognized)
    return recognized


def _ec_extreme_channel_conflict(ac_value, bc_value, base, is_boolean):
    if base == "IThrustReverser_trDoorDeplStatus_data":
        return F.lit(False)
    ac_number = ac_value.cast("double")
    bc_number = bc_value.cast("double")
    both_present = ac_number.isNotNull() & bc_number.isNotNull()

    def near(value, target):
        return F.abs(value - F.lit(float(target))) <= F.lit(EC_STATE_TOLERANCE)

    if base in EC_BINARY_PARAMETERS or is_boolean:
        opposite_endpoints = (
            (near(ac_number, 0.0) & near(bc_number, 1.0))
            | (near(ac_number, 1.0) & near(bc_number, 0.0))
        )
    else:
        opposite_endpoints = (
            (near(ac_number, 0.0) & near(bc_number, 100.0))
            | (near(ac_number, 100.0) & near(bc_number, 0.0))
        )
    return both_present & opposite_endpoints


def _ec_build_event_df(source, parameters):
    order_window = Window.partitionBy("AssetIdentifier").orderBy("Timestamp")
    state_window = order_window.rowsBetween(Window.unboundedPreceding, Window.currentRow)
    state_expressions = []
    parameter_metadata = []
    for index, base in enumerate(parameters):
        ac_column = _ec_full_column("AC", base)
        bc_column = _ec_full_column("BC", base)
        available = [
            column for column in (ac_column, bc_column) if column in _ec_scan_schema_set
        ]
        if not available:
            continue
        data_type = _ec_scan_type_by_name[available[0]]
        is_boolean = isinstance(data_type, BooleanType)
        is_numeric = isinstance(data_type, (NumericType, BooleanType))
        ac_raw = (
            _ec_quote(ac_column)
            if ac_column in _ec_scan_schema_set
            else F.lit(None).cast(data_type)
        )
        bc_raw = (
            _ec_quote(bc_column)
            if bc_column in _ec_scan_schema_set
            else F.lit(None).cast(data_type)
        )
        if is_numeric:
            ac_raw = F.when(F.isnan(ac_raw.cast("double")), F.lit(None).cast(data_type)).otherwise(ac_raw)
            bc_raw = F.when(F.isnan(bc_raw.cast("double")), F.lit(None).cast(data_type)).otherwise(bc_raw)
        ac_state = f"_ec_ac_{index}"
        bc_state = f"_ec_bc_{index}"
        state_expressions.extend(
            [
                F.last(ac_raw, ignorenulls=True).over(state_window).alias(ac_state),
                F.last(bc_raw, ignorenulls=True).over(state_window).alias(bc_state),
            ]
        )
        parameter_metadata.append(
            {
                "index": index,
                "base": base,
                "ac": ac_state,
                "bc": bc_state,
                "value": f"_ec_value_{index}",
                "disagreement": f"_ec_disagreement_{index}",
                "recognized": f"_ec_recognized_{index}",
                "tracked": f"_ec_tracked_{index}",
                "is_boolean": is_boolean,
                "is_numeric": is_numeric,
            }
        )
    state_df = source.select("Timestamp", "AssetIdentifier", *state_expressions)
    resolved_expressions = []
    for metadata in parameter_metadata:
        ac_state = F.col(metadata["ac"])
        bc_state = F.col(metadata["bc"])
        current = F.coalesce(ac_state, bc_state)
        if metadata["is_numeric"]:
            disagreement = _ec_extreme_channel_conflict(
                ac_state,
                bc_state,
                metadata["base"],
                metadata["is_boolean"],
            )
        else:
            disagreement = F.lit(False)
        recognized = (
            _ec_recognized_state(current, metadata["base"])
            if metadata["is_numeric"]
            else current
        )
        resolved_expressions.extend(
            [
                current.alias(metadata["value"]),
                F.coalesce(disagreement, F.lit(False)).alias(metadata["disagreement"]),
                recognized.alias(metadata["recognized"]),
            ]
        )
    resolved_df = state_df.select("*", *resolved_expressions)
    tracked_expressions = [
        F.last(F.col(metadata["recognized"]), ignorenulls=True)
        .over(state_window)
        .alias(metadata["tracked"])
        for metadata in parameter_metadata
    ]
    tracked_df = resolved_df.select("*", *tracked_expressions)
    previous_expressions = []
    for metadata in parameter_metadata:
        previous_expressions.extend(
            [
                F.lag(metadata["tracked"]).over(order_window).alias(f"_ec_previous_tracked_{metadata['index']}"),
                F.lag(metadata["disagreement"]).over(order_window).alias(f"_ec_previous_disagreement_{metadata['index']}"),
            ]
        )
    comparison_df = tracked_df.select("*", *previous_expressions)
    event_expressions = []
    for metadata in parameter_metadata:
        index = metadata["index"]
        current = F.col(metadata["value"])
        recognized = F.col(metadata["recognized"])
        tracked = F.col(metadata["tracked"])
        previous_tracked = F.col(f"_ec_previous_tracked_{index}")
        disagreement = F.col(metadata["disagreement"])
        previous_disagreement = F.coalesce(
            F.col(f"_ec_previous_disagreement_{index}"),
            F.lit(False),
        )
        recognized_transition = recognized.isNotNull() & (
            previous_tracked.isNull() | (tracked != previous_tracked)
        )
        event_condition = current.isNotNull() & (
            recognized_transition
            | (disagreement & ~previous_disagreement)
        )
        event_expressions.append(
            F.when(
                event_condition,
                F.struct(
                    F.lit(metadata["base"]).alias("parameter"),
                    F.lit(index).alias("parameter_order"),
                    F.col(metadata["ac"]).cast("string").alias("ac_value"),
                    F.col(metadata["bc"]).cast("string").alias("bc_value"),
                    F.when(recognized_transition, recognized.cast("string"))
                    .otherwise(current.cast("string"))
                    .alias("value"),
                    disagreement.alias("disagreement"),
                ),
            )
        )
    return (
        comparison_df.select(
            "Timestamp",
            F.col("AssetIdentifier").cast("string").alias("AssetIdentifier"),
            F.explode(F.array(*event_expressions)).alias("event"),
        )
        .where(F.col("event").isNotNull())
        .select("Timestamp", "AssetIdentifier", "event.*")
    )


def _ec_events_from_spark_frame(frame, engines):
    events_by_engine = {engine: [] for engine in engines}
    if frame.empty:
        return events_by_engine
    frame["Timestamp"] = pd.to_datetime(frame["Timestamp"], errors="coerce")
    frame = frame.dropna(subset=["Timestamp"]).sort_values(
        ["Timestamp", "parameter_order"],
        kind="stable",
    )
    for row in frame.itertuples(index=False):
        engine = str(row.AssetIdentifier)
        if engine not in events_by_engine:
            continue
        if bool(row.disagreement):
            resolved = {
                "value": row.value,
                "ac": row.ac_value,
                "bc": row.bc_value,
                "key": ("ACBC", _ec_value_key(row.ac_value), _ec_value_key(row.bc_value)),
                "disagreement": True,
            }
        else:
            resolved = {
                "value": row.value,
                "key": _ec_value_key(row.value),
                "disagreement": False,
            }
        events_by_engine[engine].append(
            {
                "timestamp": pd.Timestamp(row.Timestamp),
                "engine": engine,
                "parameter": row.parameter,
                "parameter_order": int(row.parameter_order),
                "value_key": resolved["key"],
                "text": _ec_event_text(row.parameter, resolved),
            }
        )
    return events_by_engine


def _ec_timestamp_text(timestamp):
    return pd.Timestamp(timestamp).strftime("%H:%M:%S.%f")[:-3]


def _ec_render_table(events_by_engine, engines, parameters):
    rows = {}
    for engine in engines:
        for event in events_by_engine[engine]:
            key = (event["timestamp"], event["parameter"])
            row = rows.setdefault(
                key,
                {
                    "timestamp": event["timestamp"],
                    "parameter": event["parameter"],
                    "parameter_order": event["parameter_order"],
                    "values": {},
                },
            )
            row["values"][engine] = event["text"]
    ordered_rows = sorted(
        rows.values(),
        key=lambda row: (row["timestamp"], row["parameter_order"]),
    )
    if len(ordered_rows) > EC_MAX_EVENT_ROWS:
        raise RuntimeError(
            f"The selected parameters generated {len(ordered_rows):,} change rows. "
            f"Reduce the parameter selection below {EC_MAX_EVENT_ROWS:,} rows."
        )
    color_map = dict(EC_COLORS)
    extra_index = 0
    for base in parameters:
        if base not in color_map:
            color_map[base] = EC_EXTRA_COLORS[extra_index % len(EC_EXTRA_COLORS)]
            extra_index += 1
    body = []
    for row in ordered_rows:
        base = row["parameter"]
        color = color_map[base]
        value_cells = []
        for engine in engines:
            value = row["values"].get(engine, "")
            style = f"background:{color};" if value else ""
            value_cells.append(
                f'<td class="ec-value" style="{style}">{html.escape(value)}</td>'
            )
        body.append(
            "<tr>"
            f'<td class="ec-time">{_ec_timestamp_text(row["timestamp"])}</td>'
            + "".join(value_cells)
            + "</tr>"
        )
    if not body:
        body.append('<tr><td colspan="3" class="ec-empty">No value changes were found.</td></tr>')
    return f"""
    <style>
      .ec-wrap {{ width:100%; max-height:760px; overflow:auto; border:1px solid #555; background:#fff; }}
      .ec-table {{ width:100%; min-width:720px; border-collapse:collapse; table-layout:fixed; font-size:13px; color:#111; }}
      .ec-table th {{ position:sticky; top:0; z-index:3; background:#efefef; border:1px solid #333; padding:7px 8px; text-align:center; font-weight:700; }}
      .ec-table td {{ border:1px solid #333; padding:5px 7px; vertical-align:middle; }}
      .ec-time {{ width:150px; white-space:nowrap; font-family:ui-monospace,SFMono-Regular,Menlo,Consolas,monospace; }}
      .ec-value {{ width:calc((100% - 150px) / 2); overflow-wrap:anywhere; }}
      .ec-empty {{ text-align:center; color:#666; font-style:italic; padding:18px !important; }}
    </style>
    <div class="ec-wrap">
      <table class="ec-table">
        <thead><tr><th>Time</th><th>{html.escape(engines[0])}</th><th>{html.escape(engines[1])}</th></tr></thead>
        <tbody>{''.join(body)}</tbody>
      </table>
    </div>
    """


def _ec_month_filter(start_time, end_time):
    current = date(start_time.year, start_time.month, 1)
    final = date(end_time.year, end_time.month, 1)
    expression = F.lit(False)
    while current <= final:
        expression = expression | ((F.col("year") == current.year) & (F.col("Month") == current.month))
        current = date(current.year + (current.month == 12), 1 if current.month == 12 else current.month + 1, 1)
    return expression

def _ec_selected_parameters():
    return [value for _, value in ec_parameter_widget.options]


def _ec_refresh_parameter_search():
    selected = set(_ec_selected_parameters())
    ec_parameter_search.options = [
        label for label, base in _ec_search_to_base.items() if base not in selected
    ]


def _ec_add_parameter(change):
    label = change["new"]
    base = _ec_search_to_base.get(label)
    if base is None or base in _ec_selected_parameters():
        return
    ec_parameter_widget.options = tuple(ec_parameter_widget.options) + ((_ec_parameter_label(base), base),)
    ec_parameter_search.value = ""
    _ec_refresh_parameter_search()


def _ec_remove_parameters(_):
    removed = set(ec_parameter_widget.value)
    ec_parameter_widget.options = tuple(
        option for option in ec_parameter_widget.options if option[1] not in removed
    )
    _ec_refresh_parameter_search()


def _ec_refresh_flights(_=None, raise_errors=False):
    ec_find_flights_button.disabled = True
    try:
        with ec_status_output:
            clear_output(wait=True)
            aircraft = ec_aircraft_widget.value
            flight_date = ec_date_widget.value
            if flight_date is None:
                raise ValueError("Choose a flight date.")
            calendar_id = int(flight_date.strftime("%Y%m%d"))
            rows = (
                spark.table(EC_MASTER_TABLE)
                .where(
                    (F.col("AircraftIdentifier").cast("string") == aircraft)
                    & (F.col("CalendarId") == calendar_id)
                )
                .select(
                    "StartDatetime",
                    "EndDatetime",
                    F.col("EngineSerialNumber").cast("string").alias("EngineSerialNumber"),
                )
                .where(F.col("StartDatetime").isNotNull() & F.col("EndDatetime").isNotNull())
                .groupBy("StartDatetime")
                .agg(
                    F.max("EndDatetime").alias("EndDatetime"),
                    F.sort_array(F.collect_set("EngineSerialNumber")).alias("Engines"),
                )
                .where(F.col("EndDatetime") >= F.col("StartDatetime"))
                .orderBy("StartDatetime")
                .collect()
            )
            options = []
            for row in rows:
                engine_count = len([engine for engine in row.Engines if engine is not None])
                label = (
                    f"{row.StartDatetime.strftime('%H:%M:%S.%f')[:-3]} – "
                    f"{row.EndDatetime.strftime('%H:%M:%S.%f')[:-3]} | {engine_count} engines"
                )
                options.append((label, row.StartDatetime))
            ec_flight_widget.options = options
            if options:
                ec_flight_widget.value = options[0][1]
                print(f"Found {len(options)} flights for {aircraft} on {flight_date}.")
            else:
                print(f"No flights were found for {aircraft} on {flight_date}.")
    except Exception as exc:
        ec_flight_widget.options = []
        with ec_status_output:
            print(f"Unable to find flights: {exc}")
        if raise_errors:
            raise
    finally:
        ec_find_flights_button.disabled = False


def _ec_load_comparison(_):
    ec_load_button.disabled = True
    ec_progress_widget.value = "<b>Preparing the selected flight…</b>"
    try:
        with ec_table_output:
            clear_output(wait=True)
        aircraft = ec_aircraft_widget.value
        flight_date = ec_date_widget.value
        start_time = ec_flight_widget.value
        parameters = _ec_selected_parameters()
        if flight_date is None or start_time is None:
            raise ValueError("Choose a date and flight before loading the comparison.")
        if not parameters:
            raise ValueError("Select at least one signal.")
        calendar_id = int(flight_date.strftime("%Y%m%d"))
        master_rows = (
            spark.table(EC_MASTER_TABLE)
            .where(
                (F.col("AircraftIdentifier").cast("string") == aircraft)
                & (F.col("CalendarId") == calendar_id)
                & (F.col("StartDatetime") == F.lit(start_time))
            )
            .select(
                F.col("EngineSerialNumber").cast("string").alias("EngineSerialNumber"),
                "AircraftId",
                "StartDatetime",
                "EndDatetime",
            )
            .where(
                F.col("EngineSerialNumber").isNotNull()
                & F.col("AircraftId").isNotNull()
                & F.col("EndDatetime").isNotNull()
                & (F.col("EndDatetime") >= F.col("StartDatetime"))
            )
            .collect()
        )
        engine_metadata = {}
        for row in master_rows:
            metadata = engine_metadata.setdefault(
                row.EngineSerialNumber,
                {"aircraft_ids": set(), "end_times": []},
            )
            metadata["aircraft_ids"].add(row.AircraftId)
            metadata["end_times"].append(row.EndDatetime)
        engines = sorted(engine_metadata)
        if len(engines) != 2:
            raise ValueError(
                f"Expected exactly two engines for this flight; found {len(engines)}: {engines}"
            )
        engine_windows = {}
        for engine in engines:
            aircraft_ids = engine_metadata[engine]["aircraft_ids"]
            if len(aircraft_ids) != 1:
                raise ValueError(
                    f"Engine {engine} maps to {len(aircraft_ids)} aircraft IDs for this flight."
                )
            engine_windows[engine] = {
                "aircraft_id": next(iter(aircraft_ids)),
                "end_time": max(engine_metadata[engine]["end_times"]),
            }
        end_time = max(window["end_time"] for window in engine_windows.values())
        selected_columns = ["Timestamp", "AssetIdentifier"]
        for base in parameters:
            for channel in EC_CHANNELS:
                column = _ec_full_column(channel, base)
                if column in _ec_scan_schema_set:
                    selected_columns.append(column)
        selected_columns = list(dict.fromkeys(selected_columns))
        populated_signal = F.lit(False)
        for column in selected_columns[2:]:
            populated_signal = populated_signal | _ec_quote(column).isNotNull()
        exact_engine_window = F.lit(False)
        for engine in engines:
            engine_window = engine_windows[engine]
            exact_engine_window = exact_engine_window | (
                (F.col("AircraftId") == F.lit(engine_window["aircraft_id"]))
                & (F.col("AssetIdentifier").cast("string") == F.lit(engine))
                & F.col("Timestamp").between(
                    F.lit(start_time),
                    F.lit(engine_window["end_time"]),
                )
            )
        source = (
            _ec_scan_table.where(
                (F.col("StartDatetime") == F.lit(start_time))
                & _ec_month_filter(start_time, end_time)
                & exact_engine_window
                & populated_signal
            )
            .select(*[_ec_quote(column).alias(column) for column in selected_columns])
        )
        ec_progress_widget.value = "<b>Detecting large transitions in Spark…</b>"
        event_frame = (
            _ec_build_event_df(source, parameters)
            .limit(EC_MAX_EVENT_ROWS + 1)
            .toPandas()
        )
        if len(event_frame) > EC_MAX_EVENT_ROWS:
            raise RuntimeError(
                f"More than {EC_MAX_EVENT_ROWS:,} qualifying events were found. "
                "Select fewer signals."
            )
        if event_frame.empty:
            raise ValueError("No populated signal states were found for the selected flight and engines.")
        ec_progress_widget.value = "<b>Rendering the transition table…</b>"
        events_by_engine = _ec_events_from_spark_frame(event_frame, engines)
        with ec_table_output:
            ipy_display(HTML(_ec_render_table(events_by_engine, engines, parameters)))
        ec_progress_widget.value = ""
    except Exception as exc:
        ec_progress_widget.value = (
            "<span style='color:#b3261e'><b>Comparison failed.</b></span>"
        )
        with ec_table_output:
            clear_output(wait=True)
            print(f"Unable to load comparison: {exc}")
    finally:
        ec_load_button.disabled = False
