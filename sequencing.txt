def _ec_quote(name):
    return F.col(f"`{name.replace('`', '``')}`")


def _ec_parameter_label(base):
    if base in EC_LABELS:
        return EC_LABELS[base]
    label = base
    for ending in ("_data", "_flt"):
        if label.endswith(ending):
            label = label[:-len(ending)]
    return label


def _ec_full_column(base):
    return f"{EC_PARAMETER_PREFIX}{EC_CHANNEL}_{base}"


def _ec_number(value):
    if isinstance(value, np.generic):
        value = value.item()
    if isinstance(value, (bool, np.bool_)):
        return float(bool(value))
    try:
        numeric = float(value)
        return numeric if np.isfinite(numeric) else None
    except Exception:
        return None


def _ec_format_value(value):
    number = _ec_number(value)
    if number is None:
        return str(value)
    if np.isclose(number, round(number), rtol=0.0, atol=1e-12):
        return str(int(round(number)))
    return f"{number:.9g}"


def _ec_recognized_state(value, base):
    if base in EC_ANALOG_LVT_PARAMETERS:
        numeric = value.cast("double")
        return (
            F.when(numeric <= F.lit(EC_LVT_LOW_THRESHOLD), F.lit(0.0))
            .when(numeric >= F.lit(EC_LVT_HIGH_THRESHOLD), F.lit(100.0))
            .otherwise(F.lit(None).cast("double"))
        )
    recognized = F.lit(None).cast("double")
    for state in EC_PARAMETER_STATE_VALUES.get(base, EC_GENERIC_STATE_VALUES):
        recognized = F.when(
            F.abs(value.cast("double") - F.lit(float(state))) <= F.lit(EC_STATE_TOLERANCE),
            F.lit(float(state)),
        ).otherwise(recognized)
    return recognized


def _ec_build_event_df(source, parameters):
    order_window = Window.partitionBy("AssetIdentifier").orderBy("Timestamp")
    state_window = order_window.rowsBetween(Window.unboundedPreceding, Window.currentRow)
    state_expressions = []
    parameter_metadata = []
    for index, base in enumerate(parameters):
        column = _ec_full_column(base)
        if column not in _ec_scan_schema_set:
            continue
        data_type = _ec_scan_type_by_name[column]
        is_boolean = isinstance(data_type, BooleanType)
        is_numeric = isinstance(data_type, (NumericType, BooleanType))
        raw_value = _ec_quote(column)
        if is_numeric:
            raw_value = F.when(
                F.isnan(raw_value.cast("double")),
                F.lit(None).cast(data_type),
            ).otherwise(raw_value)
        state_name = f"_ec_state_{index}"
        state_expressions.append(
            F.last(raw_value, ignorenulls=True).over(state_window).alias(state_name)
        )
        parameter_metadata.append(
            {
                "index": index,
                "base": base,
                "state": state_name,
                "recognized": f"_ec_recognized_{index}",
                "tracked": f"_ec_tracked_{index}",
                "is_boolean": is_boolean,
                "is_numeric": is_numeric,
            }
        )
    state_df = source.select("Timestamp", "AssetIdentifier", *state_expressions)
    recognized_expressions = []
    for metadata in parameter_metadata:
        state = F.col(metadata["state"])
        if metadata["is_boolean"]:
            recognized = (
                F.when(state == F.lit(True), F.lit(1.0))
                .when(state == F.lit(False), F.lit(0.0))
                .otherwise(F.lit(None).cast("double"))
            )
        elif metadata["is_numeric"]:
            recognized = _ec_recognized_state(state, metadata["base"])
        else:
            recognized = state
        recognized_expressions.append(recognized.alias(metadata["recognized"]))
    recognized_df = state_df.select("*", *recognized_expressions)
    tracked_df = recognized_df.select(
        "*",
        *[
            F.last(F.col(metadata["recognized"]), ignorenulls=True)
            .over(state_window)
            .alias(metadata["tracked"])
            for metadata in parameter_metadata
        ],
    )
    comparison_df = tracked_df.select(
        "*",
        *[
            F.lag(metadata["tracked"])
            .over(order_window)
            .alias(f"_ec_previous_{metadata['index']}")
            for metadata in parameter_metadata
        ],
    )
    event_expressions = []
    for metadata in parameter_metadata:
        tracked = F.col(metadata["tracked"])
        previous = F.col(f"_ec_previous_{metadata['index']}")
        changed = tracked.isNotNull() & (previous.isNull() | (tracked != previous))
        event_expressions.append(
            F.when(
                changed,
                F.struct(
                    F.lit(metadata["base"]).alias("parameter"),
                    F.lit(metadata["index"]).alias("parameter_order"),
                    tracked.cast("string").alias("value"),
                ),
            )
        )
    return (
        comparison_df.select(
            "Timestamp",
            F.col("AssetIdentifier").cast("string").alias("AssetIdentifier"),
            F.explode(F.array(*event_expressions)).alias("event"),
        )
        .where(F.col("event").isNotNull())
        .select("Timestamp", "AssetIdentifier", "event.*")
    )


def _ec_timestamp_text(value):
    return pd.Timestamp(value).strftime("%H:%M:%S.%f")[:-3]


def _ec_axis_definition(base, frame):
    if base in EC_Y_AXIS_STATES:
        values, labels = EC_Y_AXIS_STATES[base]
        numeric = pd.to_numeric(frame["value"], errors="coerce")
        span = max(values) - min(values)
        padding = max(span * 0.08, 0.15)
        return numeric.astype(float), values, labels, [min(values) - padding, max(values) + padding]
    numeric = pd.to_numeric(frame["value"], errors="coerce")
    if numeric.notna().all():
        values = sorted(numeric.astype(float).unique().tolist())
        if len(values) == 1:
            padding = max(abs(values[0]) * 0.08, 1.0)
        else:
            padding = max((max(values) - min(values)) * 0.08, 0.1)
        labels = [_ec_format_value(value) for value in values]
        return numeric.astype(float), values, labels, [min(values) - padding, max(values) + padding]
    categories = list(dict.fromkeys(frame["value"].astype(str).tolist()))
    mapping = {value: index for index, value in enumerate(categories)}
    plotted = frame["value"].astype(str).map(mapping).astype(float)
    values = list(range(len(categories)))
    upper = max(values) if values else 0
    return plotted, values, categories, [-0.4, upper + 0.4]


def _ec_build_timeline(event_frame, engines, parameters, start_time, engine_end_times, aircraft):
    frame = event_frame.copy()
    frame["Timestamp"] = pd.to_datetime(frame["Timestamp"], errors="coerce")
    frame = frame.dropna(subset=["Timestamp"]).sort_values(
        ["Timestamp", "parameter_order", "AssetIdentifier"],
        kind="stable",
    )
    active_parameters = [
        base for base in parameters if bool((frame["parameter"] == base).any())
    ]
    if not active_parameters:
        raise ValueError("No AC endpoint or status transitions were found for the selected flight.")
    row_count = len(active_parameters)
    figure = make_subplots(
        rows=row_count,
        cols=1,
        shared_xaxes=True,
        vertical_spacing=min(0.025, 0.16 / max(row_count, 1)),
        subplot_titles=[_ec_parameter_label(base) for base in active_parameters],
    )
    for row_index, base in enumerate(active_parameters, start=1):
        parameter_frame = frame.loc[frame["parameter"] == base].copy()
        plotted, tick_values, tick_labels, y_range = _ec_axis_definition(base, parameter_frame)
        parameter_frame["_plot_value"] = plotted.to_numpy()
        for engine_index, engine in enumerate(engines):
            engine_frame = parameter_frame.loc[
                parameter_frame["AssetIdentifier"].astype(str) == str(engine)
            ].sort_values("Timestamp", kind="stable")
            if engine_frame.empty:
                continue
            style = EC_ENGINE_STYLES[engine_index]
            event_times = engine_frame["Timestamp"].tolist()
            event_values = engine_frame["_plot_value"].astype(float).tolist()
            line_times = list(event_times)
            line_values = list(event_values)
            engine_end = pd.Timestamp(engine_end_times[engine])
            if line_times[-1] < engine_end:
                line_times.append(engine_end)
                line_values.append(line_values[-1])
            figure.add_trace(
                go.Scatter(
                    x=line_times,
                    y=line_values,
                    mode="lines",
                    name=str(engine),
                    legendgroup=str(engine),
                    showlegend=row_index == 1,
                    hoverinfo="skip",
                    line={"color": style["color"], "width": 2.2, "dash": style["dash"], "shape": "hv"},
                ),
                row=row_index,
                col=1,
            )
            custom_data = np.array(
                [
                    [str(engine), _ec_parameter_label(base), _ec_format_value(value)]
                    for value in engine_frame["value"].tolist()
                ],
                dtype=object,
            )
            figure.add_trace(
                go.Scatter(
                    x=event_times,
                    y=event_values,
                    mode="markers",
                    name=str(engine),
                    legendgroup=str(engine),
                    showlegend=False,
                    customdata=custom_data,
                    marker={
                        "color": style["color"],
                        "size": 8,
                        "symbol": style["symbol"],
                        "line": {"color": "#ffffff", "width": 1},
                    },
                    hovertemplate=(
                        "<b>Engine %{customdata[0]}</b><br>"
                        "%{customdata[1]}: %{customdata[2]}<br>"
                        "%{x|%H:%M:%S.%L}<extra></extra>"
                    ),
                ),
                row=row_index,
                col=1,
            )
        figure.update_yaxes(
            tickmode="array",
            tickvals=tick_values,
            ticktext=tick_labels,
            range=y_range,
            fixedrange=False,
            showgrid=True,
            gridcolor="rgba(100,100,100,0.16)",
            zeroline=False,
            row=row_index,
            col=1,
        )
    end_time = max(pd.Timestamp(value) for value in engine_end_times.values())
    tick_stops = [
        {"dtickrange": [None, 1_000], "value": "%H:%M:%S.%L"},
        {"dtickrange": [1_000, 60_000], "value": "%H:%M:%S.%L"},
        {"dtickrange": [60_000, 3_600_000], "value": "%H:%M:%S"},
        {"dtickrange": [3_600_000, None], "value": "%H:%M"},
    ]
    figure.update_xaxes(
        type="date",
        range=[pd.Timestamp(start_time), end_time],
        tickformatstops=tick_stops,
        hoverformat="%H:%M:%S.%L",
        showspikes=True,
        spikemode="across",
        spikesnap="cursor",
        spikethickness=1,
        spikedash="dot",
        spikecolor="#555555",
        showgrid=True,
        gridcolor="rgba(100,100,100,0.12)",
    )
    figure.update_xaxes(
        title_text="Flight timeline — drag to zoom; millisecond ticks appear automatically",
        rangeslider={"visible": True, "thickness": 0.06},
        row=row_count,
        col=1,
    )
    figure.update_layout(
        title={
            "text": (
                f"{aircraft} · AC thrust-reverser transitions · "
                f"{_ec_timestamp_text(start_time)}–{_ec_timestamp_text(end_time)}"
            ),
            "x": 0.01,
            "xanchor": "left",
        },
        template="plotly_white",
        height=max(720, 155 * row_count),
        margin={"l": 92, "r": 35, "t": 105, "b": 95},
        hovermode="closest",
        hoverdistance=40,
        spikedistance=-1,
        legend={
            "orientation": "h",
            "yanchor": "bottom",
            "y": 1.025,
            "xanchor": "right",
            "x": 1.0,
            "groupclick": "togglegroup",
            "title": {"text": "Engines"},
        },
        dragmode="zoom",
        uirevision=f"{aircraft}-{pd.Timestamp(start_time).isoformat()}",
    )
    for annotation, base in zip(figure.layout.annotations, active_parameters):
        annotation.update(
            x=0.0,
            xanchor="left",
            font={"color": EC_PARAMETER_COLORS.get(base, "#333333"), "size": 13},
        )
    return figure


def _ec_month_filter(start_time, end_time):
    current = date(start_time.year, start_time.month, 1)
    final = date(end_time.year, end_time.month, 1)
    expression = F.lit(False)
    while current <= final:
        expression = expression | (
            (F.col("year") == current.year) & (F.col("Month") == current.month)
        )
        current = date(
            current.year + (current.month == 12),
            1 if current.month == 12 else current.month + 1,
            1,
        )
    return expression

def _ec_selected_parameters():
    return [value for _, value in ec_parameter_widget.options]


def _ec_refresh_parameter_search():
    selected = set(_ec_selected_parameters())
    ec_parameter_search.options = [
        label for label, base in _ec_search_to_base.items() if base not in selected
    ]


def _ec_add_parameter(change):
    label = change["new"]
    base = _ec_search_to_base.get(label)
    if base is None or base in _ec_selected_parameters():
        return
    ec_parameter_widget.options = tuple(ec_parameter_widget.options) + (
        (_ec_parameter_label(base), base),
    )
    ec_parameter_search.value = ""
    _ec_refresh_parameter_search()


def _ec_remove_parameters(_):
    removed = set(ec_parameter_widget.value)
    ec_parameter_widget.options = tuple(
        option for option in ec_parameter_widget.options if option[1] not in removed
    )
    _ec_refresh_parameter_search()


def _ec_refresh_flights(_=None, raise_errors=False):
    ec_find_flights_button.disabled = True
    try:
        with ec_status_output:
            clear_output(wait=True)
            aircraft = ec_aircraft_widget.value
            flight_date = ec_date_widget.value
            if flight_date is None:
                raise ValueError("Choose a flight date.")
            calendar_id = int(flight_date.strftime("%Y%m%d"))
            rows = (
                spark.table(EC_MASTER_TABLE)
                .where(
                    (F.col("AircraftIdentifier").cast("string") == aircraft)
                    & (F.col("CalendarId") == calendar_id)
                )
                .select(
                    "StartDatetime",
                    "EndDatetime",
                    F.col("EngineSerialNumber").cast("string").alias("EngineSerialNumber"),
                )
                .where(F.col("StartDatetime").isNotNull() & F.col("EndDatetime").isNotNull())
                .groupBy("StartDatetime")
                .agg(
                    F.max("EndDatetime").alias("EndDatetime"),
                    F.sort_array(F.collect_set("EngineSerialNumber")).alias("Engines"),
                )
                .where(F.col("EndDatetime") >= F.col("StartDatetime"))
                .orderBy("StartDatetime")
                .collect()
            )
            options = []
            for row in rows:
                engines = [engine for engine in row.Engines if engine is not None]
                label = (
                    f"{_ec_timestamp_text(row.StartDatetime)} – "
                    f"{_ec_timestamp_text(row.EndDatetime)} | {len(engines)} engines"
                )
                options.append((label, row.StartDatetime))
            ec_flight_widget.options = options
            if options:
                ec_flight_widget.value = options[0][1]
                print(f"Found {len(options)} flights for {aircraft} on {flight_date}.")
            else:
                print(f"No flights were found for {aircraft} on {flight_date}.")
    except Exception as exc:
        ec_flight_widget.options = []
        with ec_status_output:
            print(f"Unable to find flights: {exc}")
        if raise_errors:
            raise
    finally:
        ec_find_flights_button.disabled = False

        aircraft = ec_aircraft_widget.value
        flight_date = ec_date_widget.value
        start_time = ec_flight_widget.value
        parameters = _ec_selected_parameters()
        if flight_date is None or start_time is None:
            raise ValueError("Choose a date and flight before loading the timeline.")
        if not parameters:
            raise ValueError("Select at least one signal.")
        calendar_id = int(flight_date.strftime("%Y%m%d"))
        master_rows = (
            spark.table(EC_MASTER_TABLE)
            .where(
                (F.col("AircraftIdentifier").cast("string") == aircraft)
                & (F.col("CalendarId") == calendar_id)
                & (F.col("StartDatetime") == F.lit(start_time))
            )
            .select(
                F.col("EngineSerialNumber").cast("string").alias("EngineSerialNumber"),
                "AircraftId",
                "StartDatetime",
                "EndDatetime",
            )
            .where(
                F.col("EngineSerialNumber").isNotNull()
                & F.col("AircraftId").isNotNull()
                & F.col("EndDatetime").isNotNull()
                & (F.col("EndDatetime") >= F.col("StartDatetime"))
            )
            .collect()
        )
        engine_metadata = {}
        for row in master_rows:
            metadata = engine_metadata.setdefault(
                row.EngineSerialNumber,
                {"aircraft_ids": set(), "end_times": []},
            )
            metadata["aircraft_ids"].add(row.AircraftId)
            metadata["end_times"].append(row.EndDatetime)
        engines = sorted(engine_metadata)
        if len(engines) != 2:
            raise ValueError(
                f"Expected exactly two engines for this flight; found {len(engines)}: {engines}"
            )
        engine_windows = {}
        for engine in engines:
            aircraft_ids = engine_metadata[engine]["aircraft_ids"]
            if len(aircraft_ids) != 1:
                raise ValueError(
                    f"Engine {engine} maps to {len(aircraft_ids)} aircraft IDs for this flight."
                )
            engine_windows[engine] = {
                "aircraft_id": next(iter(aircraft_ids)),
                "end_time": max(engine_metadata[engine]["end_times"]),
            }
        end_time = max(window["end_time"] for window in engine_windows.values())
        selected_columns = ["Timestamp", "AssetIdentifier"]
        for base in parameters:
            column = _ec_full_column(base)
            if column in _ec_scan_schema_set:
                selected_columns.append(column)
        selected_columns = list(dict.fromkeys(selected_columns))
        populated_signal = F.lit(False)
        for column in selected_columns[2:]:
            populated_signal = populated_signal | _ec_quote(column).isNotNull()
        exact_engine_window = F.lit(False)
        for engine in engines:
            engine_window = engine_windows[engine]
            exact_engine_window = exact_engine_window | (
                (F.col("AircraftId") == F.lit(engine_window["aircraft_id"]))
                & (F.col("AssetIdentifier").cast("string") == F.lit(engine))
                & F.col("Timestamp").between(
                    F.lit(start_time),
                    F.lit(engine_window["end_time"]),
                )
            )
        source = (
            _ec_scan_table.where(
                (F.col("StartDatetime") == F.lit(start_time))
                & _ec_month_filter(start_time, end_time)
                & exact_engine_window
                & populated_signal
            )
            .select(*[_ec_quote(column).alias(column) for column in selected_columns])
        )
        ec_progress_widget.value = "<b>Detecting AC endpoint and status transitions…</b>"
        event_frame = (
            _ec_build_event_df(source, parameters)
            .orderBy("Timestamp", "AssetIdentifier", "parameter_order")
            .limit(EC_MAX_EVENT_ROWS + 1)
            .toPandas()
        )
        if len(event_frame) > EC_MAX_EVENT_ROWS:
            raise RuntimeError(
                f"More than {EC_MAX_EVENT_ROWS:,} transitions were found. Select fewer signals."
            )
        if event_frame.empty:
            raise ValueError("No AC endpoint or status transitions were found for this flight.")
        ec_progress_widget.value = "<b>Building the interactive millisecond timeline…</b>"
        figure = _ec_build_timeline(
            event_frame,
            engines,
            parameters,
            start_time,
            {engine: engine_windows[engine]["end_time"] for engine in engines},
            aircraft,
        )
        with ec_chart_output:
            figure.show(
                config={
                    "displaylogo": False,
                    "scrollZoom": True,
                    "responsive": True,
                    "modeBarButtonsToRemove": ["lasso2d", "select2d"],
                }
            )
        ec_progress_widget.value = (
            f"<span style='color:#137333'><b>Loaded {len(event_frame):,} AC transitions. "
            "Drag over any interval to zoom to milliseconds; double-click to reset.</b></span>"
        )
    except Exception as exc:
        ec_progress_widget.value = (
            "<span style='color:#b3261e'><b>Timeline failed.</b></span>"
        )
        with ec_chart_output:
            clear_output(wait=True)
            print(f"Unable to load timeline: {exc}")
    finally:
        ec_load_button.disabled = False


ec_parameter_search.observe(_ec_add_parameter, names="value")
ec_remove_parameter_button.on_click(_ec_remove_parameters)
ec_find_flights_button.on_click(_ec_refresh_flights)
ec_load_button.on_click(_ec_load_timeline)

        ec_chart_output,
    ]
)
ipy_display(ec_controls)
