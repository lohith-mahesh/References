import ipywidgets as widgets
from IPython.display import clear_output, display as ipy_display

TR_UNLOCKED_AC_COLUMN = (
    f"{PARAMETER_PREFIX}AC_IThrustReverser_trIsUnlocked_data"
)

if TR_UNLOCKED_AC_COLUMN not in _data_df.columns:
    raise ValueError(
        f"Required AC thrust-reverser unlock parameter is missing: "
        f"{TR_UNLOCKED_AC_COLUMN}"
    )

if _data_df.empty:
    raise ValueError("No samples are available for state-change analysis.")

if "Timestamp" not in _data_df.columns:
    raise ValueError("Timestamp is missing from the Pandas dataframe.")

tr_state_df = _data_df[
    ["Timestamp", TR_UNLOCKED_AC_COLUMN]
].copy()

tr_state_df["Timestamp"] = pd.to_datetime(
    tr_state_df["Timestamp"],
    errors="coerce",
)

tr_state_df["_State"] = pd.to_numeric(
    tr_state_df[TR_UNLOCKED_AC_COLUMN],
    errors="coerce",
)

tr_state_df = tr_state_df[
    tr_state_df["Timestamp"].notna()
    & tr_state_df["_State"].isin([0, 1])
].copy()

tr_state_df = tr_state_df.sort_values(
    "Timestamp",
    kind="stable",
).reset_index(drop=True)

available_tr_dates = sorted(
    tr_state_df["Timestamp"].dt.date.unique().tolist()
)

if not available_tr_dates:
    raise ValueError(
        "No valid 0/1 trIsUnlocked_data samples are available."
    )

tr_date_widget = widgets.Dropdown(
    options=[
        (selected_date.strftime("%Y-%m-%d"), selected_date)
        for selected_date in available_tr_dates
    ],
    value=available_tr_dates[-1],
    description="Date:",
    layout=widgets.Layout(width="300px"),
)

tr_state_output = widgets.Output()


def _format_transition(previous_state, current_state):
    return f"{int(previous_state)} → {int(current_state)}"


def _show_tr_unlock_state_changes(change=None):
    selected_date = tr_date_widget.value

    day_df = tr_state_df[
        tr_state_df["Timestamp"].dt.date == selected_date
    ].copy()

    with tr_state_output:
        clear_output(wait=True)

        if day_df.empty:
            print(f"No valid trIsUnlocked_data samples found for {selected_date}.")
            return

        day_df["_PreviousState"] = day_df["_State"].shift(1)

        transitions = day_df[
            day_df["_PreviousState"].notna()
            & day_df["_State"].ne(day_df["_PreviousState"])
        ].copy()

        if transitions.empty:
            print(
                f"No trIsUnlocked_data state changes found for "
                f"{selected_date}."
            )
            return

        transitions["_Transition"] = transitions.apply(
            lambda row: _format_transition(
                row["_PreviousState"],
                row["_State"],
            ),
            axis=1,
        )

        engine_1_column = f"Engine-1 ID ({selected_esn})"
        engine_2_column = "Engine-2 ID"

        result_df = pd.DataFrame(
            {
                "Timestamp": transitions["Timestamp"],
                engine_1_column: transitions["_Transition"],
                engine_2_column: "",
            }
        )

        result_df["Timestamp"] = result_df["Timestamp"].dt.strftime(
            "%Y-%m-%d %H:%M:%S.%f"
        ).str[:-3]

        ipy_display(result_df.reset_index(drop=True))


tr_date_widget.observe(
    _show_tr_unlock_state_changes,
    names="value",
)

ipy_display(
    widgets.VBox(
        [
            tr_date_widget,
            tr_state_output,
        ]
    )
)

_show_tr_unlock_state_changes()
