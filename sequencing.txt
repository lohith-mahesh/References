from IPython.display import HTML, clear_output, display as ipy_display
from pyspark.sql import functions as F
from pyspark.sql.window import Window


TR_EVENT_SIGNALS = [
    ("LVTPosVCntrl", "IThrustReverser_trLVTPosVCtrl_data", "endpoint"),
    ("Low TRLVT Raw", "IThrustReverser_loOwnTRLVTRaw_data", "endpoint"),
    ("Up TRLVT Raw", "IThrustReverser_upOwnTRLVTRaw_data", "endpoint"),
    ("TR Armed", "IThrustReverser_trArmed_data", "discrete"),
    ("DoorDep Status", "IThrustReverser_trDoorDeplStatus_data", "discrete"),
    ("In Transit", "IThrustReverser_trInTransit_data", "discrete"),
    ("Is Unlocked", "IThrustReverser_trIsUnlocked_data", "discrete"),
    ("Door Stow", "IThrustReverser_trDoorStowStatus_data", "discrete"),
    ("Is Deployed", "IThrustReverser_trIsDeployed_data", "discrete"),
]


_previous_event_widget = globals().get("tr_event_date_widget")
_previous_event_date = getattr(_previous_event_widget, "value", None)

if _previous_event_date is None:
    _previous_event_date = getattr(
        globals().get("end_date_widget"),
        "value",
        None,
    )

if _previous_event_date is None:
    _previous_event_date = date.today()


tr_event_date_widget = widgets.DatePicker(
    value=_previous_event_date,
    description="Event date:",
    layout=widgets.Layout(width="300px"),
)

tr_event_button = widgets.Button(
    description="Show AC transitions",
    button_style="primary",
    layout=widgets.Layout(width="200px"),
)

tr_event_output = widgets.Output()


def _tr_format_value(value):
    if value is None:
        return ""

    value = float(value)

    if value.is_integer():
        return str(int(value))

    return f"{value:g}"


def _tr_render_table(
    events,
    aircraft_identifier,
    engine_serial_number,
    chosen_date,
    joined_row_count=None,
):
    if not events:
        message = (
            "<div style='font-family:Arial,Helvetica,sans-serif;'>"
            "<b>No AC thrust-reverser states/transitions found.</b><br><br>"
            f"Aircraft: <b>{html.escape(str(aircraft_identifier))}</b><br>"
            f"Engine: <b>{html.escape(str(engine_serial_number))}</b><br>"
            f"Channel: <b>AC</b><br>"
            f"Date: <b>{html.escape(str(chosen_date))}</b>"
        )

        if joined_row_count is not None:
            message += (
                f"<br>Matched scan rows: "
                f"<b>{html.escape(str(joined_row_count))}</b>"
            )

        message += "</div>"

        ipy_display(HTML(message))
        return

    signal_styles = {
        "LVTPosVCntrl": ("#eadfe8", "#000000"),
        "Low TRLVT Raw": ("#d4a900", "#000000"),
        "Up TRLVT Raw": ("#2e3192", "#ffffff"),
        "TR Armed": ("#009b35", "#000000"),
        "DoorDep Status": ("#d8c784", "#000000"),
        "In Transit": ("#5fd0d4", "#000000"),
        "Is Unlocked": ("#b7b7b7", "#000000"),
        "Door Stow": ("#f0a000", "#000000"),
        "Is Deployed": ("#269b91", "#000000"),
    }

    rows_html = []

    for event in events:
        background, foreground = signal_styles[event["Signal"]]

        timestamp_text = (
            event["Timestamp"]
            .strftime("%H:%M:%S.%f")[:-3]
        )

        event_text = (
            f"{html.escape(event['Signal'])} = "
            f"{html.escape(_tr_format_value(event['Value']))}"
        )

        rows_html.append(
            "<tr>"
            f"<td class='tr-time'>{timestamp_text}</td>"
            f"<td class='tr-event' "
            f"style='background:{background};color:{foreground};'>"
            f"{event_text}"
            "</td>"
            "</tr>"
        )

    table_html = f"""
    <style>
        .tr-container {{
            display: inline-block;
            font-family: Arial, Helvetica, sans-serif;
            margin-top: 8px;
        }}

        .tr-meta {{
            margin-bottom: 8px;
            font-size: 13px;
            line-height: 1.5;
        }}

        .tr-table {{
            border-collapse: collapse;
            border: 2px solid #222;
            font-size: 14px;
        }}

        .tr-table th {{
            border: 1px solid #222;
            background: #f2f2f2;
            padding: 5px 8px;
            text-align: left;
            font-weight: 600;
        }}

        .tr-table td {{
            border: 1px solid #222;
            padding: 2px 6px;
            line-height: 1.25;
        }}

        .tr-time {{
            white-space: nowrap;
            background: #ffffff;
            font-family:
                ui-monospace,
                SFMono-Regular,
                Menlo,
                Monaco,
                Consolas,
                monospace;
        }}

        .tr-event {{
            min-width: 230px;
            white-space: nowrap;
        }}
    </style>

    <div class="tr-container">

        <div class="tr-meta">
            Aircraft:
            <b>{html.escape(str(aircraft_identifier))}</b>

            &nbsp;|&nbsp;

            Engine:
            <b>{html.escape(str(engine_serial_number))}</b>

            &nbsp;|&nbsp;

            Channel:
            <b>AC</b>

            &nbsp;|&nbsp;

            Date:
            <b>{html.escape(str(chosen_date))}</b>

            &nbsp;|&nbsp;

            Events:
            <b>{len(events)}</b>
        </div>

        <table class="tr-table">
            <thead>
                <tr>
                    <th>Time</th>
                    <th>
                        {html.escape(str(engine_serial_number))} - AC
                    </th>
                </tr>
            </thead>

            <tbody>
                {''.join(rows_html)}
            </tbody>
        </table>

    </div>
    """

    ipy_display(HTML(table_html))


def _run_tr_event_view(_):
    tr_event_button.disabled = True

    try:
        with tr_event_output:
            clear_output(wait=True)

            required_globals = [
                "spark",
                "MASTER_TABLE",
                "SCAN_TABLE",
                "PARAMETER_PREFIX",
                "aircraft_widget",
                "engine_widget",
                "aircraft_ids_by_identifier",
                "engine_ids_by_aircraft",
                "_month_partitions_between_dates",
            ]

            missing_globals = [
                name
                for name in required_globals
                if name not in globals()
            ]

            if missing_globals:
                raise RuntimeError(
                    "Run the notebook setup/selection cells first. "
                    "Missing objects: "
                    + ", ".join(missing_globals)
                )

            chosen_date = tr_event_date_widget.value
            chosen_aircraft = aircraft_widget.value
            chosen_engine = engine_widget.value

            if chosen_date is None:
                raise ValueError("Select an event date.")

            if chosen_aircraft is None:
                raise ValueError("Select an aircraft first.")

            if chosen_engine is None:
                raise ValueError("Select an engine first.")

            if chosen_aircraft not in aircraft_ids_by_identifier:
                raise ValueError(
                    f"Aircraft '{chosen_aircraft}' was not found "
                    "in aircraft_ids_by_identifier."
                )

            valid_engines = engine_ids_by_aircraft.get(
                chosen_aircraft,
                {},
            )

            if chosen_engine not in valid_engines:
                raise ValueError(
                    f"Engine '{chosen_engine}' is not mapped to "
                    f"aircraft '{chosen_aircraft}'."
                )

            internal_aircraft_ids = aircraft_ids_by_identifier[
                chosen_aircraft
            ]

            if len(internal_aircraft_ids) != 1:
                raise ValueError(
                    f"Aircraft '{chosen_aircraft}' maps to "
                    f"{len(internal_aircraft_ids)} internal aircraft IDs. "
                    "Expected exactly one."
                )

            internal_aircraft_id = next(
                iter(internal_aircraft_ids)
            )

            calendar_id = int(
                chosen_date.strftime("%Y%m%d")
            )

            master_day_df = (
                spark.table(MASTER_TABLE)
                .where(
                    (
                        F.col("AircraftIdentifier").cast("string")
                        == str(chosen_aircraft)
                    )
                    &
                    (
                        F.col("EngineSerialNumber").cast("string")
                        == str(chosen_engine)
                    )
                    &
                    (
                        F.col("CalendarId").cast("int")
                        == calendar_id
                    )
                )
            )

            day_windows_df = (
                master_day_df
                .select(
                    F.col("StartDatetime"),
                    F.col("EndDatetime"),
                )
                .where(
                    F.col("StartDatetime").isNotNull()
                    & F.col("EndDatetime").isNotNull()
                    & (
                        F.col("EndDatetime")
                        >= F.col("StartDatetime")
                    )
                )
                .groupBy("StartDatetime")
                .agg(
                    F.max("EndDatetime").alias("EndDatetime")
                )
            )

            day_windows = (
                day_windows_df
                .orderBy("StartDatetime")
                .collect()
            )

            if not day_windows:
                ipy_display(
                    HTML(
                        "<b>No flight windows found.</b><br>"
                        f"Aircraft: {html.escape(str(chosen_aircraft))}<br>"
                        f"Engine: {html.escape(str(chosen_engine))}<br>"
                        f"Date: {html.escape(str(chosen_date))}"
                    )
                )
                return

            scan_table_df = spark.table(SCAN_TABLE)
            scan_columns = set(scan_table_df.columns)

            signal_specs = []
            missing_signal_columns = []

            for (
                display_name,
                suffix,
                signal_type,
            ) in TR_EVENT_SIGNALS:

                full_column_name = (
                    f"{PARAMETER_PREFIX}AC_{suffix}"
                )

                if full_column_name not in scan_columns:
                    missing_signal_columns.append(
                        full_column_name
                    )
                else:
                    signal_specs.append(
                        (
                            display_name,
                            full_column_name,
                            signal_type,
                        )
                    )

            if missing_signal_columns:
                raise ValueError(
                    "The following required AC columns are "
                    "missing from SCAN_TABLE:\n\n"
                    + "\n".join(missing_signal_columns)
                )

            range_start = min(
                row.StartDatetime
                for row in day_windows
            )

            range_end = max(
                row.EndDatetime
                for row in day_windows
            )

            partition_filter = F.lit(False)

            for (
                partition_year,
                partition_month,
            ) in _month_partitions_between_dates(
                range_start.date(),
                range_end.date(),
            ):
                partition_filter = (
                    partition_filter
                    |
                    (
                        (F.col("year") == partition_year)
                        &
                        (F.col("Month") == partition_month)
                    )
                )

            scan_alias = (
                scan_table_df
                .where(
                    (
                        F.col("AircraftId")
                        == internal_aircraft_id
                    )
                    &
                    (
                        F.col("AssetIdentifier")
                        .cast("string")
                        == str(chosen_engine)
                    )
                    &
                    partition_filter
                )
                .alias("scan")
            )

            window_alias = (
                F.broadcast(day_windows_df)
                .alias("flight")
            )

            joined_df = (
                scan_alias
                .join(
                    window_alias,
                    (
                        scan_alias["StartDatetime"]
                        == window_alias["StartDatetime"]
                    )
                    &
                    (
                        scan_alias["Timestamp"]
                        >= window_alias["StartDatetime"]
                    )
                    &
                    (
                        scan_alias["Timestamp"]
                        <= window_alias["EndDatetime"]
                    ),
                    "inner",
                )
                .select(
                    scan_alias["Timestamp"]
                    .alias("Timestamp"),

                    scan_alias["StartDatetime"]
                    .alias("FlightStartDatetime"),

                    *[
                        scan_alias[column_name]
                        .alias(column_name)

                        for (
                            _,
                            column_name,
                            _,
                        ) in signal_specs
                    ],
                )
            )

            joined_row_count = joined_df.count()

            if joined_row_count == 0:
                _tr_render_table(
                    [],
                    chosen_aircraft,
                    chosen_engine,
                    chosen_date,
                    joined_row_count=0,
                )

                print()
                print(
                    "No scan rows matched the selected "
                    "aircraft/engine/flight windows."
                )
                return

            population_expressions = []

            for (
                display_name,
                column_name,
                _,
            ) in signal_specs:
                population_expressions.append(
                    F.count(
                        F.col(f"`{column_name}`")
                    ).alias(display_name)
                )

            population_row = (
                joined_df
                .agg(*population_expressions)
                .collect()[0]
            )

            populated_signal_count = sum(
                1
                for display_name, _, _ in signal_specs
                if population_row[display_name] > 0
            )

            if populated_signal_count == 0:
                _tr_render_table(
                    [],
                    chosen_aircraft,
                    chosen_engine,
                    chosen_date,
                    joined_row_count=joined_row_count,
                )

                print()
                print(
                    "Matched scan rows exist, but every requested "
                    "AC thrust-reverser signal is null."
                )

                print()
                print("Non-null counts:")

                for (
                    display_name,
                    _,
                    _,
                ) in signal_specs:
                    print(
                        f"  {display_name}: "
                        f"{population_row[display_name]}"
                    )

                return

            event_frames = []

            for (
                signal_order,
                (
                    display_name,
                    column_name,
                    signal_type,
                ),
            ) in enumerate(signal_specs):

                raw_value = (
                    F.col(f"`{column_name}`")
                    .cast("double")
                )

                if signal_type == "endpoint":
                    normalized_value = (
                        F.when(
                            raw_value.between(
                                0.0,
                                3.0,
                            ),
                            F.lit(0.0),
                        )
                        .when(
                            raw_value.between(
                                97.0,
                                100.0,
                            ),
                            F.lit(100.0),
                        )
                    )
                else:
                    normalized_value = raw_value

                signal_df = (
                    joined_df
                    .select(
                        "FlightStartDatetime",
                        "Timestamp",
                        normalized_value.alias(
                            "EventValue"
                        ),
                    )
                    .where(
                        F.col("Timestamp").isNotNull()
                        & F.col("EventValue").isNotNull()
                    )
                )

                same_timestamp_window = (
                    Window
                    .partitionBy(
                        "FlightStartDatetime",
                        "Timestamp",
                    )
                    .orderBy(
                        F.col("EventValue")
                    )
                )

                signal_df = (
                    signal_df
                    .withColumn(
                        "_same_ts_rank",
                        F.row_number().over(
                            same_timestamp_window
                        ),
                    )
                    .where(
                        F.col("_same_ts_rank") == 1
                    )
                    .drop("_same_ts_rank")
                )

                chronological_window = (
                    Window
                    .partitionBy(
                        "FlightStartDatetime"
                    )
                    .orderBy(
                        F.col("Timestamp").asc()
                    )
                )

                signal_df = (
                    signal_df
                    .withColumn(
                        "PreviousValue",
                        F.lag(
                            "EventValue",
                            1,
                        ).over(
                            chronological_window
                        ),
                    )
                    .withColumn(
                        "SequenceNumber",
                        F.row_number().over(
                            chronological_window
                        ),
                    )
                )

                transition_df = (
                    signal_df
                    .where(
                        (
                            F.col("SequenceNumber")
                            == 1
                        )
                        |
                        (
                            F.col("EventValue")
                            != F.col("PreviousValue")
                        )
                    )
                    .select(
                        "Timestamp",
                        "FlightStartDatetime",
                        F.lit(
                            display_name
                        ).alias("Signal"),
                        F.col(
                            "EventValue"
                        ).alias("Value"),
                        F.lit(
                            signal_order
                        ).alias("SignalOrder"),
                    )
                )

                event_frames.append(
                    transition_df
                )

            if not event_frames:
                _tr_render_table(
                    [],
                    chosen_aircraft,
                    chosen_engine,
                    chosen_date,
                    joined_row_count=joined_row_count,
                )
                return

            events_df = event_frames[0]

            for event_df in event_frames[1:]:
                events_df = (
                    events_df
                    .unionByName(event_df)
                )

            event_rows = (
                events_df
                .orderBy(
                    F.col("Timestamp").asc(),
                    F.col("SignalOrder").asc(),
                )
                .collect()
            )

            events = [
                {
                    "Timestamp": row.Timestamp,
                    "Signal": row.Signal,
                    "Value": row.Value,
                }
                for row in event_rows
            ]

            _tr_render_table(
                events,
                chosen_aircraft,
                chosen_engine,
                chosen_date,
                joined_row_count=joined_row_count,
            )

    except Exception as exc:
        with tr_event_output:
            clear_output(wait=True)

            ipy_display(
                HTML(
                    "<div style='color:#b00020;"
                    "font-family:Arial,Helvetica,sans-serif;'>"
                    "<b>AC transition viewer error</b>"
                    "</div>"
                )
            )

            raise

    finally:
        tr_event_button.disabled = False


tr_event_button.on_click(
    _run_tr_event_view
)


ipy_display(
    widgets.VBox(
        [
            widgets.HTML(

                <div style="

                ]
            ),

            tr_event_output,
        ]
    )
)
