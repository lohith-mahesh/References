def _numeric_series(series):
    non_null = series.dropna()
    if non_null.empty:
        return pd.Series(np.nan, index=series.index, dtype="float64"), False
    is_boolean = non_null.map(lambda value: isinstance(value, (bool, np.bool_))).all()
    if is_boolean:
        return series.map({True: 1.0, False: 0.0}), True
    numeric = pd.to_numeric(series, errors="coerce")
    values = numeric.dropna().to_numpy(dtype=float)
    is_discrete = bool(
        len(values)
        and len(np.unique(values)) <= 8
        and np.allclose(values, np.round(values))
    )
    return numeric, is_discrete

def _month_partitions_between_dates(start_date, end_date):
    current = date(start_date.year, start_date.month, 1)
    final = date(end_date.year, end_date.month, 1)
    partitions = []
    while current <= final:
        partitions.append((current.year, current.month))
        current = date(current.year + (current.month == 12), 1 if current.month == 12 else current.month + 1, 1)
    return partitions

def _calendar_id_to_date(value):
    if value is None:
        return None
    text = str(value).strip()
    if text.endswith(".0"):
        text = text[:-2]
    if len(text) != 8 or not text.isdigit():
        raise ValueError(f"Invalid CalendarId value: {value!r}")
    return datetime.strptime(text, "%Y%m%d").date()

def _match_timestamp_timezone(value, reference):
    timestamp = pd.Timestamp(value)
    if reference.tzinfo is None and timestamp.tzinfo is not None:
        return timestamp.tz_localize(None)
    if reference.tzinfo is not None and timestamp.tzinfo is None:
        return timestamp.tz_localize(reference.tzinfo)
    if reference.tzinfo is not None and timestamp.tzinfo is not None:
        return timestamp.tz_convert(reference.tzinfo)
    return timestamp

def _format_gap_duration(total_seconds):
    total_seconds = max(0, int(round(total_seconds)))
    days, remainder = divmod(total_seconds, 86400)
    hours, remainder = divmod(remainder, 3600)
    minutes, seconds = divmod(remainder, 60)
    if days:
        return f"{days}d {hours:02d}h"
    if hours:
        return f"{hours}h {minutes:02d}m"
    if minutes:
        return f"{minutes}m {seconds:02d}s"
    return f"{seconds}s"

def _build_compressed_timeline(frame):
    flight_frames = []
    for flight_key, flight_frame in frame.groupby("StartDatetime", sort=False, dropna=False):
        ordered = flight_frame.sort_values("Timestamp", kind="stable").copy()
        actual_start = ordered["Timestamp"].iloc[0]
        actual_end = ordered["Timestamp"].iloc[-1]
        positive_deltas = ordered["Timestamp"].drop_duplicates().sort_values().diff().dt.total_seconds()
        positive_deltas = positive_deltas.loc[positive_deltas > 0]
        cadence_seconds = float(positive_deltas.median()) if not positive_deltas.empty else 1.0
        flight_frames.append(
            {
                "key": flight_key,
                "frame": ordered,
                "actual_start": actual_start,
                "actual_end": actual_end,
                "cadence_seconds": max(cadence_seconds, 0.001),
                "duration_seconds": max(
                    float((actual_end - actual_start).total_seconds()),
                    max(cadence_seconds, 1.0),
                ),
            }
        )
    flight_frames.sort(key=lambda item: item["actual_start"])
    active_seconds = sum(item["duration_seconds"] for item in flight_frames)
    gap_count = max(0, len(flight_frames) - 1)
    visual_gap_seconds = max(active_seconds * min(0.02, 0.15 / gap_count), 1.0) if gap_count else 0.0
    cursor = 0.0
    timeline_parts = []
    flights = []
    gaps = []
    previous = None
    for item in flight_frames:
        if previous is not None:
            gap_start = cursor
            cursor += visual_gap_seconds
            gaps.append(
                {
                    "plot_start": gap_start,
                    "plot_end": cursor,
                    "plot_center": (gap_start + cursor) / 2.0,
                    "actual_seconds": max(0.0, float((item["actual_start"] - previous["actual_end"]).total_seconds())),
                }
            )
        plot_start = cursor
        part = item["frame"].copy()
        part["_plot_x"] = plot_start + (part["Timestamp"] - item["actual_start"]).dt.total_seconds()
        plot_end = plot_start + item["duration_seconds"]
        part["_flight_order"] = len(flights)
        timeline_parts.append(part)
        flights.append({**item, "plot_start": plot_start, "plot_end": plot_end})
        cursor = plot_end
        previous = item
    timeline = pd.concat(timeline_parts, ignore_index=True) if timeline_parts else frame.iloc[0:0].copy()
    return timeline, flights, gaps, cursor

def _timeline_ticks(flights, maximum_ticks=14):
    if not flights:
        return [], []
    if len(flights) > 7:
        selected_indices = np.unique(np.linspace(0, len(flights) - 1, num=min(maximum_ticks, len(flights)), dtype=int))
        tick_specs = [(flights[index], 0.0) for index in selected_indices]
    else:
        ticks_per_flight = max(1, maximum_ticks // len(flights))
        tick_specs = []
        for flight in flights:
            for fraction in np.linspace(0.0, 1.0, num=ticks_per_flight, endpoint=True):
                tick_specs.append((flight, float(fraction)))
    tick_positions = []
    tick_labels = []
    for flight, fraction in tick_specs:
        elapsed_seconds = flight["duration_seconds"] * fraction
        actual_seconds = max(0.0, float((flight["actual_end"] - flight["actual_start"]).total_seconds()))
        actual_time = flight["actual_start"] + pd.to_timedelta(actual_seconds * fraction, unit="s")
        tick_positions.append(flight["plot_start"] + elapsed_seconds)
        tick_labels.append(actual_time.strftime("%d %b %Y\n%H:%M:%S"))
    return tick_positions, tick_labels

def _draw_axis_break(axis, gap, show_label=False):
    transform = blended_transform_factory(axis.transData, axis.transAxes)
    half_width = max((gap["plot_end"] - gap["plot_start"]) * 0.32, 0.25)
    center = gap["plot_center"]
    x_values = np.linspace(center - half_width, center + half_width, 7)
    y_values = np.array([-0.018, 0.018, -0.018, 0.018, -0.018, 0.018, -0.018])
    axis.plot(x_values, y_values, color="#333333", linewidth=1.3, transform=transform, clip_on=False, zorder=8)
    axis.axvline(center, color="#777777", linewidth=0.7, linestyle=":", alpha=0.45, zorder=1)
    if show_label:
        axis.annotate(
            f"Gap\n{_format_gap_duration(gap['actual_seconds'])}",
            xy=(center, 0),
            xycoords=transform,
            xytext=(0, -42),
            textcoords="offset points",
            ha="center",
            va="top",
            fontsize=8,
            color="#444444",
            annotation_clip=False,
        )

    )

def _format_ranked_correlations(records, top_count):
    positive = sorted((record for record in records if record[1] > 0), key=lambda record: (-record[1], record[0]))[:top_count]
    negative = sorted((record for record in records if record[1] < 0), key=lambda record: (record[1], record[0]))[:top_count]
    output_length = max(len(positive), len(negative), 1)
    def _cell(record, value_class):
        if record is None:
            return '<td class="correlation-empty">No qualifying correlation</td>'
        parameter, value, overlap = record
        visible_name = html.escape(_short_parameter_name(parameter))
        signed_value = html.escape(f"{value:+.6f}")
        pair_count = html.escape(f"n={overlap:,}")
        return (
            '<td><div class="correlation-entry">'
            f'<span class="correlation-name">{visible_name}<small>{pair_count}</small></span>'
            f'<span class="correlation-value {value_class}">{signed_value}</span>'
            "</div></td>"
        )
    rows = []
    for index in range(output_length):
        positive_record = positive[index] if index < len(positive) else None
        negative_record = negative[index] if index < len(negative) else None
        rows.append("<tr>" + _cell(positive_record, "positive-value") + _cell(negative_record, "negative-value") + "</tr>")

def _master_filter(df, aircraft, engine, aircraft_id, engine_id):
    predicate = (
        (_normalized_identifier("AircraftIdentifier") == aircraft)
        & (_normalized_identifier("EngineSerialNumber") == engine)
    )
    if "AircraftId" in df.columns:
        predicate = predicate & (F.col("AircraftId") == aircraft_id)
    if "EngineId" in df.columns:
        predicate = predicate & (F.col("EngineId") == engine_id)
    return df.where(predicate)

def _latest_available_date(aircraft, engine):
    aircraft_id = next(iter(aircraft_ids_by_identifier[aircraft]))
    engine_id = engine_ids_by_aircraft[aircraft][engine]
    master = _master_filter(spark.table(MASTER_TABLE), aircraft, engine, aircraft_id, engine_id)
    latest_row = master.agg(F.max(F.col("CalendarId").cast("int")).alias("LatestCalendarId")).first()
    return _calendar_id_to_date(latest_row.LatestCalendarId if latest_row else None)

        new_aircraft = change["new"]

def _render_parameter_panel():

        selected = set(state["selected_parameters"])
        state["channel_pair_df"]["AC selected"] = state["channel_pair_df"]["Signal"].map(lambda suffix: f"AC_{suffix}" in selected)
        state["channel_pair_df"]["BC selected"] = state["channel_pair_df"]["Signal"].map(lambda suffix: f"BC_{suffix}" in selected)

    input_display = widgets.SelectMultiple(options=tuple(state["selected_parameters"]), description="Plot inputs:", layout=widgets.Layout(width="650px", height="280px"))
    remove_button = widgets.Button(description="Remove highlighted", button_style="danger", icon="trash")
    confirm_button = widgets.Button(description="Confirm inputs", button_style="info")
    output = widgets.Output()
    def refresh_search():
        active = set(input_display.options)
        input_search.options = [name for name in combined if name not in active]
    def sync():
        state["selected_parameters"][:] = list(input_display.options)
        update_schema_selection_flags()
        globals()["selected_parameters"] = state["selected_parameters"]
    def add(change):
        value = change["new"]
        if value and value in combined and value not in input_display.options:
            input_display.options = tuple(input_display.options) + (value,)
            sync()
            input_search.value = ""
            refresh_search()

        highlighted = set(input_display.value)
        input_display.options = tuple(name for name in input_display.options if name not in highlighted)
        sync()
        refresh_search()
    def confirm(_):
        sync()
        with output:
            clear_output(wait=True)
            print(f"Confirmed {len(state['selected_parameters'])} inputs.")
    input_search.observe(add, names="value")
    remove_button.on_click(remove)
    confirm_button.on_click(confirm)
    parameter_panel.children = [widgets.VBox([input_search, input_display, widgets.HBox([remove_button, confirm_button]), output])]
    _register_dynamic_widgets(input_search, input_display, remove_button, confirm_button, output)

def _render_plot_panel():

    data_df = state["data_df"]

        plot_panel.children = [widgets.HTML("No samples are available for plotting.")]
        return

        slider_timestamps = list(all_timestamps)

        return pd.Timestamp(timestamp).strftime("%Y-%m-%d %H:%M:%S.%f").rstrip("0").rstrip(".")
    def nearest(timestamp):
        matched = _match_timestamp_timezone(timestamp, all_timestamps[0])
        nearest_index = all_timestamps.get_indexer([matched], method="nearest")[0]
        return pd.Timestamp(all_timestamps[nearest_index])
    slider = widgets.SelectionRangeSlider(options=[(label(timestamp), timestamp) for timestamp in slider_timestamps], index=(0, len(slider_timestamps) - 1), description="Time range:", orientation="horizontal", layout=widgets.Layout(width="1200px"), continuous_update=False)
    text_start = widgets.Text(value=label(all_timestamps[0]), description="Start:", layout=widgets.Layout(width="500px"))
    text_end = widgets.Text(value=label(all_timestamps[-1]), description="End:", layout=widgets.Layout(width="500px"))

    output = widgets.Output()
    def render(start_time, end_time):
        if start_time > end_time:
            start_time, end_time = end_time, start_time
        _plot_parameters(start_time, end_time, output)
    def click(_):
        try:
            start_time = nearest(text_start.value)
            end_time = nearest(text_end.value)
        except Exception:
            with output:
                clear_output(wait=True)
                print("Start and end must be valid date-time values.")

def _correlation_sample(row_limit):
    state = app["state"]
    cache_key = (state["revision"], row_limit)
    if state.get("correlation_cache_key") == cache_key and state.get("correlation_cache") is not None:
        return state["correlation_cache"], state["correlation_sample_count"]
    previous_cache = state.get("correlation_cache")
    if previous_cache is not None:
        previous_cache.unpersist()
    source = state["cda"]
    if state["row_count"] > row_limit:
        key_columns = ["StartDatetime", "Timestamp"]
        hash_expression = F.xxhash64(*[F.coalesce(_quoted_spark_column(name).cast("string"), F.lit("")) for name in key_columns])
        sample_keys = source.select(*key_columns).withColumn("_CorrelationSampleHash", hash_expression).orderBy(F.col("_CorrelationSampleHash"), F.col("StartDatetime"), F.col("Timestamp")).limit(row_limit).drop("_CorrelationSampleHash")
        sample = source.join(F.broadcast(sample_keys), key_columns, "inner")
    else:
        sample = source
    sample = sample.persist(StorageLevel.MEMORY_AND_DISK)
    sample_count = sample.count()
    state["correlation_cache"] = sample
    state["correlation_cache_key"] = cache_key
    state["correlation_sample_count"] = sample_count
    return sample, sample_count

def _render_correlation_panel():
    state = app["state"]
    top_widget = widgets.BoundedIntText(value=50, min=5, max=200, step=5, description="Top results:", layout=widgets.Layout(width="260px"))
    row_widget = widgets.BoundedIntText(value=50000, min=1000, max=500000, step=10000, description="Max rows:", layout=widgets.Layout(width="260px"))
    pairs_widget = widgets.BoundedIntText(value=30, min=3, max=10000, step=10, description="Minimum pairs:", layout=widgets.Layout(width="260px"))
    run_button = widgets.Button(description="Run correlation analysis", button_style="primary", layout=widgets.Layout(width="240px"))
    output = widgets.Output()
    def run(_):
        run_button.disabled = True
        try:
            with output:
                clear_output(wait=True)
                if not _selection_is_current():
                    print("The selectors have changed. Load the selection before running correlations.")
                    return
                input_columns = []
                for short_name in state["selected_parameters"]:
                    full_name = _full_parameter_name(short_name)
                    if full_name in state["correlation_candidate_set"] and full_name not in input_columns:
                        input_columns.append(full_name)
                if not input_columns:
                    print("None of the confirmed inputs are eligible numeric or Boolean parameters.")
                    return
                if len(input_columns) > CORRELATION_MAX_INPUTS:
                    print(f"Correlation is limited to {CORRELATION_MAX_INPUTS} inputs per run.")
                    return
                if state["row_count"] == 0:
                    print("No flight samples are available for correlation analysis.")
                    return
                row_limit = int(row_widget.value)
                minimum_pairs = int(pairs_widget.value)
                top_count = int(top_widget.value)
                sample, sample_count = _correlation_sample(row_limit)
                correlations = {column: [] for column in input_columns}
                print(f"Inputs: {len(input_columns)} | Candidates: {len(state['correlation_candidate_columns']):,} | Shared deterministic sample: {sample_count:,} rows")
                for batch_start in range(0, len(state["correlation_candidate_columns"]), CORRELATION_BATCH_SIZE):
                    batch_columns = state["correlation_candidate_columns"][batch_start:batch_start + CORRELATION_BATCH_SIZE]
                    required_columns = list(dict.fromkeys(input_columns + batch_columns))
                    batch_df = sample.select(*[_quoted_spark_column(column).alias(column) for column in required_columns])
                    expressions = []
                    result_keys = []
                    for input_index, input_column in enumerate(input_columns):
                        left_raw = _quoted_spark_column(input_column).cast("double")
                        left = F.when(left_raw.isNull() | F.isnan(left_raw), None).otherwise(left_raw)
                        for candidate_index, candidate_column in enumerate(batch_columns):
                            if candidate_column == input_column:
                                continue
                            right_raw = _quoted_spark_column(candidate_column).cast("double")
                            right = F.when(right_raw.isNull() | F.isnan(right_raw), None).otherwise(right_raw)
                            correlation_alias = f"correlation_{input_index}_{candidate_index}"
                            overlap_alias = f"overlap_{input_index}_{candidate_index}"
                            expressions.extend([F.corr(left, right).alias(correlation_alias), F.count(F.when(left.isNotNull() & right.isNotNull(), 1)).alias(overlap_alias)])
                            result_keys.append((input_column, candidate_column, correlation_alias, overlap_alias))
                    if expressions:
                        result = batch_df.agg(*expressions).first().asDict()
                        for input_column, candidate_column, correlation_alias, overlap_alias in result_keys:
                            value = result.get(correlation_alias)
                            overlap = int(result.get(overlap_alias) or 0)
                            if value is not None and math.isfinite(float(value)) and overlap >= minimum_pairs:
                                correlations[input_column].append((candidate_column, float(value), overlap))
                    print(f"Processed {min(batch_start + CORRELATION_BATCH_SIZE, len(state['correlation_candidate_columns'])):,} of {len(state['correlation_candidate_columns']):,} candidates.")
                for input_column in input_columns:
                    ipy_display(HTML("<h3 style='margin:18px 0 4px'>Input feature: " + html.escape(_short_parameter_name(input_column)) + "</h3>"))
                    ipy_display(HTML(_format_ranked_correlations(correlations[input_column], top_count)))
        except Exception as exc:
            with output:
                print(f"Correlation analysis failed: {exc}")
        finally:
            run_button.disabled = not _selection_is_current()
    run_button.on_click(run)
    correlation_panel.children = [widgets.VBox([widgets.HBox([top_widget, row_widget, pairs_widget, run_button]), output])]
    state["action_buttons"].append(run_button)
    _register_dynamic_widgets(top_widget, row_widget, pairs_widget, run_button, output)

def _load_selected_flights(_, raise_errors=False):
    load_flights_btn.disabled = True
    for selector in (aircraft_widget, engine_widget, start_date_widget, end_date_widget):
        selector.disabled = True
    temporary_caches = []
    try:
        with selection_output:
            clear_output(wait=True)
            chosen_aircraft = aircraft_widget.value
            chosen_engine = engine_widget.value
            chosen_start_date = start_date_widget.value
            chosen_end_date = end_date_widget.value
            requested_selection_key = (chosen_aircraft, chosen_engine, chosen_start_date, chosen_end_date)
            if chosen_start_date is None or chosen_end_date is None:
                raise ValueError("Both start and end dates are required.")
            if chosen_start_date > chosen_end_date:
                raise ValueError("Start date must not be later than end date.")
            chosen_aircraft_id = next(iter(aircraft_ids_by_identifier[chosen_aircraft]))
            chosen_engine_id = engine_ids_by_aircraft[chosen_aircraft][chosen_engine]
            start_calendar_id = int(chosen_start_date.strftime("%Y%m%d"))
            end_calendar_id = int(chosen_end_date.strftime("%Y%m%d"))
            master_source = spark.table(MASTER_TABLE)
            required_master_columns = {"AircraftIdentifier", "EngineSerialNumber", "CalendarId", "StartDatetime", "EndDatetime"}
            missing_master_columns = sorted(required_master_columns - set(master_source.columns))
            if missing_master_columns:
                raise ValueError(f"Master table is missing required columns: {missing_master_columns}")
            master_df = _master_filter(master_source, chosen_aircraft, chosen_engine, chosen_aircraft_id, chosen_engine_id).where(F.col("CalendarId").cast("int").between(start_calendar_id, end_calendar_id))
            master_count = master_df.count()
            invalid_window_count = master_df.where(F.col("StartDatetime").isNull() | F.col("EndDatetime").isNull() | (F.col("EndDatetime") < F.col("StartDatetime"))).count()
            calendar_mismatch_count = master_df.where(F.col("StartDatetime").isNotNull() & (F.col("CalendarId").cast("int") != F.date_format(F.col("StartDatetime"), "yyyyMMdd").cast("int"))).count()
            valid_windows = master_df.select("StartDatetime", "EndDatetime").where(F.col("StartDatetime").isNotNull() & F.col("EndDatetime").isNotNull() & (F.col("EndDatetime") >= F.col("StartDatetime"))).dropDuplicates()
            conflicts = valid_windows.groupBy("StartDatetime").agg(F.countDistinct("EndDatetime").alias("EndCount")).where(F.col("EndCount") > 1).limit(20).collect()
            if conflicts:
                conflict_times = ", ".join(str(row.StartDatetime) for row in conflicts)
                raise ValueError(f"Master data contains conflicting end times for the same flight start: {conflict_times}")
            window_count = valid_windows.count()
            duplicate_window_record_count = max(0, master_count - invalid_window_count - window_count)
            if window_count > MAX_FLIGHT_WINDOWS:
                raise RuntimeError(f"The selection contains {window_count:,} flight windows. Reduce the date range below {MAX_FLIGHT_WINDOWS:,} windows.")
            flight_rows = valid_windows.orderBy(F.col("StartDatetime"), F.col("EndDatetime")).collect()
            for previous, current in zip(flight_rows, flight_rows[1:]):
                if current.StartDatetime <= previous.EndDatetime:
                    raise ValueError(f"Overlapping flight windows detected: {previous.StartDatetime}–{previous.EndDatetime} and {current.StartDatetime}–{current.EndDatetime}")
            scan_table = spark.table(SCAN_TABLE)
            required_scan_columns = {"AircraftId", "AssetIdentifier", "Timestamp"}
            missing_scan_columns = sorted(required_scan_columns - set(scan_table.columns))
            if missing_scan_columns:
                raise ValueError(f"Scan table is missing required columns: {missing_scan_columns}")
            reserved_columns = {"_ScanStartDatetime", "_FlightEndDatetime"} & set(scan_table.columns)
            if reserved_columns:
                raise ValueError(f"Scan table uses reserved notebook columns: {sorted(reserved_columns)}")
            warnings = []
            if invalid_window_count:
                warnings.append(f"{invalid_window_count} master records have missing or invalid flight boundaries and were excluded.")
            if duplicate_window_record_count:
                warnings.append(f"{duplicate_window_record_count} duplicate master window records were collapsed without altering their shared boundaries.")
            if calendar_mismatch_count:
                warnings.append(f"{calendar_mismatch_count} master records have CalendarId values that differ from their StartDatetime calendar date.")
            unmatched_flights = []
            if flight_rows:
                range_start = flight_rows[0].StartDatetime
                range_end = max(row.EndDatetime for row in flight_rows)
                partition_filter = F.lit(False)
                if {"year", "Month"}.issubset(scan_table.columns):
                    for partition_year, partition_month in _month_partitions_between_dates(range_start.date(), range_end.date()):
                        partition_filter = partition_filter | ((F.col("year") == partition_year) & (F.col("Month") == partition_month))
                else:
                    partition_filter = F.lit(True)
                    warnings.append("Scan partition columns year/Month are unavailable; partition pruning was skipped.")
                scan_filter = (
                    (F.col("AircraftId") == chosen_aircraft_id)
                    & (_normalized_identifier("AssetIdentifier") == chosen_engine)
                    & F.col("Timestamp").between(range_start, range_end)
                    & partition_filter
                )
                if "EngineId" in scan_table.columns:
                    scan_filter = scan_filter & (F.col("EngineId") == chosen_engine_id)
                scan_alias = scan_table.where(scan_filter).alias("scan")
                window_alias = F.broadcast(valid_windows).alias("window")
                join_condition = scan_alias["Timestamp"].between(window_alias["StartDatetime"], window_alias["EndDatetime"])
                scan_projection = [scan_alias[column] for column in scan_table.columns if column != "StartDatetime"]
                if "StartDatetime" in scan_table.columns:
                    scan_projection.append(scan_alias["StartDatetime"].alias("_ScanStartDatetime"))
                cda = scan_alias.join(window_alias, join_condition, "inner").select(*scan_projection, window_alias["StartDatetime"].alias("StartDatetime"), window_alias["EndDatetime"].alias("_FlightEndDatetime"))
                audit_columns = ["StartDatetime", "_FlightEndDatetime", "Timestamp"]
                if "_ScanStartDatetime" in cda.columns:
                    audit_columns.append("_ScanStartDatetime")
                audit_df = cda.select(*audit_columns).persist(StorageLevel.MEMORY_AND_DISK)
                temporary_caches.append(audit_df)
                matched_count = audit_df.count()
                if matched_count == 0:
                    raise RuntimeError("Master flights were found, but no scan samples fell inside their time windows.")
                duplicate_rows = audit_df.groupBy("StartDatetime", "Timestamp").count().where(F.col("count") > 1).orderBy(F.col("StartDatetime"), F.col("Timestamp")).limit(20).collect()
                if duplicate_rows:
                    examples = ", ".join(f"{row.StartDatetime}/{row.Timestamp} ({row['count']} rows)" for row in duplicate_rows)
                    raise ValueError(f"Duplicate scan samples were detected for the same flight and timestamp: {examples}")
                counts = {(row.StartDatetime, row._FlightEndDatetime): int(row.SampleCount) for row in audit_df.groupBy("StartDatetime", "_FlightEndDatetime").agg(F.count(F.lit(1)).alias("SampleCount")).collect()}
                unmatched_flights = [row for row in flight_rows if counts.get((row.StartDatetime, row.EndDatetime), 0) == 0]
                if unmatched_flights:
                    warnings.append(f"{len(unmatched_flights)} master flight windows contain no scan samples; they remain visible in diagnostics and are excluded from analysis.")
                if "_ScanStartDatetime" in cda.columns:
                    start_mismatch_count = audit_df.where(F.col("_ScanStartDatetime").isNotNull() & (F.col("_ScanStartDatetime") != F.col("StartDatetime"))).count()
                    if start_mismatch_count:
                        warnings.append(f"{start_mismatch_count:,} scan samples had a non-matching scan StartDatetime and were matched safely by aircraft, engine, and timestamp containment.")
            else:
                range_start = None
                range_end = None
                cda = scan_table.limit(0)
                warnings.append("No matching master flight windows were found.")
            cda_cols = cda.columns
            scan_schema = set(cda_cols)
            phase_candidates = [field for field in cda.schema.fields if "flightphase" in field.name.lower() and isinstance(field.dataType, NumericType)]
            flight_phase_columns = {}
            for channel in CHANNELS:
                channel_prefix = f"{PARAMETER_PREFIX}{channel}_".lower()
                matches = [field.name for field in phase_candidates if field.name.lower().startswith(channel_prefix)]
                if len(matches) != 1:
                    raise ValueError(f"Expected exactly one numeric {channel} FlightPhase parameter; found {len(matches)} candidates: {matches}")
                flight_phase_columns[channel] = matches[0]
            base_identifiers = ["Timestamp", "StartDatetime", "ParentAssetIdentifier", "AssetIdentifier"]
            identifier_columns = [name for name in base_identifiers if name in scan_schema] + list(flight_phase_columns.values())
            missing_identifiers = [name for name in ("Timestamp", "StartDatetime", "AssetIdentifier") if name not in identifier_columns]
            if missing_identifiers:
                raise ValueError(f"Filtered scan data is missing required columns: {missing_identifiers}")
            requested_parameters = THRUST_REVERSER_COLUMNS + CONTEXT_COLUMNS
            unavailable_parameters = [name for name in requested_parameters if name not in scan_schema]
            parameter_columns = [name for name in requested_parameters if name in scan_schema]
            selected_columns = list(dict.fromkeys(identifier_columns + parameter_columns))
            selected_spark_df = cda.select(*[_quoted_spark_column(name).alias(name) for name in selected_columns]).orderBy(F.col("Timestamp")).persist(StorageLevel.MEMORY_AND_DISK)
            temporary_caches.append(selected_spark_df)
            row_count = selected_spark_df.count()
            if row_count > MAX_PANDAS_ROWS:
                raise RuntimeError(f"The selected range contains {row_count:,} samples. Reduce the date range below {MAX_PANDAS_ROWS:,} samples.")
            data_df = selected_spark_df.toPandas()
            if not data_df.empty:
                data_df["Timestamp"] = pd.to_datetime(data_df["Timestamp"], errors="coerce")
                data_df["StartDatetime"] = pd.to_datetime(data_df["StartDatetime"], errors="coerce")
                data_df = data_df.dropna(subset=["Timestamp", "StartDatetime"])
                for phase_column in flight_phase_columns.values():
                    data_df[phase_column] = pd.to_numeric(data_df[phase_column], errors="coerce")
                data_df = data_df.sort_values(["Timestamp", "StartDatetime"], kind="stable").reset_index(drop=True)
            correlation_candidates = [field.name for field in cda.schema.fields if _eligible_correlation_parameter(field)]
            channel_pairs = []
            for suffix in THRUST_REVERSER_SUFFIXES:
                ac_short = f"AC_{suffix}"
                bc_short = f"BC_{suffix}"
                channel_pairs.append({"Signal": suffix, "AC available": PARAMETER_PREFIX + ac_short in scan_schema, "BC available": PARAMETER_PREFIX + bc_short in scan_schema})
            channel_pair_df = pd.DataFrame(channel_pairs)
            coverage_records = []
            for suffix in THRUST_REVERSER_SUFFIXES:
                ac_column = f"{PARAMETER_PREFIX}AC_{suffix}"
                bc_column = f"{PARAMETER_PREFIX}BC_{suffix}"
                if ac_column not in data_df.columns or bc_column not in data_df.columns:
                    continue
                ac_values = data_df[ac_column]
                bc_values = data_df[bc_column]
                both = ac_values.notna() & bc_values.notna()
                disagreements = ac_values.loc[both].ne(bc_values.loc[both]).fillna(False)
                coverage_records.append({"Signal": suffix, "AC non-null": int(ac_values.notna().sum()), "BC non-null": int(bc_values.notna().sum()), "Both non-null": int(both.sum()), "Exact disagreements": int(disagreements.sum())})
            channel_coverage_df = pd.DataFrame(coverage_records, columns=["Signal", "AC non-null", "BC non-null", "Both non-null", "Exact disagreements"])
            flight_window_df = pd.DataFrame([{"StartDatetime": row.StartDatetime, "EndDatetime": row.EndDatetime, "Duration": row.EndDatetime - row.StartDatetime, "Matched samples": int(counts.get((row.StartDatetime, row.EndDatetime), 0)) if flight_rows else 0} for row in flight_rows])
            preview_columns = list(dict.fromkeys(identifier_columns + parameter_columns))
            preview_df = data_df.loc[:, [name for name in preview_columns if name in data_df.columns]].head(50).copy()
            old_state = app["state"]
            old_cache = old_state.get("correlation_cache")
            if old_cache is not None:
                old_cache.unpersist()
            revision = old_state.get("revision", 0) + 1
            preserved_parameters = list(old_state.get("selected_parameters", []))
            total_duration = sum((row.EndDatetime - row.StartDatetime for row in flight_rows), timedelta(0))
            app["state"] = {
                "revision": revision,
                "selection_key": requested_selection_key,
                "selected_acid": chosen_aircraft,
                "selected_esn": chosen_engine,
                "internal_acid": chosen_aircraft_id,
                "engineid": chosen_engine_id,
                "window_start_date": chosen_start_date,
                "window_end_date": chosen_end_date,
                "start_calendar_id": start_calendar_id,
                "end_calendar_id": end_calendar_id,
                "master_df": master_df,
                "master_count": master_count,
                "flight_windows_df": valid_windows,
                "flight_rows": flight_rows,
                "unmatched_flights": unmatched_flights,
                "start_dt": range_start,
                "end_dt": range_end,
                "flight_duration": total_duration,
                "scan_table": scan_table,
                "cda_cols": cda_cols,
                "cda": cda,
                "flight_phase_columns": flight_phase_columns,
                "identifier_columns": identifier_columns,
                "parameter_columns": parameter_columns,
                "unavailable_parameters": unavailable_parameters,
                "data_df": data_df,
                "row_count": row_count,
                "correlation_candidate_columns": correlation_candidates,
                "correlation_candidate_set": set(correlation_candidates),
                "selected_parameters": preserved_parameters,
                "correlation_cache": None,
                "correlation_cache_key": None,
                "action_buttons": [],
                "warnings": warnings,
                "channel_pair_df": channel_pair_df,
                "channel_coverage_df": channel_coverage_df,
                "flight_window_df": flight_window_df,
                "preview_df": preview_df,
            }
            globals().update(
                {
                    "selected_acid": chosen_aircraft,
                    "selected_esn": chosen_engine,
                    "internal_acid": chosen_aircraft_id,
                    "engineid": chosen_engine_id,
                    "window_start_date": chosen_start_date,
                    "window_end_date": chosen_end_date,
                    "start_calendar_id": start_calendar_id,
                    "end_calendar_id": end_calendar_id,
                    "cda_master_dt": master_df,
                    "flight_windows_df": valid_windows,
                    "flight_rows": flight_rows,
                    "flight_available": bool(flight_rows),
                    "selected_start_dt": range_start,
                    "selected_end_dt": range_end,
                    "start_dt": range_start,
                    "end_dt": range_end,
                    "flight_duration": total_duration,
                    "scan_table": scan_table,
                    "cda_cols": cda_cols,
                    "cda": cda,
                    "FLIGHT_PHASE_COLUMNS": flight_phase_columns,
                    "IDENTIFIER_COLUMNS": identifier_columns,
                    "PARAMETER_COLUMNS": parameter_columns,
                    "select_para_df": selected_spark_df,
                    "_data_df": data_df,
                    "correlation_candidate_columns": correlation_candidates,
                    "channel_pair_df": channel_pair_df,
                    "channel_coverage_df": channel_coverage_df,
                    "diagnostic_preview_df": preview_df,
                    "selection_revision": revision,
                }
            )
            _clear_dynamic_widgets()
            for render_function, panel, label in (
                (_render_parameter_panel, parameter_panel, "input controls"),
                (_render_plot_panel, plot_panel, "plot controls"),
                (_render_correlation_panel, correlation_panel, "correlation controls"),
                (_render_diagnostics, diagnostic_panel, "diagnostics"),
            ):
                try:
                    render_function()
                except Exception as render_error:
                    panel.children = [widgets.HTML(f"Unable to render {html.escape(label)}: {html.escape(str(render_error))}")]
            print(f"Aircraft: {chosen_aircraft} | Aircraft ID: {chosen_aircraft_id} | ESN: {chosen_engine} | Engine ID: {chosen_engine_id}")
            print(f"Date range: {chosen_start_date} through {chosen_end_date}")
            print(f"Flights: {len(flight_rows):,} | Samples: {row_count:,} | Combined master duration: {total_duration}")
            if unavailable_parameters:
                print(f"Unavailable optional parameters: {len(unavailable_parameters)}")
            for warning in warnings:
                print(f"Warning: {warning}")
    except Exception as exc:
        with selection_output:
            print(f"Unable to load the selected flights: {exc}")
        if raise_errors:
            raise
    finally:
        for cached_df in temporary_caches:
            try:
                cached_df.unpersist()
            except Exception:
                pass
        for selector in (aircraft_widget, engine_widget, start_date_widget, end_date_widget):
            selector.disabled = False
        load_flights_btn.disabled = False
        if app["state"].get("selection_key") is not None and not _selection_is_current():
            _mark_selection_dirty()

aircraft_widget.observe(_on_aircraft_change, names="value")
engine_widget.observe(_on_engine_change, names="value")
start_date_widget.observe(_mark_selection_dirty, names="value")
end_date_widget.observe(_mark_selection_dirty, names="value")
load_flights_btn.on_click(_load_selected_flights)
ipy_display(root_widget)
_load_selected_flights(None, raise_errors=True)
