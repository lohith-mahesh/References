def _ec_quote(name):
    return F.col(f"`{name.replace('`', '``')}`")


def _ec_parameter_label(base):
    if base in EC_LABELS:
        return EC_LABELS[base]
    label = base
    for ending in ("_data", "_flt"):
        if label.endswith(ending):
            label = label[:-len(ending)]
    return label


def _ec_full_column(channel, base):
    return f"{EC_PARAMETER_PREFIX}{channel}_{base}"


def _ec_is_missing(value):
    if value is None:
        return True
    try:
        return bool(pd.isna(value))
    except Exception:
        return False


def _ec_scalar(value):
    return value.item() if isinstance(value, np.generic) else value


def _ec_number(value):
    value = _ec_scalar(value)
    if isinstance(value, (bool, np.bool_)):
        return float(bool(value))
    if isinstance(value, (int, float, np.integer, np.floating)):
        numeric = float(value)
        return numeric if np.isfinite(numeric) else None
    try:
        numeric = float(value)
        return numeric if np.isfinite(numeric) else None
    except Exception:
        return None


def _ec_values_equal(left, right):
    left_number = _ec_number(left)
    right_number = _ec_number(right)
    if left_number is not None and right_number is not None:
        return bool(np.isclose(left_number, right_number, rtol=0.0, atol=1e-12))
    return _ec_scalar(left) == _ec_scalar(right)


def _ec_value_key(value):
    value = _ec_scalar(value)
    number = _ec_number(value)
    if number is not None:
        return ("number", round(number, 9))
    return ("text", str(value))


def _ec_format_value(value):
    value = _ec_scalar(value)
    if isinstance(value, (bool, np.bool_)):
        return "1" if bool(value) else "0"
    number = _ec_number(value)
    if number is not None:
        if np.isclose(number, round(number), rtol=0.0, atol=1e-12):
            return str(int(round(number)))
        return f"{number:.9g}"
    return str(value)


def _ec_parameter_is_discrete(frame, base, available_columns):
    observed = []
    for channel in EC_CHANNELS:
        column = _ec_full_column(channel, base)
        if column in available_columns:
            observed.extend(frame[column].dropna().tolist())
    if not observed:
        return False
    if all(isinstance(_ec_scalar(value), (bool, np.bool_)) for value in observed):
        return True
    numeric = [_ec_number(value) for value in observed]
    if any(value is None for value in numeric):
        return len({str(_ec_scalar(value)) for value in observed}) <= 16
    values = np.asarray(numeric, dtype=float)
    return len(np.unique(values)) <= 16 and np.allclose(values, np.round(values))


def _ec_resolve_channels(ac_value, bc_value, is_discrete):
    ac_missing = _ec_is_missing(ac_value)
    bc_missing = _ec_is_missing(bc_value)
    if ac_missing and bc_missing:
        return None
    if ac_missing:
        value = _ec_scalar(bc_value)
        return {"value": value, "key": _ec_value_key(value), "disagreement": False}
    if bc_missing:
        value = _ec_scalar(ac_value)
        return {"value": value, "key": _ec_value_key(value), "disagreement": False}
    ac_value = _ec_scalar(ac_value)
    bc_value = _ec_scalar(bc_value)
    if _ec_values_equal(ac_value, bc_value):
        return {"value": ac_value, "key": _ec_value_key(ac_value), "disagreement": False}
    ac_number = _ec_number(ac_value)
    bc_number = _ec_number(bc_value)
    if is_discrete or ac_number is None or bc_number is None:
        disagreement = True
    else:
        scale = max(abs(ac_number), abs(bc_number), 1e-12)
        disagreement = abs(ac_number - bc_number) / scale >= EC_DIVERGENCE_THRESHOLD
    if disagreement:
        return {
            "value": ac_value,
            "ac": ac_value,
            "bc": bc_value,
            "key": ("ACBC", _ec_value_key(ac_value), _ec_value_key(bc_value)),
            "disagreement": True,
        }
    return {"value": ac_value, "key": _ec_value_key(ac_value), "disagreement": False}


def _ec_truthy(value):
    number = _ec_number(value)
    if number is not None:
        return not np.isclose(number, 0.0, rtol=0.0, atol=1e-12)
    return str(value).strip().lower() in {"true", "on", "yes", "open", "1"}


def _ec_event_text(base, resolved):
    label = _ec_parameter_label(base)
    if resolved["disagreement"]:
        return f"{label} AC={_ec_format_value(resolved['ac'])} | BC={_ec_format_value(resolved['bc'])}"
    value = resolved["value"]
    if base in EC_ON_OFF_PARAMETERS:
        return f"{label} {'ON' if _ec_truthy(value) else 'OFF'}"
    if base == "IThrustReverser_trDoorStowStatus_data":
        return f"{label} {'Close' if _ec_truthy(value) else 'Open'}"
    if isinstance(_ec_scalar(value), (bool, np.bool_)):
        return f"{label} {'ON' if bool(value) else 'OFF'}"
    return f"{label}={_ec_format_value(value)}"


def _ec_extract_events(frame, engine, parameters, available_columns, discrete_by_parameter):
    engine_frame = frame.loc[
        frame["AssetIdentifier"].astype("string") == str(engine)
    ].copy()
    engine_frame = engine_frame.dropna(subset=["Timestamp"]).sort_values("Timestamp", kind="stable")
    channel_state = {}
    resolved_state = {}
    events = []
    parameter_order = {base: index for index, base in enumerate(parameters)}
    for timestamp, timestamp_frame in engine_frame.groupby("Timestamp", sort=True):
        touched = set()
        for base in parameters:
            for channel in EC_CHANNELS:
                column = _ec_full_column(channel, base)
                if column not in available_columns:
                    continue
                populated = timestamp_frame[column].dropna()
                if not populated.empty:
                    channel_state[(base, channel)] = populated.iloc[-1]
                    touched.add(base)
        for base in parameters:
            if base not in touched:
                continue
            resolved = _ec_resolve_channels(
                channel_state.get((base, "AC")),
                channel_state.get((base, "BC")),
                discrete_by_parameter[base],
            )
            if resolved is None:
                continue
            if resolved_state.get(base) == resolved["key"]:
                continue
            resolved_state[base] = resolved["key"]
            events.append(
                {
                    "timestamp": pd.Timestamp(timestamp),
                    "engine": str(engine),
                    "parameter": base,
                    "parameter_order": parameter_order[base],
                    "value_key": resolved["key"],
                    "text": _ec_event_text(base, resolved),
                }
            )
    return events


def _ec_format_lag(milliseconds):
    absolute = abs(milliseconds)
    if np.isclose(absolute, round(absolute), rtol=0.0, atol=1e-6):
        return f"{int(round(absolute)):,} ms"
    return f"{absolute:,.3f}".rstrip("0").rstrip(".") + " ms"


def _ec_match_lags(events_by_engine, engines):
    first_engine, second_engine = engines
    first_events = events_by_engine[first_engine]
    second_events = events_by_engine[second_engine]
    candidates_by_key = {}
    for index, event in enumerate(second_events):
        key = (event["parameter"], event["value_key"])
        candidates_by_key.setdefault(key, []).append((index, event))
    used_second_indices = set()
    lag_by_row = {}
    for first_event in first_events:
        key = (first_event["parameter"], first_event["value_key"])
        candidates = [
            (index, event)
            for index, event in candidates_by_key.get(key, [])
            if index not in used_second_indices
            and abs((event["timestamp"] - first_event["timestamp"]).total_seconds()) <= EC_MAX_LAG_SECONDS
        ]
        if not candidates:
            continue
        second_index, second_event = min(
            candidates,
            key=lambda item: abs((item[1]["timestamp"] - first_event["timestamp"]).total_seconds()),
        )
        used_second_indices.add(second_index)
        milliseconds = (second_event["timestamp"] - first_event["timestamp"]).total_seconds() * 1000.0
        if np.isclose(milliseconds, 0.0, rtol=0.0, atol=1e-9):
            row_key = (first_event["timestamp"], first_event["parameter"])
            lag_by_row[row_key] = "0 ms"
        elif milliseconds > 0:
            row_key = (second_event["timestamp"], second_event["parameter"])
            lag_by_row[row_key] = f"{_ec_format_lag(milliseconds)} ({second_engine} later)"
        else:
            row_key = (first_event["timestamp"], first_event["parameter"])
            lag_by_row[row_key] = f"{_ec_format_lag(milliseconds)} ({first_engine} later)"
    return lag_by_row


def _ec_timestamp_text(timestamp):
    return pd.Timestamp(timestamp).strftime("%H:%M:%S.%f")[:-3]


def _ec_render_table(events_by_engine, engines, parameters, context_text):
    lag_by_row = _ec_match_lags(events_by_engine, engines)
    rows = {}
    for engine in engines:
        for event in events_by_engine[engine]:
            key = (event["timestamp"], event["parameter"])
            row = rows.setdefault(
                key,
                {
                    "timestamp": event["timestamp"],
                    "parameter": event["parameter"],
                    "parameter_order": event["parameter_order"],
                    "values": {},
                },
            )
            row["values"][engine] = event["text"]
    ordered_rows = sorted(
        rows.values(),
        key=lambda row: (row["timestamp"], row["parameter_order"]),
    )
    if len(ordered_rows) > EC_MAX_EVENT_ROWS:
        raise RuntimeError(
            f"The selected parameters generated {len(ordered_rows):,} change rows. "
            f"Reduce the parameter selection below {EC_MAX_EVENT_ROWS:,} rows."
        )
    color_map = dict(EC_COLORS)
    extra_index = 0
    for base in parameters:
        if base not in color_map:
            color_map[base] = EC_EXTRA_COLORS[extra_index % len(EC_EXTRA_COLORS)]
            extra_index += 1
    body = []
    for row in ordered_rows:
        base = row["parameter"]
        color = color_map[base]
        value_cells = []
        for engine in engines:
            value = row["values"].get(engine, "")
            style = f"background:{color};" if value else ""
            value_cells.append(
                f'<td class="ec-value" style="{style}">{html.escape(value)}</td>'
            )
        lag = lag_by_row.get((row["timestamp"], base), "")
        body.append(
            "<tr>"
            f'<td class="ec-time">{_ec_timestamp_text(row["timestamp"])}</td>'
            + "".join(value_cells)
            + f'<td class="ec-lag">{html.escape(lag)}</td>'
            + "</tr>"
        )
    if not body:
        body.append('<tr><td colspan="4" class="ec-empty">No value changes were found.</td></tr>')
    return f"""
    <style>
      .ec-wrap {{ width:100%; max-height:760px; overflow:auto; border:1px solid #555; background:#fff; }}
      .ec-context {{ margin:6px 0 10px; font-size:13px; color:#333; }}
      .ec-table {{ width:100%; min-width:920px; border-collapse:collapse; table-layout:fixed; font-size:13px; color:#111; }}
      .ec-table th {{ position:sticky; top:0; z-index:3; background:#efefef; border:1px solid #333; padding:7px 8px; text-align:center; font-weight:700; }}
      .ec-table td {{ border:1px solid #333; padding:5px 7px; vertical-align:middle; }}
      .ec-time {{ width:150px; white-space:nowrap; font-family:ui-monospace,SFMono-Regular,Menlo,Consolas,monospace; }}
      .ec-value {{ width:34%; overflow-wrap:anywhere; }}
      .ec-lag {{ width:190px; white-space:nowrap; font-family:ui-monospace,SFMono-Regular,Menlo,Consolas,monospace; }}
      .ec-empty {{ text-align:center; color:#666; font-style:italic; padding:18px !important; }}
    </style>
    <div class="ec-context">{html.escape(context_text)} Change rows only; lag pairs match the same signal and resulting value within {EC_MAX_LAG_SECONDS:g} seconds.</div>
    <div class="ec-wrap">
      <table class="ec-table">
        <thead><tr><th>Time</th><th>{html.escape(engines[0])}</th><th>{html.escape(engines[1])}</th><th>Lag</th></tr></thead>
        <tbody>{''.join(body)}</tbody>
      </table>
    </div>
    """


def _ec_month_filter(start_time, end_time):
    current = date(start_time.year, start_time.month, 1)
    final = date(end_time.year, end_time.month, 1)
    expression = F.lit(False)
    while current <= final:
        expression = expression | ((F.col("year") == current.year) & (F.col("Month") == current.month))
        current = date(current.year + (current.month == 12), 1 if current.month == 12 else current.month + 1, 1)
    return expression

def _ec_refresh_flights(_=None, raise_errors=False):
    ec_find_flights_button.disabled = True
    try:
        with ec_status_output:
            clear_output(wait=True)
            aircraft = ec_aircraft_widget.value
            flight_date = ec_date_widget.value
            if flight_date is None:
                raise ValueError("Choose a flight date.")
            calendar_id = int(flight_date.strftime("%Y%m%d"))
            rows = (
                spark.table(EC_MASTER_TABLE)
                .where(
                    (F.col("AircraftIdentifier").cast("string") == aircraft)
                    & (F.col("CalendarId") == calendar_id)
                )
                .select(
                    "StartDatetime",
                    "EndDatetime",
                    F.col("EngineSerialNumber").cast("string").alias("EngineSerialNumber"),
                )
                .where(F.col("StartDatetime").isNotNull() & F.col("EndDatetime").isNotNull())
                .groupBy("StartDatetime")
                .agg(
                    F.max("EndDatetime").alias("EndDatetime"),
                    F.sort_array(F.collect_set("EngineSerialNumber")).alias("Engines"),
                )
                .where(F.col("EndDatetime") >= F.col("StartDatetime"))
                .orderBy("StartDatetime")
                .collect()
            )
            options = []
            for row in rows:
                engine_count = len([engine for engine in row.Engines if engine is not None])
                label = (
                    f"{row.StartDatetime.strftime('%H:%M:%S.%f')[:-3]} – "
                    f"{row.EndDatetime.strftime('%H:%M:%S.%f')[:-3]} | {engine_count} engines"
                )
                options.append((label, row.StartDatetime))
            ec_flight_widget.options = options
            if options:
                ec_flight_widget.value = options[0][1]
                print(f"Found {len(options)} flights for {aircraft} on {flight_date}.")
            else:
                print(f"No flights were found for {aircraft} on {flight_date}.")
    except Exception as exc:
        ec_flight_widget.options = []
        with ec_status_output:
            print(f"Unable to find flights: {exc}")
        if raise_errors:
            raise
    finally:
        ec_find_flights_button.disabled = False


def _ec_load_comparison(_):
    ec_load_button.disabled = True
    try:
        with ec_table_output:
            clear_output(wait=True)
            aircraft = ec_aircraft_widget.value
            flight_date = ec_date_widget.value
            start_time = ec_flight_widget.value
            parameters = _ec_selected_parameters()
            if flight_date is None or start_time is None:
                raise ValueError("Choose a date and flight before loading the comparison.")
            if not parameters:
                raise ValueError("Select at least one signal.")
            calendar_id = int(flight_date.strftime("%Y%m%d"))
            master_rows = (
                spark.table(EC_MASTER_TABLE)
                .where(
                    (F.col("AircraftIdentifier").cast("string") == aircraft)
                    & (F.col("CalendarId") == calendar_id)
                    & (F.col("StartDatetime") == F.lit(start_time))
                )
                .select(
                    F.col("EngineSerialNumber").cast("string").alias("EngineSerialNumber"),
                    "AircraftId",
                    "EndDatetime",
                )
                .where(
                    F.col("EngineSerialNumber").isNotNull()
                    & F.col("AircraftId").isNotNull()
                    & F.col("EndDatetime").isNotNull()
                )
                .collect()
            )
            engine_metadata = {}
            for row in master_rows:
                metadata = engine_metadata.setdefault(
                    row.EngineSerialNumber,
                    {"aircraft_ids": set(), "end_times": []},
                )
                metadata["aircraft_ids"].add(row.AircraftId)
                metadata["end_times"].append(row.EndDatetime)
            engines = sorted(engine_metadata)
            if len(engines) != 2:
                raise ValueError(
                    f"Expected exactly two engines for this flight; found {len(engines)}: {engines}"
                )
            end_time = max(
                end
                for engine in engines
                for end in engine_metadata[engine]["end_times"]
            )
            aircraft_ids = list(
                {
                    aircraft_id
                    for engine in engines
                    for aircraft_id in engine_metadata[engine]["aircraft_ids"]
                }
            )
            selected_columns = ["Timestamp", "AssetIdentifier"]
            for base in parameters:
                for channel in EC_CHANNELS:
                    column = _ec_full_column(channel, base)
                    if column in _ec_scan_schema_set:
                        selected_columns.append(column)
            selected_columns = list(dict.fromkeys(selected_columns))
            source = (
                spark.table(EC_SCAN_TABLE)
                .where(
                    F.col("AircraftId").isin(aircraft_ids)
                    & F.col("AssetIdentifier").cast("string").isin(engines)
                    & (F.col("StartDatetime") == F.lit(start_time))
                    & F.col("Timestamp").between(F.lit(start_time), F.lit(end_time))
                    & _ec_month_filter(start_time, end_time)
                )
                .select(*[_ec_quote(column).alias(column) for column in selected_columns])
                .orderBy("Timestamp")
            )
            row_count = source.count()
            if row_count == 0:
                raise ValueError("No scan samples were found for the selected flight and engines.")
            if row_count > EC_MAX_PANDAS_ROWS:
                raise RuntimeError(
                    f"The flight contains {row_count:,} samples, above the {EC_MAX_PANDAS_ROWS:,}-row safety limit."
                )
            frame = source.toPandas()
            frame["Timestamp"] = pd.to_datetime(frame["Timestamp"], errors="coerce")
            available_columns = set(frame.columns)
            discrete_by_parameter = {
                base: _ec_parameter_is_discrete(frame, base, available_columns)
                for base in parameters
            }
            events_by_engine = {
                engine: _ec_extract_events(
                    frame,
                    engine,
                    parameters,
                    available_columns,
                    discrete_by_parameter,
                )
                for engine in engines
            }
            context = (
                f"Aircraft {aircraft} | Flight {flight_date} "
                f"{_ec_timestamp_text(start_time)}–{_ec_timestamp_text(end_time)} | "
                f"{row_count:,} samples | AC primary, BC fallback | "
                f"AC/BC divergence threshold {EC_DIVERGENCE_THRESHOLD:.0%}."
            )
            ipy_display(HTML(_ec_render_table(events_by_engine, engines, parameters, context)))
    except Exception as exc:
        with ec_table_output:
            print(f"Unable to load comparison: {exc}")
    finally:
        ec_load_button.disabled = False


ec_parameter_search.observe(_ec_add_parameter, names="value")
