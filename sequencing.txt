def _tr_event_format_value(value):
    if value is None:
        return ""
    value = float(value)
    if value.is_integer():
        return str(int(value))
    return f"{value:g}"


def _tr_event_render_table(events, chosen_aircraft, chosen_engine, chosen_date):
    if not events:
        ipy_display(
            HTML(
                f"<b>No AC thrust-reverser transitions found</b><br>"
                f"Aircraft: {html.escape(str(chosen_aircraft))} &nbsp;|&nbsp; "
                f"Engine: {html.escape(str(chosen_engine))} &nbsp;|&nbsp; "
                f"Date: {html.escape(str(chosen_date))}"
            )
        )
        return

    signal_styles = {
        "LVTPosVCntrl": ("#eadfe8", "#000000"),
        "Low TRLVT Raw": ("#d4a900", "#000000"),
        "Up TRLVT Raw": ("#2e3192", "#ffffff"),
        "TR Armed": ("#009b35", "#000000"),
        "DoorDep Status": ("#d8c784", "#000000"),
        "In Transit": ("#5fd0d4", "#000000"),
        "Is Unlocked": ("#b7b7b7", "#000000"),
        "Door Stow": ("#f0a000", "#000000"),
        "Is Deployed": ("#269b91", "#000000"),
    }

    rows = []
    for event in events:
        bg, fg = signal_styles[event["Signal"]]
        timestamp_text = event["Timestamp"].strftime("%H:%M:%S.%f")[:-3]

            f"{html.escape(event['Signal'])} = "
            f"{html.escape(_tr_event_format_value(event['Value']))}"
        )
        rows.append(
            "<tr>"
            f"<td class='tr-time'>{timestamp_text}</td>"
            f"<td class='tr-event' style='background:{bg};color:{fg}'>{event_text}</td>"
            "</tr>"
        )

    table_html = f"""
    <style>
      .tr-wrap {{
        display: inline-block;
        font-family: Arial, Helvetica, sans-serif;
        margin-top: 8px;
      }}
      .tr-meta {{
        margin: 0 0 8px 0;
        font-size: 13px;
      }}
      .tr-table {{
        border-collapse: collapse;
        border: 2px solid #222;
        font-size: 14px;
      }}
      .tr-table th {{
        border: 1px solid #222;
        background: #f2f2f2;
        padding: 4px 7px;
        text-align: left;
        font-weight: 600;
      }}
      .tr-table td {{
        border: 1px solid #222;
        padding: 2px 6px;
        line-height: 1.2;
      }}
      .tr-time {{
        white-space: nowrap;
        background: #ffffff;
        font-family: ui-monospace, SFMono-Regular, Menlo, Consolas, monospace;
      }}
      .tr-event {{

    </style>
    <div class="tr-wrap">
      <div class="tr-meta">

        &nbsp;|&nbsp;

        &nbsp;|&nbsp;
        Channel: <b>AC</b>
        &nbsp;|&nbsp;

        &nbsp;|&nbsp;

      <table class="tr-table">
        <thead>
          <tr>
            <th>Time</th>
            <th>{html.escape(str(chosen_engine))} — AC</th>

def _run_tr_event_view(_):
    tr_event_button.disabled = True

    try:
        with tr_event_output:
            clear_output(wait=True)

            chosen_date = tr_event_date_widget.value
            chosen_aircraft = aircraft_widget.value
            chosen_engine = engine_widget.value

            if chosen_date is None:
                raise ValueError("Select an event date.")

            if chosen_aircraft not in aircraft_ids_by_identifier:
                raise ValueError(f"Unknown aircraft: {chosen_aircraft}")

            if chosen_engine not in engine_ids_by_aircraft.get(chosen_aircraft, {}):
                raise ValueError(
                    f"Engine {chosen_engine} is not mapped to aircraft {chosen_aircraft}."
                )

            internal_ids = aircraft_ids_by_identifier[chosen_aircraft]
            if len(internal_ids) != 1:
                raise ValueError(
                    f"Aircraft {chosen_aircraft} does not map to exactly one internal aircraft ID."
                )

            chosen_internal_aircraft_id = next(iter(internal_ids))
            chosen_calendar_id = int(chosen_date.strftime("%Y%m%d"))

            master_day_df = (
                spark.table(MASTER_TABLE)
                .where(
                    (F.col("AircraftIdentifier").cast("string") == chosen_aircraft)
                    & (F.col("EngineSerialNumber").cast("string") == chosen_engine)
                    & (F.col("CalendarId") == chosen_calendar_id)
                )
            )

            day_windows_df = (
                master_day_df
                .select("StartDatetime", "EndDatetime")
                .where(
                    F.col("StartDatetime").isNotNull()
                    & F.col("EndDatetime").isNotNull()
                )
                .groupBy("StartDatetime")
                .agg(F.max("EndDatetime").alias("EndDatetime"))
                .where(F.col("EndDatetime") >= F.col("StartDatetime"))
            )

            day_windows = (
                day_windows_df
                .orderBy("StartDatetime")
                .collect()
            )

            if not day_windows:
                print(
                    f"No flight windows found for aircraft {chosen_aircraft}, "
                    f"engine {chosen_engine}, on {chosen_date}."
                )
                return

            scan_schema = spark.table(SCAN_TABLE)
            scan_columns = set(scan_schema.columns)

            signal_specs = []
            missing_columns = []

            for display_name, suffix, signal_type in TR_EVENT_SIGNALS:
                full_column = f"{PARAMETER_PREFIX}AC_{suffix}"
                if full_column not in scan_columns:
                    missing_columns.append(full_column)
                else:
                    signal_specs.append(
                        (display_name, full_column, signal_type)
                    )

            if missing_columns:
                raise ValueError(
                    "Required AC columns are missing from the scan schema:\n"
                    + "\n".join(missing_columns)
                )

            range_start = min(row.StartDatetime for row in day_windows)
            range_end = max(row.EndDatetime for row in day_windows)

            partition_filter = F.lit(False)
            for partition_year, partition_month in _month_partitions_between_dates(
                range_start.date(),
                range_end.date(),
            ):
                partition_filter = partition_filter | (
                    (F.col("year") == partition_year)
                    & (F.col("Month") == partition_month)
                )

            scan_alias = (
                spark.table(SCAN_TABLE)
                .where(
                    (F.col("AircraftId") == chosen_internal_aircraft_id)
                    & (F.col("AssetIdentifier").cast("string") == chosen_engine)
                    & partition_filter
                )
                .alias("scan")
            )

            window_alias = F.broadcast(day_windows_df).alias("window")

            joined_df = (
                scan_alias
                .join(
                    window_alias,
                    (
                        (scan_alias["StartDatetime"] == window_alias["StartDatetime"])
                        & scan_alias["Timestamp"].between(
                            window_alias["StartDatetime"],
                            window_alias["EndDatetime"],
                        )
                    ),
                    "inner",
                )
                .select(
                    scan_alias["Timestamp"].alias("Timestamp"),
                    scan_alias["StartDatetime"].alias("FlightStartDatetime"),
                    *[
                        scan_alias[column].alias(column)
                        for _, column, _ in signal_specs
                    ],
                )
            )

            event_frames = []

            for signal_order, (display_name, column_name, signal_type) in enumerate(signal_specs):
                value_column = F.col(f"`{column_name}`").cast("double")

                if signal_type == "analog":
                    normalized_value = (
                        F.when(
                            value_column.between(0.0, 3.0),
                            F.lit(0.0),
                        )
                        .when(
                            value_column.between(97.0, 100.0),
                            F.lit(100.0),
                        )
                    )

                    normalized_df = (
                        joined_df
                        .select(
                            "FlightStartDatetime",
                            "Timestamp",
                            normalized_value.alias("EventValue"),
                        )
                        .where(F.col("EventValue").isNotNull())
                    )
                else:
                    normalized_df = (
                        joined_df
                        .select(
                            "FlightStartDatetime",
                            "Timestamp",
                            value_column.alias("EventValue"),
                        )
                        .where(F.col("EventValue").isNotNull())
                    )

                transition_window = (
                    Window
                    .partitionBy("FlightStartDatetime")
                    .orderBy(F.col("Timestamp"))
                )

                transition_df = (
                    normalized_df
                    .withColumn(
                        "PreviousValue",
                        F.lag("EventValue").over(transition_window),
                    )
                    .where(
                        F.col("PreviousValue").isNotNull()
                        & (F.col("EventValue") != F.col("PreviousValue"))
                    )
                    .select(
                        "Timestamp",
                        F.lit(display_name).alias("Signal"),
                        "EventValue",
                        F.lit(signal_order).alias("SignalOrder"),
                    )
                )

                event_frames.append(transition_df)

            if not event_frames:
                _tr_event_render_table(
                    [],
                    chosen_aircraft,
                    chosen_engine,
                    chosen_date,
                )
                return

            events_df = event_frames[0]
            for event_df in event_frames[1:]:
                events_df = events_df.unionByName(event_df)

            event_rows = (
                events_df
                .orderBy(
                    F.col("Timestamp"),
                    F.col("SignalOrder"),
                )
                .collect()
            )

            events = [
                {
                    "Timestamp": row.Timestamp,
                    "Signal": row.Signal,
                    "Value": row.EventValue,
                }
                for row in event_rows
            ]

            _tr_event_render_table(
                events,
                chosen_aircraft,
                chosen_engine,
                chosen_date,
            )

    finally:
        tr_event_button.disabled = False


tr_event_button.on_click(_run_tr_event_view)
