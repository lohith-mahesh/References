def _full_parameter_name(short_name):
    return (
        short_name
        if short_name.startswith(PARAMETER_PREFIX)
        else PARAMETER_PREFIX + short_name
    )


def _channel_and_signal(short_name):
    for channel in CHANNELS:
        prefix = f"{channel}_"
        if short_name.startswith(prefix):
            return channel, short_name[len(prefix):]
    return "", short_name


def _numeric_series(series):
    non_null = series.dropna()

    if non_null.empty:
        return pd.Series(
            np.nan,
            index=series.index,
            dtype="float64",
        ), False

    if non_null.map(
        lambda value: isinstance(value, (bool, np.bool_))
    ).all():
        return series.map({True: 1.0, False: 0.0}), True

    numeric = pd.to_numeric(series, errors="coerce")
    values = numeric.dropna().to_numpy(dtype=float)

    discrete = bool(
        len(values)
        and len(np.unique(values)) <= 8
        and np.allclose(values, np.round(values))
    )

    return numeric, discrete


def _match_timestamp_timezone(value, reference):
    timestamp = pd.Timestamp(value)

    if reference.tzinfo is None and timestamp.tzinfo is not None:
        return timestamp.tz_localize(None)

    if reference.tzinfo is not None and timestamp.tzinfo is None:
        return timestamp.tz_localize(reference.tzinfo)

    if reference.tzinfo is not None and timestamp.tzinfo is not None:
        return timestamp.tz_convert(reference.tzinfo)

    return timestamp


def _format_gap_duration(seconds):
    seconds = max(0, int(round(seconds)))
    days, remainder = divmod(seconds, 86400)
    hours, remainder = divmod(remainder, 3600)
    minutes, seconds = divmod(remainder, 60)

    if days:
        return f"{days}d {hours:02d}h"

    if hours:
        return f"{hours}h {minutes:02d}m"

    if minutes:
        return f"{minutes}m {seconds:02d}s"

    return f"{seconds}s"


def _build_compressed_timeline(frame):
    raw_flights = []

    for key, group in frame.groupby(
        "StartDatetime",
        sort=False,
        dropna=False,
    ):
        group = group.sort_values(
            "Timestamp",
            kind="stable",
        ).copy()

        actual_start = group["Timestamp"].iloc[0]
        actual_end = group["Timestamp"].iloc[-1]

        deltas = (
            group["Timestamp"]
            .drop_duplicates()
            .sort_values()
            .diff()
            .dt.total_seconds()
        )
        deltas = deltas[deltas > 0]

        cadence = (
            float(deltas.median())
            if not deltas.empty
            else 1.0
        )

        raw_flights.append(
            {
                "key": key,
                "frame": group,
                "actual_start": actual_start,
                "actual_end": actual_end,
                "cadence": max(cadence, 0.001),
                "duration": max(
                    float(
                        (
                            actual_end
                            - actual_start
                        ).total_seconds()
                    ),
                    cadence,
                    1.0,
                ),
            }
        )

    raw_flights.sort(
        key=lambda item: item["actual_start"]
    )

    active_duration = sum(
        item["duration"]
        for item in raw_flights
    )

    gap_count = max(
        0,
        len(raw_flights) - 1,
    )

    display_gap = (
        max(
            active_duration
            * min(
                0.02,
                0.15 / gap_count,
            ),
            1.0,
        )
        if gap_count
        else 0.0
    )

    cursor = 0.0
    timeline_parts = []
    flights = []
    gaps = []
    previous = None

    for item in raw_flights:
        if previous is not None:
            gap_start = cursor
            cursor += display_gap

            gaps.append(
                {
                    "start": gap_start,
                    "end": cursor,
                    "center": (
                        gap_start + cursor
                    ) / 2.0,
                    "actual_seconds": max(
                        0.0,
                        float(
                            (
                                item["actual_start"]
                                - previous["actual_end"]
                            ).total_seconds()
                        ),
                    ),
                }
            )

        plot_start = cursor
        part = item["frame"].copy()

        part["_plot_x"] = (
            plot_start
            + (
                part["Timestamp"]
                - item["actual_start"]
            ).dt.total_seconds()
        )

        plot_end = (
            plot_start
            + item["duration"]
        )

        part["_flight_order"] = len(flights)
        timeline_parts.append(part)

        flights.append(
            {
                **item,
                "plot_start": plot_start,
                "plot_end": plot_end,
            }
        )

        cursor = plot_end
        previous = item

    timeline = (
        pd.concat(
            timeline_parts,
            ignore_index=True,
        )
        if timeline_parts
        else frame.iloc[0:0].copy()
    )

    return timeline, flights, gaps, cursor


def _timeline_ticks(flights, maximum=14):
    if not flights:
        return [], []

    if len(flights) > 7:
        indices = np.unique(
            np.linspace(
                0,
                len(flights) - 1,
                min(maximum, len(flights)),
                dtype=int,
            )
        )

        specifications = [
            (flights[index], 0.0)
            for index in indices
        ]

    else:
        count = max(
            1,
            maximum // len(flights),
        )

        specifications = [
            (flight, float(fraction))
            for flight in flights
            for fraction in np.linspace(
                0.0,
                1.0,
                count,
            )
        ]

    positions = []
    labels = []

    for flight, fraction in specifications:
        actual_duration = max(
            0.0,
            float(
                (
                    flight["actual_end"]
                    - flight["actual_start"]
                ).total_seconds()
            ),
        )

        timestamp = (
            flight["actual_start"]
            + pd.to_timedelta(
                actual_duration * fraction,
                unit="s",
            )
        )

        positions.append(
            flight["plot_start"]
            + flight["duration"] * fraction
        )

        labels.append(
            timestamp.strftime(
                "%d %b %Y\n%H:%M:%S"
            )
        )

    return positions, labels


def _draw_axis_break(
    axis,
    gap,
    show_label=False,
):
    transform = blended_transform_factory(
        axis.transData,
        axis.transAxes,
    )

    half_width = max(
        (
            gap["end"]
            - gap["start"]
        )
        * 0.32,
        0.25,
    )

    x = np.linspace(
        gap["center"] - half_width,
        gap["center"] + half_width,
        7,
    )

    y = np.array(
        [
            -0.018,
            0.018,
            -0.018,
            0.018,
            -0.018,
            0.018,
            -0.018,
        ]
    )

    axis.plot(
        x,
        y,
        color="#333333",
        linewidth=1.3,
        transform=transform,
        clip_on=False,
        zorder=8,
    )

    axis.axvline(
        gap["center"],
        color="#777777",
        linewidth=0.7,
        linestyle=":",
        alpha=0.45,
    )

    if show_label:
        axis.annotate(
            (
                "Gap\n"
                f"{_format_gap_duration(gap['actual_seconds'])}"
            ),
            xy=(gap["center"], 0),
            xycoords=transform,
            xytext=(0, -42),
            textcoords="offset points",
            ha="center",
            va="top",
            fontsize=8,
            color="#444444",
            annotation_clip=False,
        )


def _plot_parameters(
    start_time,
    end_time,
):
    chosen = list(selected_parameters)

    if not chosen:
        print(
            "No plot inputs are applied. "
            "Add at least one parameter "
            "in the previous cell."
        )
        return

    grouped = OrderedDict()

    for short_name in chosen:
        channel, signal = (
            _channel_and_signal(
                short_name
            )
        )

        grouped.setdefault(
            signal,
            [],
        ).append(
            (
                channel,
                short_name,
            )
        )

    if len(grouped) > MAX_SIGNAL_GROUPS:
        print(
            f"Reduce the selection to "
            f"{MAX_SIGNAL_GROUPS} "
            "signal groups or fewer."
        )
        return

    visible = data_df.loc[
        data_df["Timestamp"].between(
            start_time,
            end_time,
            inclusive="both",
        )
    ].copy()

    if visible.empty:
        print(
            "No samples fall inside "
            "the requested time range."
        )
        return

    (
        timeline,
        flights,
        gaps,
        timeline_end,
    ) = _build_compressed_timeline(
        visible
    )

    width = float(
        plot_width_widget.value
    )

    height = max(
        5.5,
        float(
            subplot_height_widget.value
        ) * len(grouped),
    )

    dpi = int(
        plot_dpi_widget.value
    )

    if (
        width
        * height
        * dpi**2
        > MAX_RENDER_PIXELS
    ):
        print(
            "The requested graph size is too large. "
            "Reduce width, height, or DPI."
        )
        return

    figure, axes = plt.subplots(
        len(grouped),
        1,
        figsize=(width, height),
        dpi=dpi,
        sharex=True,
        squeeze=False,
    )

    skipped = []

    for axis, (
        signal,
        members,
    ) in zip(
        axes[:, 0],
        grouped.items(),
    ):
        plotted = False
        red_segments = []

        for channel, short_name in members:
            full_name = (
                _full_parameter_name(
                    short_name
                )
            )

            if full_name not in timeline.columns:
                skipped.append(
                    short_name
                )
                continue

            frame = timeline[
                [
                    "Timestamp",
                    "_plot_x",
                    "_flight_order",
                    full_name,
                ]
            ].copy()

            (
                frame["_value"],
                discrete,
            ) = _numeric_series(
                frame[full_name]
            )

            frame = frame.dropna(
                subset=["_value"]
            )

            if frame.empty:
                skipped.append(
                    short_name
                )
                continue

            solid_x = []
            solid_y = []
            previous_flight_endpoint = None

            for (
                flight_order,
                flight_frame,
            ) in frame.groupby(
                "_flight_order",
                sort=True,
            ):
                flight_frame = (
                    flight_frame.sort_values(
                        "Timestamp",
                        kind="stable",
                    )
                )

                deltas = (
                    flight_frame[
                        "Timestamp"
                    ]
                    .diff()
                    .dt.total_seconds()
                )

                threshold = max(
                    3.0
                    * flights[
                        int(flight_order)
                    ]["cadence"],
                    5.0,
                )

                chunks = [
                    chunk
                    for _, chunk
                    in flight_frame.groupby(
                        deltas
                        .gt(threshold)
                        .cumsum(),
                        sort=True,
                    )
                ]

                first = chunks[0].iloc[0]

                if (
                    previous_flight_endpoint
                    is not None
                ):
                    red_segments.append(
                        [
                            previous_flight_endpoint,
                            (
                                float(
                                    first[
                                        "_plot_x"
                                    ]
                                ),
                                float(
                                    first[
                                        "_value"
                                    ]
                                ),
                            ),
                        ]
                    )

                for index, chunk in enumerate(
                    chunks
                ):
                    if index:
                        left = (
                            chunks[
                                index - 1
                            ].iloc[-1]
                        )

                        right = chunk.iloc[0]

                        red_segments.append(
                            [
                                (
                                    float(
                                        left[
                                            "_plot_x"
                                        ]
                                    ),
                                    float(
                                        left[
                                            "_value"
                                        ]
                                    ),
                                ),
                                (
                                    float(
                                        right[
                                            "_plot_x"
                                        ]
                                    ),
                                    float(
                                        right[
                                            "_value"
                                        ]
                                    ),
                                ),
                            ]
                        )

                    solid_x.extend(
                        chunk[
                            "_plot_x"
                        ].to_numpy(
                            dtype=float
                        )
                    )

                    solid_y.extend(
                        chunk[
                            "_value"
                        ].to_numpy(
                            dtype=float
                        )
                    )

                    solid_x.append(np.nan)
                    solid_y.append(np.nan)

                last = chunks[-1].iloc[-1]

                previous_flight_endpoint = (
                    float(
                        last[
                            "_plot_x"
                        ]
                    ),
                    float(
                        last[
                            "_value"
                        ]
                    ),
                )

            axis.plot(
                solid_x,
                solid_y,
                color=CHANNEL_COLORS.get(
                    channel,
                    "#333333",
                ),
                linewidth=1.25,
                drawstyle=(
                    "steps-post"
                    if discrete
                    else "default"
                ),
                label=(
                    channel
                    or short_name
                ),
                zorder=3,
            )

            plotted = True

        if red_segments:
            axis.add_collection(
                LineCollection(
                    red_segments,
                    colors=NO_DATA_COLOR,
                    linewidths=1.2,
                    linestyles=(
                        0,
                        (2, 3),
                    ),
                    zorder=2,
                ),
                autolim=False,
            )

        axis.set_ylabel(
            signal,
            fontsize=9,
        )

        axis.grid(
            True,
            alpha=0.22,
        )

        axis.margins(
            y=0.08
        )

        if axis is not axes[-1, 0]:
            for gap in gaps:
                _draw_axis_break(
                    axis,
                    gap,
                )

        if plotted:
            axis.legend(
                loc="upper right",
                ncol=max(
                    1,
                    len(members),
                ),
                framealpha=0.85,
            )
        else:
            axis.text(
                0.5,
                0.5,
                "No numeric samples",
                ha="center",
                va="center",
                transform=axis.transAxes,
            )

    bottom = axes[-1, 0]

    (
        tick_positions,
        tick_labels,
    ) = _timeline_ticks(
        flights
    )

    bottom.set_xticks(
        tick_positions
    )

    bottom.set_xticklabels(
        tick_labels,
        rotation=25,
        ha="right",
        fontsize=8,
    )

    bottom.set_xlabel(
        (
            "Flight time - calendar timestamps shown; "
            "inter-flight gaps compressed"
        ),
        labelpad=(
            48
            if gaps
            else 12
        ),
    )

    for gap in gaps:
        _draw_axis_break(
            bottom,
            gap,
            show_label=True,
        )

    for axis in axes[:, 0]:
        axis.set_xlim(
            0.0,
            max(
                timeline_end,
                1.0,
            ),
        )

    figure.legend(
        handles=[
            Line2D(
                [0],
                [0],
                color=NO_DATA_COLOR,
                linewidth=1.5,
                linestyle=(
                    0,
                    (2, 3),
                ),
                label="No recorded data",
            )
        ],
        loc="lower center",
        bbox_to_anchor=(
            0.5,
            0.012,
        ),
        frameon=False,
    )

    figure.suptitle(
        (
            f"Aircraft {selected_acid} | "
            f"Engine {selected_esn} | "
            f"{len(flights)} flights | "
            f"{start_time} through {end_time}"
        ),
        fontsize=14,
    )

    figure.text(
        0.5,
        0.055,
        (
            "Axis breaks compress inter-flight gaps. "
            "Red dotted connectors are not measured data."
        ),
        ha="center",
        fontsize=9,
        color="#444444",
    )

    figure.tight_layout(
        rect=(
            0.02,
            0.095,
            0.99,
            0.95,
        )
    )

    ipy_display(figure)
    plt.close(figure)

    if skipped:
        print(
            "Skipped parameters without numeric samples: "
            + ", ".join(
                sorted(
                    set(skipped)
                )
            )
        )


if data_df.empty:
    print(
        "No samples are available "
        "for the selected flights and engine."
    )

else:
    all_timestamps = pd.DatetimeIndex(
        data_df[
            "Timestamp"
        ]
        .drop_duplicates()
        .sort_values()
    )

    if len(all_timestamps) > MAX_SLIDER_POINTS:
        indices = np.unique(
            np.linspace(
                0,
                len(all_timestamps) - 1,
                MAX_SLIDER_POINTS,
                dtype=int,
            )
        )

        slider_timestamps = [
            all_timestamps[index]
            for index in indices
        ]

    else:
        slider_timestamps = list(
            all_timestamps
        )

    def _timestamp_label(timestamp):
        return (
            pd.Timestamp(timestamp)
            .strftime(
                "%Y-%m-%d %H:%M:%S.%f"
            )
            .rstrip("0")
            .rstrip(".")
        )


    def _nearest_timestamp(value):
        timestamp = (
            _match_timestamp_timezone(
                value,
                all_timestamps[0],
            )
        )

        index = (
            all_timestamps.get_indexer(
                [timestamp],
                method="nearest",
            )[0]
        )

        return pd.Timestamp(
            all_timestamps[index]
        )


    time_slider = SelectionRangeSlider(
        options=[
            (
                _timestamp_label(

                ),
                timestamp,
            )
            for timestamp
            in slider_timestamps
        ],

            0,
            len(slider_timestamps) - 1,
        ),
        description="Time range:",
        continuous_update=False,
        layout=widgets.Layout(
            width="1200px"
        ),
    )

    txt_start = widgets.Text(
        value=_timestamp_label(
            all_timestamps[0]
        ),
        description="Start:",
        layout=widgets.Layout(
            width="500px"
        ),
    )

    txt_end = widgets.Text(
        value=_timestamp_label(
            all_timestamps[-1]
        ),
        description="End:",
        layout=widgets.Layout(
            width="500px"
        ),
    )

    plot_width_widget = (
        widgets.BoundedFloatText(
            value=22.0,
            min=10.0,
            max=60.0,
            step=2.0,
            description="Width (in):",
            layout=widgets.Layout(
                width="230px"
            ),
        )
    )

    subplot_height_widget = (
        widgets.BoundedFloatText(
            value=3.2,
            min=2.0,
            max=10.0,
            step=0.5,
            description="Height/plot:",
            layout=widgets.Layout(
                width="230px"
            ),
        )
    )

    plot_dpi_widget = (
        widgets.BoundedIntText(
            value=110,
            min=72,
            max=220,
            step=10,
            description="DPI:",
            layout=widgets.Layout(
                width="190px"
            ),
        )
    )

    plot_btn = widgets.Button(
        description="Plot",
        button_style="success",
        icon="line-chart",
        layout=widgets.Layout(
            width="150px"
        ),
    )

    plot_output = widgets.Output(
        layout=widgets.Layout(
            width="100%",
            overflow_x="auto",
            overflow_y="auto",
        )
    )

    def _render_range(
        start_time,
        end_time,
    ):
        plot_output.clear_output(
            wait=True
        )

        with plot_output:
            try:
                _plot_parameters(
                    min(
                        start_time,
                        end_time,
                    ),
                    max(
                        start_time,
                        end_time,
                    ),
                )
            except Exception:
                traceback.print_exc()


    def _on_plot_click(_):
        try:
            start_time = (
                _nearest_timestamp(
                    txt_start.value
                )
            )

            end_time = (
                _nearest_timestamp(
                    txt_end.value
                )
            )

        except Exception:
            plot_output.clear_output(
                wait=True
            )

            with plot_output:
                print(
                    "Start and end must be "
                    "valid date-time values."
                )

            return

        txt_start.value = (
            _timestamp_label(
                start_time
            )
        )

        txt_end.value = (
            _timestamp_label(
                end_time
            )
        )

        _render_range(
            start_time,
            end_time,
        )


    def _on_slider_change(change):
        start_time, end_time = map(
            pd.Timestamp,
            change["new"],
        )

        txt_start.value = (
            _timestamp_label(
                start_time
            )
        )

        txt_end.value = (
            _timestamp_label(
                end_time
            )
        )

        _render_range(
            start_time,
            end_time,
        )


    plot_btn.on_click(
        _on_plot_click
    )

    time_slider.observe(
        _on_slider_change,
        names="value",
    )

    size_controls = widgets.HBox(
        [
            plot_width_widget,
            subplot_height_widget,
            plot_dpi_widget,
            plot_btn,
        ]
    )

    controls = widgets.VBox(
        [
            txt_start,

            size_controls,
        ]
    )

    ipy_display(
        widgets.VBox(
            [
                controls,
                plot_output,
            ]
        )
    )

    _render_range(
        all_timestamps[0],
        all_timestamps[-1],
    )
