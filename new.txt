def _filter_cda(_):
    global filtered_cda_df
    with filter_output:
        clear_output(wait=True)
        flight_no = flight_no_widget.value.strip()
        start_date = start_date_widget.value
        end_date = end_date_widget.value

        if not flight_no:
            raise ValueError("Flight No. is required.")
        if start_date is None or end_date is None:
            raise ValueError("Both Start Date and End Date are required.")
        if start_date > end_date:
            raise ValueError("Start Date must not be later than End Date.")

        start_timestamp = datetime.combine(start_date, datetime.min.time())
        end_timestamp_exclusive = datetime.combine(end_date + timedelta(days=1), datetime.min.time())

        filtered_cda_df = (
            cda_catalog
            .where(
                (F.col("ParentAssetIdentifier").cast("string") == flight_no)
                & (F.col("Timestamp") >= F.lit(start_timestamp))
                & (F.col("Timestamp") < F.lit(end_timestamp_exclusive))
            )
            .select(*(OUTPUT_IDENTIFIER_COLUMNS + available_thrust_reverser_columns))
            .orderBy(F.col("Timestamp").asc(), F.col("AssetIdentifier").cast("string").asc())
        )

        print(f"Flight No.: {flight_no}")
        print(f"Date range: {start_date} through {end_date}")
        print(f"Thrust-reverser columns returned: {len(available_thrust_reverser_columns)} / {len(THRUST_REVERSER_COLUMNS)}")
        if missing_thrust_reverser_columns:
            print(f"Requested thrust-reverser columns absent from the CDA schema: {len(missing_thrust_reverser_columns)}")
        display(filtered_cda_df)

filter_button.on_click(_filter_cda)

    raise RuntimeError("Run Cell 1, choose the aircraft/date range, and click 'Filter CDA' before running Cell 2.")

AC_PREFIX = "Parameter:EVHMU_EEC_AC_"
EVENT_SOURCES = [
    ("TR Armed", "IThrustReverser_trArmed_data", "raw"),
    ("DoorDep Status", "IThrustReverser_trDoorDeplStatus_data", "raw"),
    ("Door Stow Status", "IThrustReverser_trDoorStowStatus_data", "raw"),
    ("In Transit", "IThrustReverser_trInTransit_data", "boolean"),
    ("IsDeployed", "IThrustReverser_trIsDeployed_data", "raw"),

    ("Low TRLVT", "IThrustReverser_loOwnTRLVTRaw_data", "lvt"),
    ("Up TRLVT", "IThrustReverser_upOwnTRLVTRaw_data", "lvt"),
]

filtered_columns = set(filtered_cda_df.columns)
required_columns = {"Timestamp", "AssetIdentifier"}
missing_required = sorted(required_columns - filtered_columns)
if missing_required:
    raise ValueError(f"filtered_cda_df is missing required columns: {missing_required}")

available_events = []
missing_events = []
for label, suffix, event_type in EVENT_SOURCES:
    source_column = f"{AC_PREFIX}{suffix}"
    if source_column in filtered_columns:
        available_events.append((label, source_column, event_type))
    else:
        missing_events.append(label)

if not available_events:
    raise ValueError("None of the required AC thrust-reverser event parameters are present in filtered_cda_df.")

engine_ids = [
    row.EngineNo
    for row in (
        filtered_cda_df
        .select(F.col("AssetIdentifier").cast("string").alias("EngineNo"))
        .where(F.col("EngineNo").isNotNull())
        .distinct()
        .orderBy(F.col("EngineNo"))
        .collect()
    )
]

if not engine_ids:
    raise ValueError("filtered_cda_df contains no engine identifiers.")
if len(engine_ids) > 2:
    raise ValueError(f"Expected no more than two engine IDs, found {len(engine_ids)}: {engine_ids}")


def _quoted_col(column_name):
    return F.col(f"`{column_name.replace('`', '``')}`")


def _normalized_raw_state(column_name):
    source = _quoted_col(column_name)
    numeric = source.cast("double")
    return (
        F.when(source.isNull(), F.lit(None).cast("string"))
        .when(numeric.isNotNull() & (numeric == F.floor(numeric)), F.format_string("%.0f", numeric))
        .when(numeric.isNotNull(), F.regexp_replace(F.regexp_replace(F.format_string("%.12f", numeric), "0+$", ""), "[.]$", ""))
        .otherwise(source.cast("string"))
    )


def _boolean_state(column_name):
    source = _quoted_col(column_name)
    numeric = source.cast("double")
    return (
        F.when(source.isNull(), F.lit(None).cast("string"))
        .when(numeric == 0.0, F.lit("OFF"))
        .when(numeric == 1.0, F.lit("ON"))
        .otherwise(_normalized_raw_state(column_name))
    )


def _lvt_state(column_name):
    source = _quoted_col(column_name)
    numeric = source.cast("double")
    return (
        F.when(source.isNull() | numeric.isNull(), F.lit(None).cast("string"))
        .when(numeric.between(0.0, 1.0), F.lit("STOWED [0-1]"))
        .when((numeric > 1.0) & (numeric < 98.0), F.lit("TRANSIT (1-98)"))
        .when(numeric.between(98.0, 100.0), F.lit("DEPLOYED [98-100]"))
        .otherwise(F.concat(F.lit("OUT OF RANGE ["), _normalized_raw_state(column_name), F.lit("]")))
    )


def _state_expression(column_name, event_type):
    if event_type == "lvt":
        return _lvt_state(column_name)
    if event_type == "boolean":
        return _boolean_state(column_name)
    return _normalized_raw_state(column_name)


def _event_frame(label, source_column, event_type):
    state_expression = _state_expression(source_column, event_type)
    states = (
        filtered_cda_df
        .select(
            F.col("Timestamp"),
            F.col("AssetIdentifier").cast("string").alias("EngineNo"),
            state_expression.alias("State"),
        )
        .where(F.col("Timestamp").isNotNull() & F.col("EngineNo").isNotNull())
        .where(F.col("State").isNotNull())
        .dropDuplicates(["Timestamp", "EngineNo", "State"])
    )

    ambiguous = (
        states
        .groupBy("Timestamp", "EngineNo")
        .agg(F.countDistinct("State").alias("StateCount"))
        .where(F.col("StateCount") > 1)
        .limit(1)
        .count()
    )
    if ambiguous:
        raise ValueError(
            f"{label} has more than one distinct state for the same engine and timestamp. "
            "The source ordering is ambiguous, so the timeline was not guessed."
        )

    states = states.dropDuplicates(["Timestamp", "EngineNo"])
    sequence = Window.partitionBy("EngineNo").orderBy(F.col("Timestamp").asc())
    changed = (
        states
        .withColumn("PreviousState", F.lag("State").over(sequence))
        .where(F.col("PreviousState").isNull() | (F.col("State") != F.col("PreviousState")))
        .select(
            "Timestamp",
            "EngineNo",
            F.lit(label).alias("EventName"),
            "State",
            F.concat(F.lit(f"{label} = "), F.col("State")).alias("Event"),
        )
    )
    return changed


event_frames = [_event_frame(*event_spec) for event_spec in available_events]
event_long_df = event_frames[0]
for frame in event_frames[1:]:
    event_long_df = event_long_df.unionByName(frame)

event_long_df = event_long_df.orderBy(
    F.col("Timestamp").asc(),
    F.col("EngineNo").asc(),
    F.col("EventName").asc(),
)

event_rows = event_long_df.collect()

if event_rows:
    event_long_pd = pd.DataFrame(
        [(row.Timestamp, row.EngineNo, row.EventName, row.State, row.Event) for row in event_rows],
        columns=["Timestamp", "EngineNo", "EventName", "State", "Event"],
    )
    event_long_pd["Timestamp"] = pd.to_datetime(event_long_pd["Timestamp"], errors="raise")
    grouped_events = (
        event_long_pd
        .groupby(["Timestamp", "EngineNo"], sort=True, as_index=False)["Event"]
        .agg("\n".join)
    )
    event_df = (
        grouped_events
        .pivot(index="Timestamp", columns="EngineNo", values="Event")
        .reindex(columns=engine_ids)
        .reset_index()
        .sort_values("Timestamp", kind="stable")
        .reset_index(drop=True)
    )
    event_df.columns.name = None
else:
    event_long_pd = pd.DataFrame(columns=["Timestamp", "EngineNo", "EventName", "State", "Event"])
    event_df = pd.DataFrame(columns=["Timestamp", *engine_ids])

print(f"AC event parameters used: {len(available_events)} / {len(EVENT_SOURCES)}")
print(f"Engine IDs: {', '.join(engine_ids)}")
print(f"Event timestamps: {len(event_df):,}")
if missing_events:
    print("Unavailable AC event parameters: " + ", ".join(missing_events))

ipy_display(event_df)
