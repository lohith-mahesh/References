from pyspark.sql import functions as F

CDA_TABLE = (
    "ehm_fleetstore_prd1eun82719_internal."
    "`g700-pearl700`."
    "`emucontinuousscan-cda`"
)

FLIGHT_NO = "87024"

START_TIMESTAMP = "2026-04-06 00:00:00"
END_TIMESTAMP_EXCLUSIVE = "2026-04-16 00:00:00"

FLIGHT_COLUMN = "ParentAssetIdentifier"
ENGINE_COLUMN = "AssetIdentifier"
TIMESTAMP_COLUMN = "Timestamp"

AC_PREFIX = "Parameter:EVHMU_EEC_AC_"
BC_PREFIX = "Parameter:EVHMU_EEC_BC_"

THRUST_REVERSER_KEYWORDS = (
    "ThrustReverser",
    "Thrust_Reverser",
)

cda_source = spark.table(CDA_TABLE)

required_columns = {
    FLIGHT_COLUMN,
    ENGINE_COLUMN,
    TIMESTAMP_COLUMN,
}

missing_required_columns = required_columns.difference(cda_source.columns)

if missing_required_columns:
    raise ValueError(
        "Required CDA column(s) are missing: "
        + ", ".join(sorted(missing_required_columns))
    )

all_columns = cda_source.columns

ac_thrust_reverser_columns = sorted(
    [
        column
        for column in all_columns
        if column.startswith(AC_PREFIX)
        and any(
            keyword.lower() in column.lower()
            for keyword in THRUST_REVERSER_KEYWORDS
        )
    ]
)

bc_thrust_reverser_columns = sorted(
    [
        column
        for column in all_columns
        if column.startswith(BC_PREFIX)
        and any(
            keyword.lower() in column.lower()
            for keyword in THRUST_REVERSER_KEYWORDS
        )
    ]
)

thrust_reverser_columns = (
    ac_thrust_reverser_columns
    + bc_thrust_reverser_columns
)

if not thrust_reverser_columns:
    raise ValueError(
        "No AC or BC thrust-reverser parameter columns were found "
        "in the CDA table."
    )

cda_filtered = (
    cda_source
    .filter(
        (F.col(FLIGHT_COLUMN).cast("string") == F.lit(FLIGHT_NO))
        & (F.col(TIMESTAMP_COLUMN) >= F.to_timestamp(F.lit(START_TIMESTAMP)))
        & (F.col(TIMESTAMP_COLUMN) < F.to_timestamp(F.lit(END_TIMESTAMP_EXCLUSIVE)))
    )
    .select(
        F.col(TIMESTAMP_COLUMN).alias("Timestamp"),
        F.col(FLIGHT_COLUMN).alias("FlightNo"),
        F.col(ENGINE_COLUMN).alias("EngineNo"),
        *[
            F.col(column)
            for column in thrust_reverser_columns
        ],
    )
    .orderBy(
        F.col("Timestamp").asc(),
        F.col("EngineNo").asc(),
    )
)

row_count = cda_filtered.count()

engine_count = (
    cda_filtered
    .select("EngineNo")
    .where(F.col("EngineNo").isNotNull())
    .distinct()
    .count()
)

print("=" * 80)
print("CDA THRUST REVERSER DATA")
print("=" * 80)
print(f"Flight No.        : {FLIGHT_NO}")
print(f"From              : {START_TIMESTAMP}")
print(f"To                : 2026-04-15 23:59:59.999...")
print(f"Rows              : {row_count:,}")
print(f"Engines           : {engine_count}")
print(f"AC parameters     : {len(ac_thrust_reverser_columns)}")
print(f"BC parameters     : {len(bc_thrust_reverser_columns)}")
print(f"Total TR params   : {len(thrust_reverser_columns)}")
print("=" * 80)

if row_count == 0:
    print(
        f"No CDA records were found for FlightNo {FLIGHT_NO} "
        "between 2026-04-06 and 2026-04-15."
    )
else:
    display(cda_filtered)
