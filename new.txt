def _normalize_raw_value(value):
    if pd.isna(value):
        return None
    try:
        numeric = float(value)
        if pd.isna(numeric):
            return None
        if numeric.is_integer():
            return str(int(numeric))
        return f"{numeric:.12f}".rstrip("0").rstrip(".")
    except (TypeError, ValueError, OverflowError):
        return str(value)


def _boolean_value(value):
    if pd.isna(value):
        return None
    try:
        numeric = float(value)
        if numeric == 0.0:
            return "OFF"
        if numeric == 1.0:
            return "ON"
    except (TypeError, ValueError, OverflowError):
        pass
    return _normalize_raw_value(value)


def _lvt_value(value):
    if pd.isna(value):
        return None
    try:
        numeric = float(value)
    except (TypeError, ValueError, OverflowError):
        return None
    if pd.isna(numeric):
        return None
    if 0.0 <= numeric <= 1.0:
        return "STOWED [0-1]"
    if 1.0 < numeric < 98.0:
        return "TRANSIT (1-98)"
    if 98.0 <= numeric <= 100.0:
        return "DEPLOYED [98-100]"
    return f"OUT OF RANGE [{_normalize_raw_value(value)}]"


def _state_series(series, event_type):
    if event_type == "lvt":
        return series.map(_lvt_value)
    if event_type == "boolean":
        return series.map(_boolean_value)
    return series.map(_normalize_raw_value)


def _build_event_timeline(source_pd):
    available_events = []
    missing_events = []
    for label, suffix, event_type in EVENT_SOURCES:
        source_column = f"{AC_PREFIX}{suffix}"
        if source_column in source_pd.columns:
            available_events.append((label, source_column, event_type))
        else:
            missing_events.append(label)

    if not available_events:
        raise ValueError("None of the required AC thrust-reverser event parameters are present in the filtered data.")

    working = source_pd.copy()
    working["Timestamp"] = pd.to_datetime(working["Timestamp"], errors="coerce")
    working["EngineNo"] = working["AssetIdentifier"].astype("string")
    working = working.loc[working["Timestamp"].notna() & working["EngineNo"].notna()].copy()

    if working.empty:
        return pd.DataFrame(columns=["Timestamp", "EngineNo", "EventName", "State", "Event"]), pd.DataFrame(columns=["Timestamp"]), [], missing_events

    engine_ids = sorted(working["EngineNo"].dropna().unique().tolist())
    if len(engine_ids) > 2:
        raise ValueError(f"Expected no more than two engine IDs, found {len(engine_ids)}: {engine_ids}")

    frames = []
    for label, source_column, event_type in available_events:
        states = working[["Timestamp", "EngineNo", source_column]].copy()
        states["State"] = _state_series(states[source_column], event_type)
        states = states.loc[states["State"].notna(), ["Timestamp", "EngineNo", "State"]]
        states = states.drop_duplicates(subset=["Timestamp", "EngineNo", "State"], keep="first")

        state_counts = states.groupby(["Timestamp", "EngineNo"], sort=False)["State"].nunique(dropna=True)
        ambiguous = state_counts[state_counts > 1]
        if not ambiguous.empty:
            timestamp, engine_no = ambiguous.index[0]
            raise ValueError(
                f"{label} has more than one distinct state for engine {engine_no} at {timestamp}. "
                "The source ordering is ambiguous, so the timeline was not guessed."
            )

        states = states.drop_duplicates(subset=["Timestamp", "EngineNo"], keep="first")
        states = states.sort_values(["EngineNo", "Timestamp"], kind="stable").reset_index(drop=True)
        states["PreviousState"] = states.groupby("EngineNo", sort=False)["State"].shift(1)
        changed = states.loc[states["PreviousState"].isna() | states["State"].ne(states["PreviousState"])].copy()
        changed["EventName"] = label
        changed["Event"] = label + " = " + changed["State"].astype(str)
        frames.append(changed[["Timestamp", "EngineNo", "EventName", "State", "Event"]])

    if not frames:
        event_long = pd.DataFrame(columns=["Timestamp", "EngineNo", "EventName", "State", "Event"])
        timeline = pd.DataFrame(columns=["Timestamp", *engine_ids])
        return event_long, timeline, engine_ids, missing_events

    event_long = pd.concat(frames, ignore_index=True)
    event_long = event_long.sort_values(["Timestamp", "EngineNo", "EventName"], kind="stable").reset_index(drop=True)

    grouped = (
        event_long
        .groupby(["Timestamp", "EngineNo"], sort=True, as_index=False)["Event"]
        .agg("\n".join)
    )
    timeline = (
        grouped
        .pivot(index="Timestamp", columns="EngineNo", values="Event")
        .reindex(columns=engine_ids)
        .reset_index()
        .sort_values("Timestamp", kind="stable")
        .reset_index(drop=True)
    )
    timeline.columns.name = None
    return event_long, timeline, engine_ids, missing_events


def _run_pipeline(_):
    global filtered_cda_df, filtered_cda_pd, event_long_pd, event_df
    with filter_output:
        clear_output(wait=True)
        filter_button.disabled = True
        try:
            flight_no = flight_no_widget.value.strip()
            start_date = start_date_widget.value
            end_date = end_date_widget.value

            if not flight_no:
                raise ValueError("Flight No. is required.")
            if start_date is None or end_date is None:
                raise ValueError("Both Start Date and End Date are required.")
            if start_date > end_date:
                raise ValueError("Start Date must not be later than End Date.")

            start_timestamp = datetime.combine(start_date, datetime.min.time())
            end_timestamp_exclusive = datetime.combine(end_date + timedelta(days=1), datetime.min.time())

            filtered_cda_df = (
                cda_catalog
                .where(
                    (F.col("ParentAssetIdentifier").cast("string") == F.lit(flight_no))
                    & (F.col("Timestamp") >= F.lit(start_timestamp))
                    & (F.col("Timestamp") < F.lit(end_timestamp_exclusive))
                )
                .select(*(IDENTIFIER_COLUMNS + available_thrust_reverser_columns))
                .orderBy(F.col("Timestamp").asc(), F.col("AssetIdentifier").cast("string").asc())
            )

            row_count = filtered_cda_df.limit(MAX_PANDAS_ROWS + 1).count()
            if row_count == 0:
                filtered_cda_pd = pd.DataFrame(columns=IDENTIFIER_COLUMNS + available_thrust_reverser_columns)
                event_long_pd = pd.DataFrame(columns=["Timestamp", "EngineNo", "EventName", "State", "Event"])
                event_df = pd.DataFrame(columns=["Timestamp"])
                print(f"No CDA rows found for {flight_no} from {start_date} through {end_date}.")
                return
            if row_count > MAX_PANDAS_ROWS:
                raise MemoryError(
                    f"The filtered selection exceeds {MAX_PANDAS_ROWS:,} rows. "
                    "Choose a smaller date range before converting the CDA data to Pandas."
                )

            filtered_cda_pd = filtered_cda_df.toPandas()
            filtered_cda_pd["Timestamp"] = pd.to_datetime(filtered_cda_pd["Timestamp"], errors="raise")
            filtered_cda_pd = filtered_cda_pd.sort_values(
                ["Timestamp", "AssetIdentifier"], kind="stable"
            ).reset_index(drop=True)

            event_long_pd, event_df, engine_ids, missing_events = _build_event_timeline(filtered_cda_pd)

            print(f"Flight No.: {flight_no}")
            print(f"Date range: {start_date} through {end_date}")
            print(f"Filtered CDA rows converted to Pandas: {len(filtered_cda_pd):,}")
            print(f"Thrust-reverser columns returned: {len(available_thrust_reverser_columns)} / {len(THRUST_REVERSER_COLUMNS)}")
            print(f"Engine IDs: {', '.join(engine_ids) if engine_ids else 'None'}")
            print(f"Event timestamps: {len(event_df):,}")
            if missing_thrust_reverser_columns:
                print(f"Requested thrust-reverser columns absent from the CDA schema: {len(missing_thrust_reverser_columns)}")
            if missing_events:
                print("Unavailable AC event parameters: " + ", ".join(missing_events))

            ipy_display(event_df)
        except Exception as exc:
            filtered_cda_df = None
            filtered_cda_pd = None
            event_long_pd = None
            event_df = None
            print(f"ERROR: {type(exc).__name__}: {exc}")
            raise
        finally:
            filter_button.disabled = False


filter_button.on_click(_run_pipeline)
ipy_display(
    widgets.VBox(
        [
            flight_no_widget,
            widgets.HBox([start_date_widget, end_date_widget]),
            filter_button,
            filter_output,
        ]
    )
)
