from datetime import datetime

import ipywidgets as widgets
import pandas as pd
from IPython.display import clear_output, display as ipy_display
from pyspark.sql import functions as F
from pyspark.sql.window import Window

SCAN_TABLE = "ehm_fleetstore_prd1eun82719_internal.`g700-pearl700`.`emucontinuousscan-cda`"
PARAMETER_PREFIX = "Parameter:EVHMU_EEC_AC_"

EVENT_PARAMETERS = {
    "TR Armed": "IThrustReverser_trArmed_data",
    "DoorDep Status": "IThrustReverser_trDoorDeplStatus_data",
    "Door Stow Status": "IThrustReverser_trDoorStowStatus_data",
    "In Transit": "IThrustReverser_trInTransit_data",
    "IsDeployed": "IThrustReverser_trIsDeployed_data",
    "Is Unlocked": "IThrustReverser_trIsUnlocked_data",
    "LVTPosVCntrl": "IThrustReverser_trLVTPosVCtrl_data",
    "Low TRLVT": "IThrustReverser_loOwnTRLVTRaw_data",
    "Up TRLVT": "IThrustReverser_upOwnTRLVTRaw_data",
}

LVT_EVENTS = {"LVTPosVCntrl", "Low TRLVT", "Up TRLVT"}
BOOLEAN_STYLE = {
    "TR Armed": ("0", "1"),
    "In Transit": ("OFF", "ON"),
    "IsDeployed": ("0", "1"),
    "Is Unlocked": ("OFF", "ON"),
}

scan_schema = spark.table(SCAN_TABLE).schema
scan_columns = {field.name for field in scan_schema.fields}
required_identifier_columns = {"Timestamp", "ParentAssetIdentifier", "AssetIdentifier"}
missing_identifiers = sorted(required_identifier_columns - scan_columns)
if missing_identifiers:
    raise ValueError(f"Scan table is missing required columns: {missing_identifiers}")

available_event_parameters = {
    label: f"{PARAMETER_PREFIX}{suffix}"
    for label, suffix in EVENT_PARAMETERS.items()
    if f"{PARAMETER_PREFIX}{suffix}" in scan_columns
}
missing_event_parameters = [
    label for label in EVENT_PARAMETERS if label not in available_event_parameters
]
if not available_event_parameters:
    raise ValueError("None of the required AC thrust-reverser event parameters exist in the scan schema.")

existing_aircraft = globals().get("tr_aircraft_widget")
existing_start = globals().get("tr_start_widget")
existing_end = globals().get("tr_end_widget")
previous_aircraft = getattr(existing_aircraft, "value", "") or ""
previous_start = getattr(existing_start, "value", "") or ""
previous_end = getattr(existing_end, "value", "") or ""

tr_aircraft_widget = widgets.Text(
    value=previous_aircraft,
    description="Flight No. / Aircraft:",
    placeholder="ParentAssetIdentifier",
    layout=widgets.Layout(width="520px"),
    style={"description_width": "150px"},
)
tr_start_widget = widgets.Text(
    value=previous_start,
    description="Start time:",
    placeholder="YYYY-MM-DD HH:MM:SS.sss",
    layout=widgets.Layout(width="520px"),
    style={"description_width": "120px"},
)
tr_end_widget = widgets.Text(
    value=previous_end,
    description="End time:",
    placeholder="YYYY-MM-DD HH:MM:SS.sss",
    layout=widgets.Layout(width="520px"),
    style={"description_width": "120px"},
)
tr_load_button = widgets.Button(
    description="Build event timeline",
    button_style="primary",
    icon="table",
    layout=widgets.Layout(width="220px"),
)
tr_output = widgets.Output()


def _parse_timestamp(text, field_name):
    value = str(text).strip()
    if not value:
        raise ValueError(f"{field_name} is required.")
    parsed = pd.to_datetime(value, errors="raise")
    if getattr(parsed, "tzinfo", None) is not None:
        raise ValueError(f"{field_name} must be timezone-naive to match CDA timestamps.")
    return parsed.to_pydatetime()


def _spark_col(name):
    return F.col(f"`{name.replace('`', '``')}`")


def _lvt_state(column_name):
    value = _spark_col(column_name).cast("double")
    return (
        F.when(value.between(0.0, 1.0), F.lit("STOWED [0-1]"))
        .when((value > 1.0) & (value < 98.0), F.lit("TRANSIT (1-98)"))
        .when(value.between(98.0, 100.0), F.lit("DEPLOYED [98-100]"))
    )


def _plain_state(column_name):
    value = _spark_col(column_name)
    return F.when(value.isNotNull(), value.cast("string"))


def _format_boolean_event(label, state_column):
    off_text, on_text = BOOLEAN_STYLE[label]
    numeric = state_column.cast("double")
    return (
        F.when(numeric == 0.0, F.lit(f"{label} = {off_text}"))
        .when(numeric == 1.0, F.lit(f"{label} = {on_text}"))
        .otherwise(F.concat(F.lit(f"{label} = "), state_column))
    )


def _event_text(label, state_column):
    if label in LVT_EVENTS:
        return F.concat(F.lit(f"{label} = "), state_column)
    if label in BOOLEAN_STYLE:
        return _format_boolean_event(label, state_column)
    return F.concat(F.lit(f"{label} = "), state_column)


def _build_event_timeline(_):
    tr_load_button.disabled = True
    try:
        with tr_output:
            clear_output(wait=True)
            aircraft_id = tr_aircraft_widget.value.strip()
            if not aircraft_id:
                raise ValueError("Aircraft ID is required.")

            start_ts = _parse_timestamp(tr_start_widget.value, "Start time")
            end_ts = _parse_timestamp(tr_end_widget.value, "End time")
            if start_ts > end_ts:
                raise ValueError("Start time must not be later than end time.")

            selected_columns = [
                "Timestamp",
                "ParentAssetIdentifier",
                "AssetIdentifier",
                *available_event_parameters.values(),
            ]
            filtered_cda_df = (
                spark.table(SCAN_TABLE)
                .where(
                    (F.col("ParentAssetIdentifier").cast("string") == aircraft_id)
                    & F.col("Timestamp").between(F.lit(start_ts), F.lit(end_ts))
                )
                .select(*selected_columns)
            )

            engine_ids = [
                row.AssetIdentifier
                for row in (
                    filtered_cda_df
                    .select(F.col("AssetIdentifier").cast("string").alias("AssetIdentifier"))
                    .where(F.col("AssetIdentifier").isNotNull())
                    .distinct()
                    .orderBy("AssetIdentifier")
                    .collect()
                )
            ]
            if not engine_ids:
                raise ValueError("No CDA samples were found for the selected aircraft and time interval.")
            if len(engine_ids) > 2:
                raise ValueError(
                    f"Expected at most two engine IDs in the selected interval, found {len(engine_ids)}: {engine_ids}"
                )

            event_frames = []
            for label, source_column in available_event_parameters.items():
                state_expr = _lvt_state(source_column) if label in LVT_EVENTS else _plain_state(source_column)
                state_df = (
                    filtered_cda_df
                    .where(F.col("AssetIdentifier").isNotNull())
                    .where(_spark_col(source_column).isNotNull())
                    .where(state_expr.isNotNull())
                    .select(
                        "Timestamp",
                        F.col("AssetIdentifier").cast("string").alias("EngineNo"),
                        state_expr.alias("State"),
                    )
                    .dropDuplicates(["Timestamp", "EngineNo", "State"])
                )

                order_window = Window.partitionBy("EngineNo").orderBy(F.col("Timestamp"))
                previous_non_null_state = F.last("State", ignorenulls=True).over(
                    order_window.rowsBetween(Window.unboundedPreceding, -1)
                )
                changed_df = (
                    state_df
                    .withColumn("PreviousState", previous_non_null_state)
                    .where(F.col("PreviousState").isNull() | (F.col("State") != F.col("PreviousState")))
                    .select(
                        "Timestamp",
                        "EngineNo",
                        _event_text(label, F.col("State")).alias("Event"),
                    )
                )
                event_frames.append(changed_df)

            if not event_frames:
                raise ValueError("No event streams could be constructed from the available AC parameters.")

            event_long_df = event_frames[0]
            for frame in event_frames[1:]:
                event_long_df = event_long_df.unionByName(frame)

            event_long_df = event_long_df.orderBy("Timestamp", "EngineNo", "Event")
            event_rows = event_long_df.collect()

            if event_rows:
                event_long_pd = pd.DataFrame(
                    [(row.Timestamp, row.EngineNo, row.Event) for row in event_rows],
                    columns=["Timestamp", "EngineNo", "Event"],
                )
                event_long_pd["Timestamp"] = pd.to_datetime(event_long_pd["Timestamp"])
                grouped = (
                    event_long_pd
                    .groupby(["Timestamp", "EngineNo"], sort=True, as_index=False)["Event"]
                    .agg("\n".join)
                )
                event_df = (
                    grouped
                    .pivot(index="Timestamp", columns="EngineNo", values="Event")
                    .reindex(columns=engine_ids)
                    .reset_index()
                    .sort_values("Timestamp", kind="stable")
                    .reset_index(drop=True)
                )
                event_df.columns.name = None
            else:
                event_df = pd.DataFrame(columns=["Timestamp", *engine_ids])

            globals().update(
                {
                    "filtered_cda_df": filtered_cda_df,
                    "event_long_df": event_long_df,
                    "event_df": event_df,
                }
            )

            print(f"Aircraft ID: {aircraft_id}")
            print(f"Interval: {start_ts} through {end_ts}")
            print(f"Engine IDs: {', '.join(engine_ids)}")
            print(f"Event rows: {len(event_df):,}")
            if missing_event_parameters:
                print("Unavailable AC event parameters: " + ", ".join(missing_event_parameters))
            ipy_display(event_df)
    except Exception as exc:
        with tr_output:
            clear_output(wait=True)
            print(f"Unable to build the thrust-reverser event timeline: {exc}")
    finally:
        tr_load_button.disabled = False


tr_load_button.on_click(_build_event_timeline)
ipy_display(
    widgets.VBox(
        [
            widgets.HTML("<b>AC Thrust-Reverser Event Timeline</b>"),
            tr_aircraft_widget,
            tr_start_widget,
            tr_end_widget,
            tr_load_button,
            tr_output,
        ]
    )
)
