from datetime import datetime

import numpy as np
import pandas as pd
import ipywidgets as widgets
from IPython.display import clear_output, display
from pyspark.sql import functions as F

SCAN_TABLE = "ehm_fleetstore_prd1eun82719_internal.`g700-pearl700`.`emucontinuousscan-cda`"
PARAMETER_PREFIX = "Parameter:EVHMU_EEC_AC_"

EVENT_PARAMETERS = {
    "TR Armed": "IThrustReverser_trArmed_data",
    "DoorDep Status": "IThrustReverser_trDoorDeplStatus_data",
    "Door Stow Status": "IThrustReverser_trDoorStowStatus_data",
    "In Transit": "IThrustReverser_trInTransit_data",
    "IsDeployed": "IThrustReverser_trIsDeployed_data",
    "Is Unlocked": "IThrustReverser_trIsUnlocked_data",
    "LVTPosVCntrl": "IThrustReverser_trLVTPosVCtrl_data",
    "Low TRLVT": "IThrustReverser_loOwnTRLVTRaw_data",
    "Up TRLVT": "IThrustReverser_upOwnTRLVTRaw_data",
}

POSITION_EVENTS = {"LVTPosVCntrl", "Low TRLVT", "Up TRLVT"}
ON_OFF_EVENTS = {"TR Armed", "In Transit", "Is Unlocked"}
ZERO_ONE_EVENTS = {"IsDeployed"}
REQUIRED_BASE_COLUMNS = {"Timestamp", "ParentAssetIdentifier", "AssetIdentifier"}


def _parse_timestamp(value, field_name):
    value = value.strip()
    if not value:
        raise ValueError(f"{field_name} is required.")
    try:
        parsed = pd.Timestamp(value)
    except Exception as exc:
        raise ValueError(
            f"{field_name} must be a valid timestamp, for example 2026-09-27 20:03:55.600."
        ) from exc
    if pd.isna(parsed):
        raise ValueError(f"{field_name} is not a valid timestamp.")
    if parsed.tzinfo is not None:
        raise ValueError(f"{field_name} must not contain a timezone offset.")
    return parsed


def _position_state(value):
    try:
        numeric = float(value)
    except (TypeError, ValueError):
        return None
    if not np.isfinite(numeric):
        return None
    if 0.0 <= numeric <= 1.0:
        return "STOWED [0–1]"
    if 1.0 < numeric < 98.0:
        return "TRANSIT (1–98)"
    if 98.0 <= numeric <= 100.0:
        return "DEPLOYED [98–100]"
    return f"OUT OF RANGE [{numeric:g}]"


def _normalized_scalar(value):
    if pd.isna(value):
        return None
    if isinstance(value, (np.bool_, bool)):
        return int(value)
    if isinstance(value, (np.integer, int)):
        return int(value)
    if isinstance(value, (np.floating, float)):
        numeric = float(value)
        if not np.isfinite(numeric):
            return None
        if numeric.is_integer():
            return int(numeric)
        return numeric
    return str(value).strip()


def _binary_value(value, true_text, false_text):
    normalized = _normalized_scalar(value)
    if normalized is None:
        return None
    if normalized in (1, "1", "1.0", "True", "true", "TRUE"):
        return true_text
    if normalized in (0, "0", "0.0", "False", "false", "FALSE"):
        return false_text
    return str(normalized)


def _event_text(event_name, value):
    if event_name in POSITION_EVENTS:
        state = _position_state(value)
        return None if state is None else f"{event_name} = {state}"
    if event_name in ON_OFF_EVENTS:
        state = _binary_value(value, "ON", "OFF")
        return None if state is None else f"{event_name} = {state}"
    if event_name in ZERO_ONE_EVENTS:
        state = _binary_value(value, "1", "0")
        return None if state is None else f"{event_name} = {state}"
    normalized = _normalized_scalar(value)
    return None if normalized is None else f"{event_name} = {normalized}"


def _change_key(event_name, value):
    if event_name in POSITION_EVENTS:
        return _position_state(value)
    if event_name in ON_OFF_EVENTS or event_name in ZERO_ONE_EVENTS:
        return _normalized_scalar(value)
    return _normalized_scalar(value)


def _extract_events(source_pdf, engine_ids, available_parameters):
    records = []
    for engine_id in engine_ids:
        engine_pdf = source_pdf.loc[source_pdf["AssetIdentifier"].astype(str) == engine_id]
        engine_pdf = engine_pdf.sort_values("Timestamp", kind="stable")
        for event_name, full_column in available_parameters.items():
            previous_key = object()
            has_previous = False
            for timestamp, value in engine_pdf[["Timestamp", full_column]].itertuples(index=False, name=None):
                if pd.isna(timestamp) or pd.isna(value):
                    continue
                key = _change_key(event_name, value)
                text = _event_text(event_name, value)
                if key is None or text is None:
                    continue
                if not has_previous or key != previous_key:
                    records.append(
                        {
                            "Timestamp": timestamp,
                            "EngineNo": engine_id,
                            "Event": text,
                            "_ParameterOrder": list(EVENT_PARAMETERS).index(event_name),
                        }
                    )
                    previous_key = key
                    has_previous = True

    if not records:
        return pd.DataFrame(columns=["Timestamp", *engine_ids])

    long_df = pd.DataFrame.from_records(records)
    long_df = long_df.sort_values(
        ["Timestamp", "_ParameterOrder", "EngineNo"], kind="stable"
    ).reset_index(drop=True)
    long_df["_EventSlot"] = long_df.groupby(
        ["Timestamp", "EngineNo"], sort=False
    ).cumcount()

    wide_df = long_df.pivot(
        index=["Timestamp", "_EventSlot"], columns="EngineNo", values="Event"
    ).reset_index()
    wide_df.columns.name = None
    for engine_id in engine_ids:
        if engine_id not in wide_df.columns:
            wide_df[engine_id] = pd.NA
    wide_df = wide_df[["Timestamp", *engine_ids]]
    wide_df = wide_df.sort_values("Timestamp", kind="stable").reset_index(drop=True)
    return wide_df


def _month_pairs(start_ts, end_ts):
    periods = pd.period_range(start=start_ts.to_period("M"), end=end_ts.to_period("M"), freq="M")
    return [(int(period.year), int(period.month)) for period in periods]


def _build_timeline(_=None):
    run_button.disabled = True
    try:
        with output:
            clear_output(wait=True)
            aircraft_id = aircraft_input.value.strip()
            if not aircraft_id:
                raise ValueError("Flight No. / Aircraft ID is required.")

            start_ts = _parse_timestamp(start_input.value, "Start timestamp")
            end_ts = _parse_timestamp(end_input.value, "End timestamp")
            if start_ts > end_ts:
                raise ValueError("Start timestamp must not be later than end timestamp.")

            scan_df = spark.table(SCAN_TABLE)
            schema_columns = set(scan_df.columns)
            missing_base = sorted(REQUIRED_BASE_COLUMNS - schema_columns)
            if missing_base:
                raise ValueError(f"The CDA scan table is missing required columns: {missing_base}")

            full_parameters = {
                event_name: f"{PARAMETER_PREFIX}{suffix}"
                for event_name, suffix in EVENT_PARAMETERS.items()
            }
            missing_parameters = [
                column for column in full_parameters.values() if column not in schema_columns
            ]
            if missing_parameters:
                raise ValueError(
                    "The CDA scan table is missing required AC thrust-reverser parameters: "
                    + ", ".join(missing_parameters)
                )

            filter_condition = (
                (F.col("ParentAssetIdentifier").cast("string") == aircraft_id)
                & F.col("Timestamp").between(start_ts.to_pydatetime(), end_ts.to_pydatetime())
            )

            if "year" in schema_columns and "Month" in schema_columns:
                partition_condition = F.lit(False)
                for year_value, month_value in _month_pairs(start_ts, end_ts):
                    partition_condition = partition_condition | (
                        (F.col("year") == year_value) & (F.col("Month") == month_value)
                    )
                filter_condition = filter_condition & partition_condition

            selected_columns = [
                "Timestamp",
                "ParentAssetIdentifier",
                "AssetIdentifier",
                *full_parameters.values(),
            ]
            filtered_spark_df = (
                scan_df.where(filter_condition)
                .select(*selected_columns)
                .orderBy(F.col("Timestamp"), F.col("AssetIdentifier"))
            )

            engine_ids = [
                row["AssetIdentifier"]
                for row in filtered_spark_df.select(
                    F.col("AssetIdentifier").cast("string").alias("AssetIdentifier")
                )
                .where(F.col("AssetIdentifier").isNotNull())
                .distinct()
                .orderBy("AssetIdentifier")
                .collect()
            ]
            if not engine_ids:
                raise ValueError("No CDA records were found for the requested aircraft and time interval.")
            if len(engine_ids) != 2:
                raise ValueError(
                    f"Expected exactly two engine identifiers in the requested interval; found {len(engine_ids)}: {engine_ids}"
                )

            source_pdf = filtered_spark_df.toPandas()
            if source_pdf.empty:
                raise ValueError("No CDA records were found after filtering.")
            source_pdf["Timestamp"] = pd.to_datetime(source_pdf["Timestamp"], errors="coerce")
            source_pdf["AssetIdentifier"] = source_pdf["AssetIdentifier"].astype("string")
            source_pdf = source_pdf.loc[source_pdf["Timestamp"].notna()].copy()
            if source_pdf.empty:
                raise ValueError("All filtered CDA timestamps are invalid or null.")

            event_df = _extract_events(source_pdf, engine_ids, full_parameters)
            filtered_cda_df = source_pdf
            globals()["filtered_cda_df"] = filtered_cda_df
            globals()["event_df"] = event_df

            print(f"Aircraft ID: {aircraft_id}")
            print(f"Interval: {start_ts} through {end_ts}")
            print(f"Engines: {engine_ids[0]}, {engine_ids[1]}")
            print(f"Filtered CDA rows: {len(filtered_cda_df):,}")
            print(f"Event rows: {len(event_df):,}")
            display(event_df)
    except Exception as exc:
        print(f"Unable to build the thrust-reverser event timeline: {exc}")
        raise
    finally:
        run_button.disabled = False


existing_aircraft_input = globals().get("aircraft_input")
existing_start_input = globals().get("start_input")
existing_end_input = globals().get("end_input")

previous_aircraft = getattr(existing_aircraft_input, "value", "")
previous_start = getattr(existing_start_input, "value", "")
previous_end = getattr(existing_end_input, "value", "")

aircraft_input = widgets.Text(
    value=previous_aircraft,
    description="Flight No.:",
    placeholder="Aircraft ID",
    layout=widgets.Layout(width="420px"),
)
start_input = widgets.Text(
    value=previous_start,
    description="Start:",
    placeholder="YYYY-MM-DD HH:MM:SS.sss",
    layout=widgets.Layout(width="420px"),
)
end_input = widgets.Text(
    value=previous_end,
    description="End:",
    placeholder="YYYY-MM-DD HH:MM:SS.sss",
    layout=widgets.Layout(width="420px"),
)
run_button = widgets.Button(
    description="Build event timeline",
    button_style="primary",
    layout=widgets.Layout(width="220px"),
)
output = widgets.Output()
run_button.on_click(_build_timeline)

display(
    widgets.VBox(
        [
            widgets.HBox([aircraft_input, start_input, end_input]),
            run_button,
            output,
        ]
    )
)
