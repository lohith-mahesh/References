flight_no_widget = widgets.Text(
    description="Flight No.:",

)

start_date_widget = widgets.DatePicker(
    description="Start Date:",
    layout=widgets.Layout(width="300px"),
)

end_date_widget = widgets.DatePicker(
    description="End Date:",
    layout=widgets.Layout(width="300px"),
)

filter_button = widgets.Button(
    description="Filter Data",
    button_style="primary",
    layout=widgets.Layout(width="150px"),
)

output = widgets.Output()

def filter_cda(_):
    global filtered_df

    with output:
        clear_output(wait=True)

        flight_no = flight_no_widget.value.strip()
        start_date = start_date_widget.value
        end_date = end_date_widget.value

        if not flight_no:
            raise ValueError("Flight No. is required.")

        if start_date is None or end_date is None:
            raise ValueError("Both Start Date and End Date are required.")

        if start_date > end_date:
            raise ValueError("Start Date must not be later than End Date.")

        start_datetime = datetime.combine(start_date, datetime.min.time())
        end_datetime_exclusive = datetime.combine(
            end_date + timedelta(days=1),
            datetime.min.time(),
        )

        cda = spark.table(CDA_TABLE)
        cda_columns = set(cda.columns)

        required_columns = [
            "ParentAssetIdentifier",
            "AssetIdentifier",
            "Timestamp",
        ]

        missing_required_columns = [
            column for column in required_columns
            if column not in cda_columns
        ]

        if missing_required_columns:
            raise ValueError(
                f"CDA table is missing required columns: {missing_required_columns}"
            )

        thrust_reverser_columns = [
            f"{PARAMETER_PREFIX}{channel}_{suffix}"
            for channel in CHANNELS
            for suffix in THRUST_REVERSER_SUFFIXES
        ]

        available_thrust_reverser_columns = [
            column for column in thrust_reverser_columns
            if column in cda_columns
        ]

        missing_thrust_reverser_columns = [
            column for column in thrust_reverser_columns
            if column not in cda_columns
        ]

        if not available_thrust_reverser_columns:
            raise ValueError(
                "None of the requested AC/BC Thrust Reverser parameters exist in the CDA table."
            )

        filtered_df = (
            cda
            .where(
                (F.col("ParentAssetIdentifier").cast("string") == flight_no)
                & (F.col("Timestamp") >= F.lit(start_datetime))
                & (F.col("Timestamp") < F.lit(end_datetime_exclusive))
            )
            .select(
                "ParentAssetIdentifier",
                "AssetIdentifier",
                "Timestamp",
                *available_thrust_reverser_columns,
            )
            .orderBy(
                F.col("Timestamp"),
                F.col("AssetIdentifier"),
            )
        )

        print(f"Flight No.: {flight_no}")
        print(f"Date range: {start_date} through {end_date}")
        print(f"Thrust Reverser parameters selected: {len(available_thrust_reverser_columns)} / {len(thrust_reverser_columns)}")

        if missing_thrust_reverser_columns:
            print(f"Thrust Reverser parameters unavailable in schema: {len(missing_thrust_reverser_columns)}")

        display(filtered_df)

filter_button.on_click(filter_cda)
