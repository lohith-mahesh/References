def _normalized_transition_expression(column_name, mode):
    raw_column = F.col(f"`{column_name}`")
    raw_text = F.trim(raw_column.cast("string"))
    folded_text = F.lower(raw_text)
    numeric_value = F.expr(f"try_cast(`{column_name}` as double)")
    numeric_nonfinite = F.isnan(numeric_value) | numeric_value.isin(float("inf"), float("-inf"))
    if mode == "endpoint":
        return (
            F.when(
                raw_column.isNull() | numeric_value.isNull() | numeric_nonfinite,
                F.lit(None).cast("string"),
            )
            .when(numeric_value <= F.lit(TRU_ENDPOINT_LOW_MAX), F.lit("0"))
            .when(numeric_value >= F.lit(TRU_ENDPOINT_HIGH_MIN), F.lit("100"))
            .otherwise(F.lit(None).cast("string"))
        )
    return (
        F.when(raw_column.isNull() | (raw_text == ""), F.lit(None).cast("string"))
        .when(folded_text.isin("true", "on", "yes"), F.lit("1"))
        .when(folded_text.isin("false", "off", "no"), F.lit("0"))
        .when(numeric_nonfinite, F.lit(None).cast("string"))
        .when(numeric_value == F.lit(1.0), F.lit("1"))
        .when(numeric_value == F.lit(0.0), F.lit("0"))
        .otherwise(folded_text)
    )


def _millisecond_timestamp(value):
    timestamp = pd.Timestamp(value)
    return timestamp.strftime("%Y-%m-%d %H:%M:%S.%f")[:-3]


def _raw_value_list(value):
    if value is None:
        return []
    if isinstance(value, (list, tuple, set)):
        return [str(item) for item in value]
    if hasattr(value, "tolist") and not isinstance(value, str):
        converted = value.tolist()
        if isinstance(converted, list):
            return [str(item) for item in converted]
    try:
        if pd.isna(value):
            return []
    except (TypeError, ValueError):
        pass
    return [str(value)]


def _render_coverage_table(coverage_records):

        "ESN",

        "Conflicts",

def _event_badge(event):
    parameter = str(event["Parameter"])
    mode = str(event["Mode"])
    event_type = str(event["EventType"])
    state = "" if event.get("State") is None else str(event.get("State"))
    raw_values = _raw_value_list(event.get("RawValues"))
    raw_label = " / ".join(raw_values) if raw_values else "no value"
    exact_time = pd.Timestamp(event["Timestamp"]).isoformat()
    if event_type == "conflict":
        label = f"{parameter}: conflict [{raw_label}]"
        value_class = "value-conflict"
    elif mode == "endpoint":
        label = f"{parameter}={raw_label} (≈{state})"
        value_class = "endpoint-low" if state == "0" else "endpoint-high"
    else:
        label = f"{parameter}={raw_label}"
        if state == "1":
            value_class = "state-one"
        elif state == "0":
            value_class = "state-zero"
        else:
            value_class = "state-categorical"
    initial_class = " event-initial" if event_type == "initialization" else ""
    initial_tag = "<span class='event-tag'>initial</span>" if event_type == "initialization" else ""
    return (
        f"<div class='event-badge {value_class}{initial_class}' title='{html.escape(exact_time)}'>"
        f"{html.escape(label)}{initial_tag}</div>"
    )


def _render_transition_table(event_rows, esns, positions):
    columns = [(str(esn), channel) for esn in esns for channel in CHANNELS]
    rows_by_millisecond = OrderedDict()
    for event in event_rows:
        timestamp = pd.Timestamp(event["Timestamp"])
        millisecond = timestamp.floor("ms")
        row = rows_by_millisecond.setdefault(millisecond, {column: [] for column in columns})
        key = (str(event["ESN"]), str(event["Channel"]))
        if key in row:
            row[key].append(event)
    header_cells = ["<th class='timestamp-header'>Timestamp</th>"]
    for esn, channel in columns:
        position = positions.get(esn) or ""
        position_label = f" · {html.escape(str(position))}" if position else ""
        channel_class = "channel-ac" if channel == "AC" else "channel-bc"
        header_cells.append(
            f"<th class='{channel_class}'>ESN {html.escape(esn)}{position_label}<br>{html.escape(channel)}</th>"
        )
    body_rows = []
    for millisecond, values in rows_by_millisecond.items():
        cells = [
            f"<td class='transition-time' title='{html.escape(millisecond.isoformat())}'>"
            f"{html.escape(_millisecond_timestamp(millisecond))}</td>"
        ]
        for column in columns:
            badges = "".join(_event_badge(event) for event in values[column])
            cells.append(f"<td class='event-cell'>{badges}</td>")
        body_rows.append("<tr>" + "".join(cells) + "</tr>")
    return """

def _build_transition_comparison(chosen_day, raise_errors=False):
    try:
        print("Building the TRU transition table...")
        chosen_aircraft = selected_acid
        if chosen_day is None:
            raise ValueError("A comparison day is required.")
        day_start, day_end = _calendar_bounds(chosen_day, chosen_day)
        master_df, windows_df, esns, positions, aircraft_ids = _resolve_engine_windows(
            chosen_aircraft, day_start, day_end
        )
        esns = [str(esn) for esn in esns]
        positions = {str(esn): position for esn, position in positions.items()}
        if not esns:
            print(f"WARNING: No engine master records overlap {chosen_day} for aircraft {chosen_aircraft}.")
            globals().update(
                {
                    "transition_event_df": pd.DataFrame(),
                    "transition_coverage_df": pd.DataFrame(),
                    "transition_esns": [],
                }
            )
            return
        scan_source = spark.table(SCAN_TABLE)
        schema_types = {
            field.name: field.dataType.simpleString() for field in scan_source.schema.fields
        }
        available_columns = {}
        for parameter, channel_columns in TRU_TRANSITION_COLUMNS.items():
            for channel, column_name in channel_columns.items():
                if column_name in schema_types:
                    available_columns[(parameter, channel)] = column_name
                else:
                    print(
                        f"WARNING: {channel} {parameter} is absent from the scan schema "
                        f"({column_name})."
                    )
        comparison_scan, _ = _windowed_scan(
            windows_df,
            aircraft_ids,
            esns,
            requested_columns=list(available_columns.values()),
            strict_start=day_start,
            strict_end=day_end,
        )
        coverage_expressions = [F.count(F.lit(1)).alias("_Rows")]
        coverage_aliases = {}
        for parameter_index, (parameter, specification) in enumerate(
            TRU_TRANSITION_PARAMETERS.items()
        ):
            mode = specification["mode"]
            for channel in CHANNELS:
                column_name = available_columns.get((parameter, channel))
                if column_name is None:
                    continue
                alias_prefix = f"_P{parameter_index}_{channel}"
                coverage_aliases[(parameter, channel)] = alias_prefix
                raw_column = F.col(f"`{column_name}`")
                numeric_value = F.expr(f"try_cast(`{column_name}` as double)")
                numeric_nonfinite = F.isnan(numeric_value) | numeric_value.isin(
                    float("inf"), float("-inf")
                )
                normalized_value = _normalized_transition_expression(column_name, mode)
                ignored_condition = (
                    raw_column.isNotNull()
                    & numeric_value.isNotNull()
                    & ~numeric_nonfinite
                    & (numeric_value > F.lit(TRU_ENDPOINT_LOW_MAX))
                    & (numeric_value < F.lit(TRU_ENDPOINT_HIGH_MIN))
                    if mode == "endpoint"
                    else F.lit(False)
                )
                invalid_condition = (
                    raw_column.isNotNull() & (numeric_value.isNull() | numeric_nonfinite)
                    if mode == "endpoint"
                    else raw_column.isNotNull() & normalized_value.isNull()
                )
                coverage_expressions.extend(
                    [
                        F.count(raw_column).alias(f"{alias_prefix}_NonNull"),
                        F.sum(F.when(normalized_value.isNotNull(), 1).otherwise(0)).alias(
                            f"{alias_prefix}_Accepted"
                        ),
                        F.sum(F.when(ignored_condition, 1).otherwise(0)).alias(
                            f"{alias_prefix}_Ignored"
                        ),
                        F.sum(F.when(invalid_condition, 1).otherwise(0)).alias(
                            f"{alias_prefix}_Invalid"
                        ),
                        F.min(F.when(raw_column.isNotNull(), F.col("Timestamp"))).alias(
                            f"{alias_prefix}_FirstSample"
                        ),
                        F.max(F.when(raw_column.isNotNull(), F.col("Timestamp"))).alias(
                            f"{alias_prefix}_LastSample"
                        ),
                    ]
                )
        coverage_rows = (
            comparison_scan.groupBy(F.col("AssetIdentifier").cast("string").alias("ESN"))
            .agg(*coverage_expressions)
            .collect()
        )
        total_rows = sum(int(row["_Rows"] or 0) for row in coverage_rows)
        print(f"Aircraft {chosen_aircraft} | Day {chosen_day} | ESNs: {', '.join(esns)}")
        print(f"Strict calendar-day scan rows: {total_rows:,}")
        coverage_lookup = {}
        for row in coverage_rows:
            esn = str(row["ESN"])
            for parameter, channel in available_columns:
                alias_prefix = coverage_aliases[(parameter, channel)]
                first_sample = row[f"{alias_prefix}_FirstSample"]
                last_sample = row[f"{alias_prefix}_LastSample"]
                coverage_lookup[(esn, parameter, channel)] = {
                    "Rows": int(row["_Rows"] or 0),
                    "Non-null": int(row[f"{alias_prefix}_NonNull"] or 0),
                    "Accepted": int(row[f"{alias_prefix}_Accepted"] or 0),
                    "Ignored": int(row[f"{alias_prefix}_Ignored"] or 0),
                    "Invalid": int(row[f"{alias_prefix}_Invalid"] or 0),
                    "First sample": _millisecond_timestamp(first_sample) if first_sample else "",
                    "Last sample": _millisecond_timestamp(last_sample) if last_sample else "",
                }
        event_frames = []
        for parameter_index, (parameter, specification) in enumerate(
            TRU_TRANSITION_PARAMETERS.items()
        ):
            mode = specification["mode"]
            for channel in CHANNELS:
                column_name = available_columns.get((parameter, channel))
                if column_name is None:
                    continue
                raw_column = F.col(f"`{column_name}`")
                normalized_value = _normalized_transition_expression(column_name, mode)
                parameter_rows = comparison_scan.select(
                    F.col("Timestamp"),
                    F.col("StartDatetime"),
                    F.col("AssetIdentifier").cast("string").alias("ESN"),
                    raw_column.cast("string").alias("RawValue"),
                    normalized_value.alias("State"),
                ).where(normalized_value.isNotNull())
                grouped = parameter_rows.groupBy("Timestamp", "StartDatetime", "ESN").agg(
                    F.sort_array(F.collect_set("State")).alias("States"),
                    F.sort_array(F.collect_set("RawValue")).alias("RawValues"),
                )
                valid_unique = grouped.where(F.size("States") == 1).select(
                    "Timestamp",
                    "StartDatetime",
                    "ESN",
                    F.element_at("States", 1).cast("string").alias("State"),
                    "RawValues",
                )
                sequence_window = Window.partitionBy("ESN", "StartDatetime").orderBy("Timestamp")
                sequenced = valid_unique.withColumn(
                    "PreviousState", F.lag("State").over(sequence_window)
                )
                events = sequenced.where(
                    F.col("PreviousState").isNull()
                    | (F.col("State") != F.col("PreviousState"))
                ).select(
                    "Timestamp",
                    "StartDatetime",
                    "ESN",
                    F.lit(channel).alias("Channel"),
                    F.lit(parameter).alias("Parameter"),
                    F.lit(parameter_index).cast("integer").alias("ParameterOrder"),
                    F.lit(mode).alias("Mode"),
                    "State",
                    F.when(F.col("PreviousState").isNull(), F.lit("initialization"))
                    .otherwise(F.lit("transition"))
                    .alias("EventType"),
                    "RawValues",
                )
                conflicts = grouped.where(F.size("States") > 1).select(
                    "Timestamp",
                    "StartDatetime",
                    "ESN",
                    F.lit(channel).alias("Channel"),
                    F.lit(parameter).alias("Parameter"),
                    F.lit(parameter_index).cast("integer").alias("ParameterOrder"),
                    F.lit(mode).alias("Mode"),
                    F.lit(None).cast("string").alias("State"),
                    F.lit("conflict").alias("EventType"),
                    "RawValues",
                )
                event_frames.extend([events, conflicts])
        event_columns = [
            "Timestamp",
            "StartDatetime",
            "ESN",
            "Channel",
            "Parameter",
            "ParameterOrder",
            "Mode",
            "State",
            "EventType",
            "RawValues",
        ]
        if event_frames:
            all_events = event_frames[0]
            for frame in event_frames[1:]:
                all_events = all_events.unionByName(frame)
            event_pdf = all_events.orderBy(
                "Timestamp", "ESN", "Channel", "ParameterOrder"
            ).toPandas()
            if not event_pdf.empty:
                event_pdf["TimestampMs"] = pd.to_datetime(event_pdf["Timestamp"]).dt.floor("ms")
        else:
            event_pdf = pd.DataFrame(columns=event_columns + ["TimestampMs"])
        event_counts = {}
        if not event_pdf.empty:
            grouped_counts = event_pdf.groupby(
                ["ESN", "Parameter", "Channel", "EventType"], dropna=False
            ).size()
            for key, count in grouped_counts.items():
                esn, parameter, channel, event_type = key
                event_counts[(str(esn), str(parameter), str(channel), str(event_type))] = int(count)
        coverage_records = []
        warnings = []
        for esn in esns:
            for channel in CHANNELS:
                for parameter, specification in TRU_TRANSITION_PARAMETERS.items():
                    column_name = TRU_TRANSITION_COLUMNS[parameter][channel]
                    base = coverage_lookup.get(
                        (esn, parameter, channel),
                        {
                            "Rows": 0,
                            "Non-null": 0,
                            "Accepted": 0,
                            "Ignored": 0,
                            "Invalid": 0,
                            "First sample": "",
                            "Last sample": "",
                        },
                    )
                    record = {
                        "ESN": esn,
                        "Position": positions.get(esn) or "",
                        "Channel": channel,
                        "Parameter": parameter,
                        "Source type": schema_types.get(column_name, "missing"),
                        **base,
                        "Initializations": event_counts.get(
                            (esn, parameter, channel, "initialization"), 0
                        ),
                        "Transitions": event_counts.get(
                            (esn, parameter, channel, "transition"), 0
                        ),
                        "Conflicts": event_counts.get(
                            (esn, parameter, channel, "conflict"), 0
                        ),
                    }
                    coverage_records.append(record)
                    if (parameter, channel) not in available_columns:
                        warnings.append(f"ESN {esn} {channel} {parameter}: parameter missing from schema")
                    elif record["Non-null"] == 0:
                        warnings.append(f"ESN {esn} {channel} {parameter}: no non-null samples")
                    elif specification["mode"] == "endpoint" and record["Accepted"] == 0:
                        warnings.append(
                            f"ESN {esn} {channel} {parameter}: no values reached ≤"
                            f"{TRU_ENDPOINT_LOW_MAX:g} or ≥{TRU_ENDPOINT_HIGH_MIN:g}"
                        )
                    if record["Invalid"]:
                        warnings.append(
                            f"ESN {esn} {channel} {parameter}: {record['Invalid']:,} invalid values"
                        )
                    if record["Conflicts"]:
                        warnings.append(
                            f"ESN {esn} {channel} {parameter}: {record['Conflicts']:,} "
                            "timestamps contain conflicting values"
                        )
        for warning in warnings:
            print("WARNING: " + warning)
        ipy_display(HTML(_render_coverage_table(coverage_records)))
        if event_pdf.empty:
            print("WARNING: No initialization or transition events were found for the selected day.")
        else:
            ipy_display(HTML(_render_transition_table(event_pdf.to_dict("records"), esns, positions)))
            print(f"Transition-ledger milliseconds: {event_pdf['TimestampMs'].nunique():,}")
            print(f"Transition events: {len(event_pdf):,}")
        globals().update(
            {
                "transition_event_df": event_pdf,
                "transition_coverage_df": pd.DataFrame(coverage_records),
                "transition_esns": esns,
                "transition_master_df": master_df,
                "transition_windows_df": windows_df,
            }
        )
    except Exception as exc:
        print(f"Unable to build the transition table: {exc}")
        if raise_errors:
            raise
